function extractTextFromParts(parts: unknown): { content: string; thought: string } {
  if (!Array.isArray(parts)) return { content: '', thought: '' };
  const textParts: string[] = [];
  const thoughtParts: string[] = [];

  for (const part of parts) {
    if (!part || typeof part !== 'object') continue;
    const p = part as { text?: unknown; thought?: unknown; isThought?: unknown };
    if (typeof p.text !== 'string' || !p.text) continue;
    if (p.thought || p.isThought) {
      thoughtParts.push(p.text);
    } else {
      textParts.push(p.text);
    }
  }
  return {
    content: textParts.join(''),
    thought: thoughtParts.join(''),
  };
}

function cleanModelName(raw: unknown): string | undefined {
  if (typeof raw !== 'string' || !raw.trim()) return undefined;
  return raw.trim().replace(/^models\//i, '');
}

export function condenseLongCodeBlocks(text: string, maxLines = 80): string {
  return text.replace(/```([\w-]*)\n([\s\S]*?)```/g, (full, lang: string, body: string) => {
    const lines = body.split('\n');
    if (lines.length <= maxLines) return full;
    const preview = lines.slice(0, 12).join('\n');
    return `\`\`\`${lang}\n${preview}\n// ... [${lines.length - 12} baris kode ${lang || 'text'} diringkas]\n\`\`\``;
  });
}

// Pertahankan N blok kode besar terakhir secara utuh (zero-loss untuk kode terbaru & seluruh teks chat)
export function applyKeepLatestCodeToMessages(
  messages: ChatMessage[],
  keepLatestBlocks = 2
): ChatMessage[] {
  const codeBlockLocations: Array<{ msgIdx: number; matchIndex: number }> = [];
  const regex = /```[\w-]*\n[\s\S]*?```/g;

  messages.forEach((msg, msgIdx) => {
    let m: RegExpExecArray | null;
    regex.lastIndex = 0;
    while ((m = regex.exec(msg.content)) !== null) {
      if (m[0].split('\n').length > 25) {
        codeBlockLocations.push({ msgIdx, matchIndex: m.index });
      }
    }
  });

  if (codeBlockLocations.length <= keepLatestBlocks) {
    return messages;
  }

  const preserveSet = new Set(
    codeBlockLocations.slice(-keepLatestBlocks).map((loc) => `${loc.msgIdx}:${loc.matchIndex}`)
  );

  return messages.map((msg, msgIdx) => {
    let replaced = false;
    const nextContent = msg.content.replace(
      /```([\w-]*)\n([\s\S]*?)```/g,
      (full, lang: string, body: string, offset: number) => {
        const lineCount = body.split('\n').length;
        if (lineCount <= 25 || preserveSet.has(`${msgIdx}:${offset}`)) {
          return full;
        }
        replaced = true;
        return `[Versi kode ${lang || 'text'} lama (${lineCount} baris) sudah diperbarui pada versi terbaru di akhir percakapan]`;
      }
    );
    return replaced ? { ...msg, content: nextContent } : msg;
  });
}

export function estimateMessagesTokenTotal(messages: ChatMessage[]): number {
  return messages.reduce((acc, m) => {
    const attTokens = (m.attachments || []).reduce((s, a) => s + (a.tokenCount ?? 0), 0);
    const vtTokens = m.visualTarget ? estimateTokens(m.visualTarget.outerHtml) : 0;
    return acc + estimateTokens(m.content) + attTokens + vtTokens;
  }, 0);
}

export function buildEffectiveApiHistory(
  messages: ChatMessage[],
  compacted?: CompactedContextState
): ChatMessage[] {
  if (!compacted) return messages;

  if (compacted.mode === 'keep-latest-code') {
    return applyKeepLatestCodeToMessages(messages, 2);
  }

  const keepCount = Math.max(2, compacted.keepRecentCount);
  const recentRaw = messages.slice(-keepCount);
  const recentClean = applyKeepLatestCodeToMessages(recentRaw, 2);

  if (!compacted.summaryPrefix) {
    return recentClean;
  }

  const prefixMessages: ChatMessage[] = [
    {
      id: 'compacted-history-user',
      role: 'user',
      content: `[Ringkasan Konteks Historis Lengkap dari Sesi Sebelumnya]\n\n${compacted.summaryPrefix}`,
      createdAt: messages[0]?.createdAt ?? Date.now(),
    },
    {
      id: 'compacted-history-assistant',
      role: 'assistant',
      content:
        'Konteks historis lengkap dan spesifikasi proyek di atas sudah saya pahami sepenuhnya. Saya siap melanjutkan dari percakapan terbaru di bawah.',
      createdAt: messages[0]?.createdAt ?? Date.now(),
    },
  ];

  return [...prefixMessages, ...recentClean];
}

export function buildCompactionChunks(messages: ChatMessage[], maxChunkTokens = 105_000): string[] {
  const deduped = applyKeepLatestCodeToMessages(messages, 1);
  const chunks: string[] = [];
  let currentText = '';
  let currentTokens = 0;

  for (let i = 0; i < deduped.length; i++) {
    const m = deduped[i];
    const speaker = m.role === 'user' ? 'USER' : 'ASSISTANT';
    const entry = `[Turn #${i + 1} - ${speaker}]\n${m.content}\n\n`;
    const entryTokens = estimateTokens(entry);

    if (currentTokens + entryTokens > maxChunkTokens && currentText) {
      chunks.push(currentText.trim());
      currentText = entry;
      currentTokens = entryTokens;
    } else {
      currentText += entry;
      currentTokens += entryTokens;
    }
  }

  if (currentText.trim()) {
    chunks.push(currentText.trim());
  }
  return chunks;
}

export async function runLosslessContextCompaction(params: {
  messages: ChatMessage[];
  keepRecentCount: number;
  model: string;
  apiKey?: string;
  onStatus?: (msg: string) => void;
}): Promise<{ ok: boolean; summary?: string; error?: string }> {
  const olderMessages =
    params.messages.length > params.keepRecentCount
      ? params.messages.slice(0, -params.keepRecentCount)
      : params.messages;

  const chunks = buildCompactionChunks(olderMessages, 105_000);
  if (chunks.length === 0) {
    return { ok: false, error: 'Tidak ada pesan lama untuk dirangkum.' };
  }

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
  };
  const normalizedKeys = (params.apiKey || '')
    .split(/[\r\n,;]+/)
    .map((k) => k.trim())
    .filter(Boolean);
  if (normalizedKeys.length > 0) {
    headers['x-gemini-api-key'] = normalizedKeys.join(',');
  }

  const res = await fetch('/api/compact', {
    method: 'POST',
    headers,
    body: JSON.stringify({
      model: params.model,
      chunks,
    }),
  });

  if (!res.ok || !res.body) {
    const txt = await res.text().catch(() => '');
    return { ok: false, error: txt || `Gagal merangkum konteks (${res.status})` };
  }

  const reader = res.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';
  let finalSummary = '';
  let errorMsg = '';

  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');

    let boundaryIndex: number;
    while ((boundaryIndex = buffer.indexOf('\n\n')) !== -1) {
      const rawBlock = buffer.slice(0, boundaryIndex);
      buffer = buffer.slice(boundaryIndex + 2);
      if (!rawBlock.trim() || rawBlock.startsWith(':')) continue;

      let eventName = 'message';
      const dataLines: string[] = [];
      for (const line of rawBlock.split('\n')) {
        if (line.startsWith('event:')) eventName = line.slice(6).trim();
        else if (line.startsWith('data:')) dataLines.push(line.slice(5).trimStart());
      }
      if (dataLines.length === 0) continue;

      let parsed: Record<string, unknown>;
      try {
        parsed = JSON.parse(dataLines.join('\n'));
      } catch {
        continue;
      }

      if (eventName === 'status' && typeof parsed.message === 'string') {
        params.onStatus?.(parsed.message);
      } else if (eventName === 'result' && typeof parsed.summary === 'string') {
        finalSummary = parsed.summary;
      } else if (eventName === 'error' && typeof parsed.message === 'string') {
        errorMsg = parsed.message;
      }
    }
  }

  if (finalSummary) return { ok: true, summary: finalSummary };
  return { ok: false, error: errorMsg || 'Ringkasan gagal dihasilkan.' };
}

function parseGoogleAiStudioJson(
  obj: Record<string, unknown>,
  fallbackTitle: string
): ParsedImportResult | null {
  const chunked = obj.chunkedPrompt as { chunks?: unknown[] } | undefined;
  if (!chunked || !Array.isArray(chunked.chunks)) return null;

  const messages: ChatMessage[] = [];
  let pendingThought = '';
  const now = Date.now();

  for (let i = 0; i < chunked.chunks.length; i++) {
    const rawChunk = chunked.chunks[i];
    if (!rawChunk || typeof rawChunk !== 'object') continue;
    const c = rawChunk as {
      role?: string;
      text?: string;
      isThought?: boolean;
      parts?: unknown[];
    };

    const roleRaw = (c.role || 'user').toLowerCase();
    const role: 'user' | 'assistant' =
      roleRaw === 'model' || roleRaw === 'assistant' ? 'assistant' : 'user';

    let text = typeof c.text === 'string' ? c.text : '';
    let thought = '';

    if (Array.isArray(c.parts)) {
      const extracted = extractTextFromParts(c.parts);
      if (!text) text = extracted.content;
      thought = extracted.thought;
    }

    if (c.isThought && role === 'assistant') {
      pendingThought += (pendingThought ? '\n' : '') + text;
      continue;
    }

    const combinedThought = [pendingThought, thought].filter(Boolean).join('\n').trim();
    pendingThought = '';

    if (!text.trim() && !combinedThought) continue;

    const prev = messages[messages.length - 1];
    if (prev && prev.role === role && role === 'assistant' && !prev.content && combinedThought) {
      prev.content = text;
      prev.thought = [prev.thought, combinedThought].filter(Boolean).join('\n');
      continue;
    }

    messages.push({
      id: generateId(),
      role,
      content: text.trim(),
      thought: combinedThought || undefined,
      createdAt: now + i,
    });
  }

  if (pendingThought && messages.length > 0 && messages[messages.length - 1].role === 'assistant') {
    const last = messages[messages.length - 1];
    last.thought = [last.thought, pendingThought].filter(Boolean).join('\n');
  }

  if (messages.length === 0) return null;

  const runSettings = obj.runSettings as { model?: unknown } | undefined;
  const sysObj = obj.systemInstruction as { text?: unknown; parts?: unknown[] } | undefined;
  let sysText = typeof sysObj?.text === 'string' ? sysObj.text : '';
  if (!sysText && Array.isArray(sysObj?.parts)) {
    sysText = extractTextFromParts(sysObj.parts).content;
  }

  const totalTok = estimateMessagesTokenTotal(messages);

  return {
    ok: true,
    sourceLabel: 'Google AI Studio Drive JSON',
    title: typeof obj.title === 'string' && obj.title.trim() ? obj.title.trim() : fallbackTitle,
    model: cleanModelName(runSettings?.model),
    sessionSystemInstruction: sysText.trim() || undefined,
    messages,
    estimatedTokens: totalTok,
  };
}

function parseChatGptMappingJson(
  obj: Record<string, unknown>,
  fallbackTitle: string
): ParsedImportResult | null {
  const mapping = obj.mapping as Record<string, unknown> | undefined;
  if (!mapping || typeof mapping !== 'object') return null;

  const nodes = Object.values(mapping).filter(
    (n): n is {
      message?: {
        author?: { role?: string };
        content?: { parts?: unknown[] };
        create_time?: number;
      };
    } => Boolean(n && typeof n === 'object')
  );

  nodes.sort((a, b) => (a.message?.create_time ?? 0) - (b.message?.create_time ?? 0));

  const messages: ChatMessage[] = [];
  for (const node of nodes) {
    const msg = node.message;
    if (!msg) continue;
    const roleRaw = msg.author?.role;
    if (roleRaw !== 'user' && roleRaw !== 'assistant') continue;
    const parts = Array.isArray(msg.content?.parts) ? msg.content.parts : [];
    const text = parts.filter((p): p is string => typeof p === 'string').join('\n').trim();
    if (!text) continue;
    messages.push({
      id: generateId(),
      role: roleRaw,
      content: text,
      createdAt: msg.create_time ? Math.round(msg.create_time * 1000) : Date.now(),
    });
  }

  if (messages.length === 0) return null;

  return {
    ok: true,
    sourceLabel: 'ChatGPT Export JSON',
    title: typeof obj.title === 'string' && obj.title.trim() ? obj.title.trim() : fallbackTitle,
    messages,
    estimatedTokens: estimateMessagesTokenTotal(messages),
  };
}

function parseGenericMessagesArray(
  arr: unknown[],
  sourceLabel: string,
  title: string,
  model?: string,
  sessionSystemInstruction?: string
): ParsedImportResult | null {
  const messages: ChatMessage[] = [];
  const now = Date.now();

  for (let i = 0; i < arr.length; i++) {
    const item = arr[i];
    if (!item || typeof item !== 'object') continue;
    const m = item as {
      role?: string;
      content?: unknown;
      text?: unknown;
      thought?: unknown;
      parts?: unknown[];
      model?: unknown;
    };

    const roleRaw = (m.role || '').toLowerCase();
    if (roleRaw === 'system') continue;
    const role: 'user' | 'assistant' =
      roleRaw === 'assistant' || roleRaw === 'model' || roleRaw === 'ai' ? 'assistant' : 'user';

    let content = '';
    let thought = typeof m.thought === 'string' ? m.thought : '';

    if (typeof m.content === 'string') {
      content = m.content;
    } else if (typeof m.text === 'string') {
      content = m.text;
    } else if (Array.isArray(m.parts)) {
      const extracted = extractTextFromParts(m.parts);
      content = extracted.content;
      if (!thought) thought = extracted.thought;
    }

    if (!content.trim() && !thought.trim()) continue;
    messages.push({
      id: generateId(),
      role,
      content: content.trim(),
      thought: thought.trim() || undefined,
      model: cleanModelName(m.model),
      createdAt: now + i,
    });
  }

  if (messages.length === 0) return null;

  return {
    ok: true,
    sourceLabel,
    title,
    model,
    sessionSystemInstruction,
    messages,
    estimatedTokens: estimateMessagesTokenTotal(messages),
  };
}

function parseMarkdownChat(raw: string, fallbackTitle: string): ParsedImportResult {
  const text = raw.replace(/\r\n/g, '\n').trim();
  const titleMatch = /^#\s+(.+)$/m.exec(text);
  const extractedTitle = titleMatch ? titleMatch[1].trim() : fallbackTitle;

  const roleHeaderRegex =
    /^(?:#{1,4}\s*|\*\*)(User|Pengguna|You|Assistant|Model|AI|Gemini|Claude)(?:\*\*)?:?\s*$/gim;

  const matches = Array.from(text.matchAll(roleHeaderRegex));
  const messages: ChatMessage[] = [];
  const now = Date.now();

  if (matches.length >= 1) {
    for (let i = 0; i < matches.length; i++) {
      const curr = matches[i];
      const next = matches[i + 1];
      const start = (curr.index ?? 0) + curr[0].length;
      const end = next ? (next.index ?? text.length) : text.length;
      const body = text.slice(start, end).trim();
      if (!body) continue;

      const label = curr[1].toLowerCase();
      const role: 'user' | 'assistant' =
        label === 'user' || label === 'pengguna' || label === 'you' ? 'user' : 'assistant';

      messages.push({
        id: generateId(),
        role,
        content: body,
        createdAt: now + i,
      });
    }
  }

  if (messages.length === 0) {
    messages.push({
      id: generateId(),
      role: 'user',
      content: `[Konteks Referensi dari Dokumen Markdown]\n\n${text}`,
      createdAt: now,
    });
    messages.push({
      id: generateId(),
      role: 'assistant',
      content:
        'Siap! Konteks dokumen di atas sudah saya pahami untuk sesi ini. Silakan lanjutkan pertanyaan atau instruksi berikutnya.',
      createdAt: now + 1,
    });
  }

  return {
    ok: true,
    sourceLabel: matches.length >= 1 ? 'Markdown Chat (.md)' : 'Markdown Context (.md)',
    title: extractedTitle.slice(0, 64),
    messages,
    estimatedTokens: estimateMessagesTokenTotal(messages),
  };
}

export function parseImportedChatContent(
  rawText: string,
  fileName = 'Imported Chat'
): ParsedImportResult {
  const trimmed = rawText.trim();
  const baseTitle = fileName.replace(/\.(json|md|txt)$/i, '') || 'Imported Chat';

  if (!trimmed) {
    return { ok: false, messages: [], estimatedTokens: 0, error: 'Konten masih kosong.' };
  }

  if (trimmed.startsWith('{') || trimmed.startsWith('[')) {
    try {
      const parsed = JSON.parse(trimmed) as unknown;

      if (Array.isArray(parsed)) {
        const res = parseGenericMessagesArray(parsed, 'Gemini / Chat Array JSON', baseTitle);
        if (res) return res;
      } else if (parsed && typeof parsed === 'object') {
        const obj = parsed as Record<string, unknown>;

        const aiStudioRes = parseGoogleAiStudioJson(obj, baseTitle);
        if (aiStudioRes) return aiStudioRes;

        if (Array.isArray(obj.messages)) {
          const res = parseGenericMessagesArray(
            obj.messages,
            'Adam Studio / Chat JSON',
            typeof obj.title === 'string' && obj.title.trim() ? obj.title.trim() : baseTitle,
            cleanModelName(obj.model),
            typeof obj.sessionSystemInstruction === 'string'
              ? obj.sessionSystemInstruction
              : undefined
          );
          if (res) return res;
        }

        if (Array.isArray(obj.contents)) {
          const res = parseGenericMessagesArray(
            obj.contents,
            'Gemini API Contents JSON',
            baseTitle,
            cleanModelName(obj.model)
          );
          if (res) return res;
        }

        const gptRes = parseChatGptMappingJson(obj, baseTitle);
        if (gptRes) return gptRes;
      }

      return {
        ok: false,
        messages: [],
        estimatedTokens: 0,
        error:
          'Struktur JSON tidak dikenali. Pastikan berasal dari Google AI Studio (Drive JSON), Adam Studio Export, ChatGPT, atau array { role, content }.',
      };
    } catch (err) {
      return {
        ok: false,
        messages: [],
        estimatedTokens: 0,
        error: `JSON tidak valid: ${err instanceof Error ? err.message : 'Syntax error'}`,
      };
    }
  }

  return parseMarkdownChat(trimmed, baseTitle);
}

let roundRobinCursor = Math.floor(Math.random() * 1000);

function parseKeyPool(...sources) {
  const pool = [];
  const seen = new Set();
  for (const src of sources) {
    if (!src || typeof src !== 'string') continue;
    for (const part of src.split(/[\r\n,;]+/)) {
      const k = part.trim();
      if (k && !seen.has(k)) {
        seen.add(k);
        pool.push(k);
      }
    }
  }
  return pool;
}

function extractGeminiError(rawText, status) {
  let msg = `Gemini API error (${status})`;
  try {
    const parsed = JSON.parse(rawText);
    if (parsed?.error?.message) msg = parsed.error.message;
  } catch {
    if (rawText) msg = rawText;
  }
  return msg;
}

async function callGeminiSummary({ model, apiKeys, prompt, signal }) {
  const candidates = Array.from(
    new Set(['gemini-2.5-flash', model || 'gemini-4-hydrogen', 'gemini-2.0-flash'].filter(Boolean))
  );

  const startOffset = roundRobinCursor % apiKeys.length;
  roundRobinCursor = (roundRobinCursor + 1) % apiKeys.length;

  let lastError = 'Gagal meringkas konteks.';

  for (const targetModel of candidates) {
    const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(
      targetModel
    )}:generateContent`;

    for (let i = 0; i < apiKeys.length; i++) {
      const key = apiKeys[(startOffset + i) % apiKeys.length];
      let useZeroThinking = targetModel.includes('2.5');
      let triedNoThinking = false;

      while (true) {
        try {
          const generationConfig = {
            temperature: 0.2,
            maxOutputTokens: 4096,
          };
          if (useZeroThinking) {
            generationConfig.thinkingConfig = { thinkingBudget: 0 };
          }

          const res = await fetch(endpoint, {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
              'x-goog-api-key': key,
            },
            body: JSON.stringify({
              contents: [{ role: 'user', parts: [{ text: prompt }] }],
              generationConfig,
            }),
            signal,
          });

          if (res.ok) {
            const json = await res.json();
            const text = json?.candidates?.[0]?.content?.parts
              ?.filter((p) => !p?.thought)
              ?.map((p) => p?.text || '')
              .join('')
              .trim();
            if (text) return { ok: true, text };
          } else {
            const errRaw = await res.text();
            lastError = extractGeminiError(errRaw, res.status);
            if (res.status === 400 && useZeroThinking && !triedNoThinking && /thinking/i.test(errRaw)) {
              useZeroThinking = false;
              triedNoThinking = true;
              continue;
            }
            if (res.status === 404) break;
          }
        } catch (err) {
          if (signal?.aborted) throw err;
          lastError = err instanceof Error ? err.message : 'Network error';
        }
        break;
      }
    }
  }

  return { ok: false, error: lastError };
}

function parseKeyPool(...sources) {
  const pool = [];
  const seen = new Set();
  for (const src of sources) {
    if (!src || typeof src !== 'string') continue;
    for (const part of src.split(/[\r\n,;]+/)) {
      const k = part.trim();
      if (k && !seen.has(k)) {
        seen.add(k);
        pool.push(k);
      }
    }
  }
  return pool;
}

async function fetchWithTimeout(url, options, parentSignal, timeoutMs = 10000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  const onParentAbort = () => controller.abort();
  parentSignal?.addEventListener('abort', onParentAbort, { once: true });

  try {
    return await fetch(url, { ...options, signal: controller.signal });
  } finally {
    clearTimeout(timer);
    parentSignal?.removeEventListener('abort', onParentAbort);
  }
}

function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) {
      reject(new DOMException('Aborted', 'AbortError'));
      return;
    }
    const timer = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();
    }, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

function buildGeminiContents(messages) {
  return messages.map((msg) => {
    const parts = [];
    if (msg.content && msg.content.trim().length > 0) {
      parts.push({ text: msg.content });
    }
    if (Array.isArray(msg.attachments)) {
      for (const att of msg.attachments) {
        parts.push(...buildAttachmentParts(att));
      }
    }
    if (parts.length === 0) {

    };
  });
}

function buildGeminiPayload({

function extractGeminiError(rawText, status) {
  let msg = `Gemini API error (${status})`;
  try {
    const parsed = JSON.parse(rawText);
    if (parsed?.error?.message) msg = parsed.error.message;

    if (rawText) msg = rawText;
  }
  return msg;
}

function buildOrderedCandidateKeys(keys) {
  const now = Date.now();
  const start = roundRobinCursor % keys.length;
  roundRobinCursor = (roundRobinCursor + 1) % keys.length;

  const rotated = [...keys.slice(start), ...keys.slice(0, start)];
  const ready = [];
  const cooling = [];
  for (const item of rotated) {
    const origIdx = keys.indexOf(item);
    const coolUntil = keyCooldownMap.get(item) || 0;
    if (coolUntil > now) {
      cooling.push({ key: item, idx: origIdx });
    } else {
      ready.push({ key: item, idx: origIdx });
    }
  }
  return [...ready, ...cooling];
}

        })
      ),
      signal,
    });

    if (upstream.ok) {
      if (attempt > 0 && orderedKeys.length > 1) {
        onInfo(`Berhasil terhubung menggunakan ${keyLabel} (pool ${orderedKeys.length} keys)`);
      }
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    const errBody = await upstream.text();
    lastStatus = upstream.status;
    lastErrorMessage = extractGeminiError(errBody, upstream.status);

    if (upstream.status === 400 && wantsThinking && /thinking/i.test(errBody)) {
      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    const isBadKey =
      upstream.status === 401 ||
      upstream.status === 403 ||
      (upstream.status === 400 && /api.?key|expired|invalid/i.test(lastErrorMessage));

    const canRotateToAnotherKey = orderedKeys.length > 1;
    const isRetryable = RETRYABLE_STATUSES.has(upstream.status) || (isBadKey && canRotateToAnotherKey);

    if (isRetryable && attempt + 1 < maxAttempts) {
      const coolMs = isBadKey ? 120_000 : upstream.status === 429 ? 60_000 : 12_000;
      keyCooldownMap.set(current.key, Date.now() + coolMs);

      if (canRotateToAnotherKey) {
        keyPointer += 1;
        const nextCandidate = orderedKeys[keyPointer % orderedKeys.length];
        const nextLabel = maskKeyLabel(nextCandidate.key, nextCandidate.idx);
        const fastRotateDelay = ROTATE_IMMEDIATELY_STATUSES.has(upstream.status) || isBadKey ? 200 : 450;

        onStatus(
          `${keyLabel} terkena (${upstream.status}). Merotasi ke ${nextLabel} (${attempt + 2}/${maxAttempts})...`
        );
        await sleep(fastRotateDelay, signal);
        continue;
      }

      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${maxAttempts - 1}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: lastStatus,
      message: lastErrorMessage,
    };
  }

  return { ok: false, status: lastStatus, message: lastErrorMessage };
}

      messages,
      systemInstruction,
      temperature,
      maxOutputTokens,
      includeThoughts,
      thinkingBudget,

      fallbackModel &&

      writeSseEvent(res, 'error', {
        status: result.status,
        message: result.message,
      });
      return;
    }

    const upstream = result.response;
    const activeModel = result.usedModel;

    if (!upstream.body) {
      writeSseEvent(res, 'error', {
        status: 502,
        message: 'Stream respons dari Gemini API kosong.',
      });
      return;
    }

    const decoder = new TextDecoder();
    let buffer = '';
    let lastFinishReason = undefined;

    while (true) {
      const { value, done } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');

      let boundaryIndex;
      while ((boundaryIndex = buffer.indexOf('\n\n')) !== -1) {
        const rawEvent = buffer.slice(0, boundaryIndex);
        buffer = buffer.slice(boundaryIndex + 2);

        const dataLines = rawEvent
          .split('\n')
          .filter((line) => line.startsWith('data:'))
          .map((line) => line.slice(5).trimStart());

        if (dataLines.length === 0) continue;
        const jsonString = dataLines.join('\n').trim();
        if (!jsonString || jsonString === '[DONE]') continue;

        try {
          const chunk = JSON.parse(jsonString);
          const candidate = chunk?.candidates?.[0];
          const parts = candidate?.content?.parts;

          if (Array.isArray(parts)) {
            for (const part of parts) {
              if (part?.thought && typeof part?.text === 'string' && part.text.length > 0) {
                writeSseEvent(res, 'thought', { text: part.text });
              } else if (typeof part?.text === 'string' && part.text.length > 0) {
                writeSseEvent(res, 'delta', { text: part.text });
              }
            }
          }

          if (candidate?.finishReason) lastFinishReason = candidate.finishReason;
          if (chunk?.usageMetadata) writeSseEvent(res, 'usage', chunk.usageMetadata);
        } catch {
          // Abaikan JSON parsial
        }
      }
    }

    writeSseEvent(res, 'done', {
      model: activeModel,
      finishReason: lastFinishReason || 'STOP',
      isDemo: false,
    });
  } catch (err) {
    if (err instanceof Error && err.name === 'AbortError') return;
    writeSseEvent(res, 'error', {
      status: 500,
      message: err instanceof Error ? err.message : 'Terjadi kesalahan internal server.',
    });
  } finally {
    clearInterval(keepAlive);
    if (!res.writableEnded) {
      res.end();
    }
  }
});

function getDateBucket(timestamp: number): 'Hari Ini' | 'Kemarin' | 'Sebelumnya' {
  const now = new Date();
  const startOfToday = new Date(now.getFullYear(), now.getMonth(), now.getDate()).getTime();
  const startOfYesterday = startOfToday - 86_400_000;
  if (timestamp >= startOfToday) return 'Hari Ini';
  if (timestamp >= startOfYesterday) return 'Kemarin';
  return 'Sebelumnya';
}

function formatRelativeShort(ts: number): string {
  const diffMin = Math.floor((Date.now() - ts) / 60_000);
  if (diffMin < 1) return 'baru saja';
  if (diffMin < 60) return `${diffMin}m`;
  const diffHr = Math.floor(diffMin / 60);
  if (diffHr < 24) return `${diffHr}j`;
  const diffDay = Math.floor(diffHr / 24);
  if (diffDay === 1) return 'kemarin';
  if (diffDay < 7) return `${diffDay}h`;
  return new Date(ts).toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
}

function shortModelTag(modelId: string): string {
  if (modelId === 'gemini-4-hydrogen') return 'H4';
  if (modelId === 'gemini-2.5-pro') return '2.5P';
  if (modelId === 'gemini-2.5-flash') return '2.5F';
  if (modelId === 'gemini-2.0-flash') return '2.0F';
  return modelId.replace(/^gemini-/i, '').slice(0, 8);
}

    if (!parsed?.ok) return 0;
    if (!keepLatestCodeForApi) return parsed.estimatedTokens;

    return estimateMessagesTokenTotal(deduped);
  }, [parsed, keepLatestCodeForApi]);

  const handleReadFile = async (file: File) => {
    const text = await file.text();
    setFileName(file.name);

  const handleFileChange = async (e: ChangeEvent<HTMLInputElement>) => {

  };

  const handleDrop = async (e: DragEvent<HTMLDivElement>) => {
    e.preventDefault();

    const file = e.dataTransfer.files?.[0];

  };

  const resetForm = () => {
    setRawInput('');
    setFileName('');
    setCustomTitle('');
    setKeepLatestCodeForApi(true);
    setMode('new');
  };

  const handleConfirm = () => {

export default function App() {
  const [conversations, setConversations] = useState<Conversation[]>(() => loadConversations());

    const params = new URLSearchParams(window.location.search);
    if (params.get('action') === 'new') return null;
    const existing = loadConversations();

  });
  const [selectedModel, setSelectedModel] = useState<string>(DEFAULT_MODEL_ID);
  const [settings, setSettings] = useState<AppSettings>(() => loadSettings());
  const [serverHealth, setServerHealth] = useState<ServerHealth | null>(null);

  const [sidebarOpen, setSidebarOpen] = useState(true);
  const [mobileSidebarOpen, setMobileSidebarOpen] = useState(false);
  const [zenMode, setZenMode] = useState(false);

  const [modelDialogOpen, setModelDialogOpen] = useState(false);
  const [settingsDialogOpen, setSettingsDialogOpen] = useState(false);
  const [commandOpen, setCommandOpen] = useState(false);
  const [bookmarksOpen, setBookmarksOpen] = useState(false);
  const [importDialogOpen, setImportDialogOpen] = useState(false);
  const [contextDialogOpen, setContextDialogOpen] = useState(false);
  const [installDialogOpen, setInstallDialogOpen] = useState(false);
  const [lightboxUrl, setLightboxUrl] = useState<string | null>(null);

  const [activeArtifact, setActiveArtifact] = useState<ArtifactItem | null>(null);
  const [artifactFullscreen, setArtifactFullscreen] = useState(false);
  const [artifactWidthPct, setArtifactWidthPct] = useState(46);
  const [isResizingPane, setIsResizingPane] = useState(false);

  const [welcomeTab, setWelcomeTab] = useState<WelcomeTab>('All');
  const [externalDraft, setExternalDraft] = useState<string | null>(null);
  const [visualTarget, setVisualTarget] = useState<VisualTarget | null>(null);
  const [activeStream, setActiveStream] = useState<ActiveStreamState | null>(null);
  const [showScrollBottom, setShowScrollBottom] = useState(false);
  const [scrollViewport, setScrollViewport] = useState<HTMLElement | null>(null);

  const { isStandalone, isIos, isOnline, canNativeInstall, triggerNativeInstall } = usePwaInstall();

  const {
    needRefresh: [needRefresh],
    updateServiceWorker,
  } = useRegisterSW();

  useEffect(() => {
    if (needRefresh) {
      toast('Pembaruan Adam Studio tersedia', {
        description: 'Muat ulang untuk mendapatkan versi terbaru.',

        action: {
          label: 'Update',
          onClick: () => {
            void updateServiceWorker(true);
          },
        },
      });
    }
  }, [needRefresh, updateServiceWorker]);

  const handleInstallAction = useCallback(async () => {
    if (canNativeInstall) {
      const result = await triggerNativeInstall();
      if (result === 'installed') {
        toast.success('Adam Studio berhasil diinstal!');
        return;
      }
      if (result === 'dismissed') return;
    }
    setInstallDialogOpen(true);
  }, [canNativeInstall, triggerNativeInstall]);

  const abortControllerRef = useRef<AbortController | null>(null);
  const activeStreamRef = useRef<ActiveStreamState | null>(null);
  const chatScrollRef = useRef<StudioScrollHandle | null>(null);
  const autoScrollEnabledRef = useRef(true);
  const flushTimerRef = useRef<number | null>(null);

  useEffect(() => {
    activeStreamRef.current = activeStream;
  }, [activeStream]);

  useEffect(() => {
    saveConversations(conversations);
  }, [conversations]);

  useEffect(() => {
    saveSettings(settings);
    document.documentElement.setAttribute('data-skin', settings.skin);
    if (settings.skin === 'paper') {
      document.documentElement.classList.remove('dark');
    } else {
      document.documentElement.classList.add('dark');
    }
  }, [settings]);

  useEffect(() => {
    fetch('/api/health')
      .then((res) => (res.ok ? res.json() : null))
      .then((data: ServerHealth | null) => {
        if (data) setServerHealth(data);
      })
      .catch(() => {});
  }, []);

  const activeConversation = useMemo(
    () => conversations.find((c) => c.id === activeId) ?? null,
    [conversations, activeId]
  );

  const messageList = activeConversation?.messages ?? [];
  const currentModelId = activeConversation?.model ?? selectedModel;
  const currentModelInfo = getModelInfo(currentModelId);
  const isStreamingActive = activeStream !== null;
  const thinkingEnabled = settings.includeThoughts && settings.thinkingBudget !== 0;

  const activeKeyPoolCount =
    countApiKeys(settings.apiKey) ||
    serverHealth?.serverKeyCount ||
    (serverHealth?.hasServerApiKey ? 1 : 0);

  const virtualizer = useVirtualizer({
    count: messageList.length,
    getScrollElement: () => scrollViewport,
    estimateSize: () => 180,
    overscan: 5,
    getItemKey: (index) => messageList[index]?.id ?? index,
  });

  const conversationArtifacts = useMemo(() => {
    if (!activeConversation) return [];
    const list: ArtifactItem[] = [];
    for (const msg of activeConversation.messages) {
      if (msg.role === 'assistant' && !msg.isStreaming && msg.content) {
        list.push(...extractArtifactsFromMarkdown(msg.id, msg.content));
      }
    }
    return list;
  }, [activeConversation]);

  const contextTokenEstimate = useMemo(() => {
    if (!activeConversation) return estimateTokens(settings.systemInstruction);
    const effective = buildEffectiveApiHistory(
      activeConversation.messages,
      activeConversation.compactedContext
    );
    return (
      estimateMessagesTokenTotal(effective) +
      estimateTokens(activeConversation.sessionSystemInstruction || settings.systemInstruction)
    );
  }, [activeConversation, settings.systemInstruction]);

  const contextUsagePct = Math.min(
    100,
    Math.max(1, Math.round((contextTokenEstimate / currentModelInfo.contextLimit) * 100))
  );

  const isContextOverFreeTier = contextTokenEstimate > 235_000;

  const scrollToBottom = useCallback(
    (smooth = false) => {
      const el = scrollViewport ?? chatScrollRef.current?.getViewport();
      if (!el) return;
      if (messageList.length > 0 && !smooth) {
        virtualizer.scrollToIndex(messageList.length - 1, { align: 'end' });
        requestAnimationFrame(() => {
          el.scrollTop = el.scrollHeight;
        });
      } else {
        el.scrollTo({
          top: el.scrollHeight,
          behavior: smooth ? 'smooth' : 'auto',
        });
      }
    },
    [messageList.length, scrollViewport, virtualizer]
  );

  useEffect(() => {
    if (!activeId || messageList.length === 0 || !scrollViewport) return;
    autoScrollEnabledRef.current = true;
    setShowScrollBottom(false);
    requestAnimationFrame(() => {
      virtualizer.scrollToIndex(messageList.length - 1, { align: 'end' });
      requestAnimationFrame(() => {
        scrollViewport.scrollTop = scrollViewport.scrollHeight;
      });
    });
  }, [activeId, scrollViewport]); // eslint-disable-line react-hooks/exhaustive-deps

  useEffect(() => {
    if (autoScrollEnabledRef.current && messageList.length > 0) {
      scrollToBottom(false);
    }
  }, [messageList.length, activeStream?.content, activeStream?.thought, scrollToBottom]);

  const handleViewportScroll = useCallback((el: HTMLElement) => {
    const distanceFromBottom = el.scrollHeight - el.scrollTop - el.clientHeight;
    const nearBottom = distanceFromBottom < 95;
    autoScrollEnabledRef.current = nearBottom;
    setShowScrollBottom(!nearBottom);
  }, []);

  const handleStopStream = useCallback(() => {
    abortControllerRef.current?.abort();
    abortControllerRef.current = null;
    if (flushTimerRef.current !== null) {
      window.clearTimeout(flushTimerRef.current);
      flushTimerRef.current = null;
    }

    const current = activeStreamRef.current;
    activeStreamRef.current = null;
    setActiveStream(null);

    if (current) {
      setConversations((prev) =>
        prev.map((conv) =>
          conv.id !== current.convId
            ? conv
            : {
                ...conv,
                updatedAt: Date.now(),
                messages: conv.messages.map((m) =>
                  m.id === current.msgId
                    ? {
                        ...m,
                        content: current.content,
                        thought: current.thought || m.thought,
                        statusText: undefined,
                        usage: current.usage ?? m.usage,
                        metrics: current.thinkingDurationMs
                          ? { ...m.metrics, thinkingDurationMs: current.thinkingDurationMs }
                          : m.metrics,
                        isStreaming: false,
                      }
                    : m
                ),
              }
        )
      );
    }
  }, []);

  const handleNewChat = useCallback(() => {
    if (isStreamingActive) handleStopStream();
    setActiveId(null);
    setActiveArtifact(null);
    setArtifactFullscreen(false);
    setVisualTarget(null);
  }, [handleStopStream, isStreamingActive]);

  const handleCycleSkin = useCallback(() => {
    setSettings((prev) => {
      const idx = SKIN_ORDER.indexOf(prev.skin);
      const nextSkin = SKIN_ORDER[(idx + 1) % SKIN_ORDER.length];
      toast.success(`Tema diganti ke ${nextSkin.toUpperCase()}`);
      return { ...prev, skin: nextSkin };
    });
  }, []);

  useEffect(() => {
    const onKeyDown = (e: globalThis.KeyboardEvent) => {
      const mod = e.metaKey || e.ctrlKey;
      if (mod && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        setCommandOpen((prev) => !prev);
      } else if (mod && e.key.toLowerCase() === 'b') {
        e.preventDefault();
        setSidebarOpen((prev) => !prev);
      } else if (mod && e.key === '\\') {
        e.preventDefault();
        setZenMode((prev) => !prev);
      } else if (mod && e.shiftKey && e.key.toLowerCase() === 'o') {
        e.preventDefault();
        handleNewChat();
      }
    };
    window.addEventListener('keydown', onKeyDown);
    return () => window.removeEventListener('keydown', onKeyDown);
  }, [handleNewChat]);

  const startResizingPane = (e: ReactPointerEvent<HTMLDivElement>) => {
    e.preventDefault();
    setIsResizingPane(true);

    const onMove = (moveEvent: PointerEvent) => {
      const widthPct = ((window.innerWidth - moveEvent.clientX) / window.innerWidth) * 100;
      setArtifactWidthPct(Math.min(66, Math.max(30, widthPct)));
    };

    const onUp = () => {
      setIsResizingPane(false);
      window.removeEventListener('pointermove', onMove);
      window.removeEventListener('pointerup', onUp);
    };

    window.addEventListener('pointermove', onMove);
    window.addEventListener('pointerup', onUp);
  };

  const runStreamForConversation = useCallback(
    async (
      convId: string,
      modelId: string,
      historyMessages: ChatMessageType[],
      assistantMessageId: string,
      sessionSystemInstruction?: string,
      compactedContext?: CompactedContextState
    ) => {
      abortControllerRef.current?.abort();
      const controller = new AbortController();
      abortControllerRef.current = controller;
      autoScrollEnabledRef.current = true;

      const effectiveHistory = buildEffectiveApiHistory(historyMessages, compactedContext);
      const trimmedHistory =
        settings.maxHistoryMessages > 0 && effectiveHistory.length > settings.maxHistoryMessages
          ? effectiveHistory.slice(-settings.maxHistoryMessages)
          : effectiveHistory;

      const effectiveSystemInstruction = [
        settings.systemInstruction,
        sessionSystemInstruction ? `[Konteks Sistem Sesi Impor]\n${sessionSystemInstruction}` : '',
      ]
        .filter(Boolean)
        .join('\n\n');

      const startedAt = performance.now();
      let firstThoughtAt: number | null = null;
      let firstAnswerAt: number | null = null;
      let thinkingDurationMs: number | undefined = undefined;
      let accContent = '';
      let accThought = '';
      let currentStatus: string | undefined = undefined;
      let accUsage: TokenUsage | undefined = undefined;
      let chunkCount = 0;

      const initialStream: ActiveStreamState = {
        convId,
        msgId: assistantMessageId,
        content: '',
        thought: '',
        tps: 0,
      };
      activeStreamRef.current = initialStream;
      setActiveStream(initialStream);

      const commitStreamSnapshot = () => {
        flushTimerRef.current = null;
        if (!activeStreamRef.current) return;
        const elapsed = Math.max(100, performance.now() - (firstAnswerAt ?? startedAt));
        const estTok = Math.max(chunkCount, Math.round(accContent.length / 4));
        const tps = firstAnswerAt ? Number(((estTok / elapsed) * 1000).toFixed(1)) : 0;
        const snapshot: ActiveStreamState = {
          convId,
          msgId: assistantMessageId,
          content: accContent,
          thought: accThought,
          statusText: currentStatus,
          tps,
          thinkingDurationMs,
          usage: accUsage,
        };
        activeStreamRef.current = snapshot;
        setActiveStream(snapshot);
      };

      const scheduleFlush = () => {
        if (flushTimerRef.current !== null) return;
        const cadence = accContent.length > 3000 ? 55 : 24;
        flushTimerRef.current = window.setTimeout(commitStreamSnapshot, cadence);
      };

      try {
        await streamChatCompletion(
          {
            model: modelId,
            messages: trimmedHistory.map((m) => ({
              role: m.role,
              content: m.content,
              visualTarget: m.visualTarget,
              attachments: m.attachments,
            })),
            systemInstruction: effectiveSystemInstruction,
            temperature: settings.temperature,
            maxOutputTokens: settings.maxOutputTokens,
            includeThoughts: settings.includeThoughts,
            thinkingBudget: settings.thinkingBudget,
            autoFallbackOnOverload: settings.autoFallbackOnOverload,
            fallbackModel: settings.fallbackModel,
            demoMode: settings.demoMode,
            apiKey: settings.apiKey,
          },
          {
            signal: controller.signal,
            onStatus: (statusMessage) => {
              currentStatus = statusMessage;
              commitStreamSnapshot();
            },
            onThought: (thoughtChunk) => {
              currentStatus = undefined;
              if (firstThoughtAt === null) firstThoughtAt = performance.now();
              accThought += thoughtChunk;
              scheduleFlush();
            },
            onDelta: (textChunk) => {
              currentStatus = undefined;
              if (firstAnswerAt === null) {
                firstAnswerAt = performance.now();
                const thinkOrigin = firstThoughtAt ?? startedAt;
                thinkingDurationMs = Math.max(100, Math.round(firstAnswerAt - thinkOrigin));
              }
              accContent += textChunk;
              chunkCount += 1;
              scheduleFlush();
            },
            onUsage: (usage) => {
              accUsage = usage;
            },
            onInfo: (infoMsg) => {
              toast.info(infoMsg);
            },
            onDone: ({ model: resolvedModel, isDemo }) => {
              if (flushTimerRef.current !== null) {
                window.clearTimeout(flushTimerRef.current);
                flushTimerRef.current = null;
              }
              activeStreamRef.current = null;
              setActiveStream(null);

              const finishedAt = performance.now();
              const durationMs = Math.max(1, Math.round(finishedAt - startedAt));
              const firstAny = firstAnswerAt ?? firstThoughtAt;
              const ttftMs = firstAny !== null ? Math.round(firstAny - startedAt) : durationMs;

              if (thinkingDurationMs === undefined && accThought) {
                thinkingDurationMs = Math.max(
                  100,
                  Math.round(finishedAt - (firstThoughtAt ?? startedAt))
                );
              }

              const answerWindow = Math.max(1, finishedAt - (firstAnswerAt ?? startedAt));
              const estTokens = Math.max(chunkCount, Math.round(accContent.length / 4));
              const tokensPerSec = Number(((estTokens / answerWindow) * 1000).toFixed(1));

              if (settings.soundChime && durationMs >= 2500) {
                playCompletionChime();
              }

              setConversations((prev) =>
                prev.map((conv) =>
                  conv.id !== convId
                    ? conv
                    : {
                        ...conv,
                        updatedAt: Date.now(),
                        messages: conv.messages.map((m) =>
                          m.id === assistantMessageId
                            ? {
                                ...m,
                                model: resolvedModel || m.model,
                                content: accContent,
                                thought: accThought || undefined,
                                statusText: undefined,
                                usage: accUsage,
                                isStreaming: false,
                                isDemo,
                                metrics: {
                                  ttftMs,
                                  thinkingDurationMs,
                                  durationMs,
                                  tokensPerSec,
                                },
                              }
                            : m
                        ),
                      }
                )
              );

              if (settings.autoOpenArtifacts && accContent) {
                const detected = extractArtifactsFromMarkdown(assistantMessageId, accContent);
                const previewable = detected.find(
                  (a) => a.kind === 'html' || a.kind === 'svg' || a.kind === 'mermaid'
                );
                if (previewable) setActiveArtifact(previewable);
              }
            },
            onError: (errorMessage) => {
              if (flushTimerRef.current !== null) {
                window.clearTimeout(flushTimerRef.current);
                flushTimerRef.current = null;
              }
              activeStreamRef.current = null;
              setActiveStream(null);
              toast.error(errorMessage);
              setConversations((prev) =>
                prev.map((conv) =>
                  conv.id !== convId
                    ? conv
                    : {
                        ...conv,
                        messages: conv.messages.map((m) =>
                          m.id === assistantMessageId
                            ? {
                                ...m,
                                content: accContent,
                                thought: accThought || undefined,
                                statusText: undefined,
                                isStreaming: false,
                                error: errorMessage,
                              }
                            : m
                        ),
                      }
                )
              );
            },
          }
        );
      } catch (err) {
        if (err instanceof Error && err.name === 'AbortError') return;
        const message = err instanceof Error ? err.message : 'Gagal terhubung ke server SSE.';
        toast.error(message);
        activeStreamRef.current = null;
        setActiveStream(null);
        setConversations((prev) =>
          prev.map((conv) =>
            conv.id !== convId
              ? conv
              : {
                  ...conv,
                  messages: conv.messages.map((m) =>
                    m.id === assistantMessageId
                      ? {
                          ...m,
                          content: accContent,
                          statusText: undefined,
                          isStreaming: false,
                          error: message,
                        }
                      : m
                  ),
                }
          )
        );
      } finally {
        if (abortControllerRef.current === controller) {
          abortControllerRef.current = null;
          if (activeStreamRef.current?.msgId === assistantMessageId) {
            activeStreamRef.current = null;
            setActiveStream(null);
            setConversations((prev) =>
              prev.map((conv) =>
                conv.id !== convId
                  ? conv
                  : {
                      ...conv,
                      messages: conv.messages.map((m) =>
                        m.id === assistantMessageId && m.isStreaming
                          ? {
                              ...m,
                              content: accContent,
                              thought: accThought || undefined,
                              statusText: undefined,
                              isStreaming: false,
                            }
                          : m
                      ),
                    }
              )
            );
          }
        }
      }
    },
    [settings]
  );

  const handleSend = useCallback(
    (text: string, attachments: Attachment[]) => {
      if (isStreamingActive) return;

      const currentVisualTarget = visualTarget;
      setVisualTarget(null);

      const userMessage: ChatMessageType = {
        id: generateId(),
        role: 'user',
        content: text,
        visualTarget: currentVisualTarget || undefined,
        attachments: attachments.length > 0 ? attachments : undefined,
        createdAt: Date.now(),
      };

      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: currentModelId,
        isStreaming: true,
      };

      if (!activeConversation) {
        const newId = generateId();
        const title = (text || attachments[0]?.name || 'New Chat').slice(0, 50);
        const newConv: Conversation = {
          id: newId,
          title,
          model: currentModelId,
          messages: [userMessage, assistantPlaceholder],
          createdAt: Date.now(),
          updatedAt: Date.now(),
        };

        setConversations((prev) => [newConv, ...prev]);
        setActiveId(newId);
        void runStreamForConversation(newId, currentModelId, [userMessage], assistantMessageId);
      } else {
        const historyForModel = [...activeConversation.messages, userMessage];
        const nextCompacted: CompactedContextState | undefined = activeConversation.compactedContext
          ? activeConversation.compactedContext.mode === 'ai-summary'
            ? {
                ...activeConversation.compactedContext,
                keepRecentCount: activeConversation.compactedContext.keepRecentCount + 2,
              }
            : activeConversation.compactedContext
          : undefined;

        setConversations((prev) =>
          prev.map((conv) =>
            conv.id !== activeConversation.id
              ? conv
              : {
                  ...conv,
                  compactedContext: nextCompacted,
                  updatedAt: Date.now(),
                  messages: [...conv.messages, userMessage, assistantPlaceholder],
                }
          )
        );
        void runStreamForConversation(
          activeConversation.id,
          currentModelId,
          historyForModel,
          assistantMessageId,
          activeConversation.sessionSystemInstruction,
          nextCompacted
        );
      }
    },
    [activeConversation, currentModelId, isStreamingActive, runStreamForConversation, visualTarget]
  );

  const handleRegenerate = useCallback(
    (overrideModel?: string) => {
      if (!activeConversation || isStreamingActive) return;
      const msgs = activeConversation.messages;
      if (msgs.length === 0) return;

      const lastMsg = msgs[msgs.length - 1];
      const baseMessages = lastMsg.role === 'assistant' ? msgs.slice(0, -1) : msgs;
      if (baseMessages.length === 0) return;

      const targetModel = overrideModel || activeConversation.model;
      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: targetModel,
        isStreaming: true,
      };

      setConversations((prev) =>
        prev.map((conv) =>
          conv.id !== activeConversation.id
            ? conv
            : {
                ...conv,
                model: targetModel,
                updatedAt: Date.now(),
                messages: [...baseMessages, assistantPlaceholder],
              }
        )
      );

      void runStreamForConversation(
        activeConversation.id,
        targetModel,
        baseMessages,
        assistantMessageId,
        activeConversation.sessionSystemInstruction,
        activeConversation.compactedContext
      );
    },
    [activeConversation, isStreamingActive, runStreamForConversation]
  );

  const handleEditUserMessage = useCallback(
    (messageId: string, newContent: string) => {
      if (!activeConversation || isStreamingActive) return;
      const idx = activeConversation.messages.findIndex((m) => m.id === messageId);
      if (idx === -1) return;

      const updatedMsg: ChatMessageType = {
        ...activeConversation.messages[idx],
        content: newContent,
      };
      const baseHistory = [...activeConversation.messages.slice(0, idx), updatedMsg];

      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: activeConversation.model,
        isStreaming: true,
      };

            ? conv
            : {
                ...conv,
                updatedAt: Date.now(),
                messages: [...baseHistory, assistantPlaceholder],
              }
        )
      );

      void runStreamForConversation(
        activeConversation.id,
        activeConversation.model,
        baseHistory,
        assistantMessageId,
        activeConversation.sessionSystemInstruction,
        activeConversation.compactedContext
      );
    },
    [activeConversation, isStreamingActive, runStreamForConversation]

      if (!activeConversation) return;

      if (idx === -1) return;

        id: forkedId,
        title: `Branch: ${activeConversation.title}`.slice(0, 52),
        model: activeConversation.model,
        sessionSystemInstruction: activeConversation.sessionSystemInstruction,
        compactedContext: activeConversation.compactedContext,
        messages: activeConversation.messages.slice(0, idx + 1).map((m) => ({
          ...m,
          id: generateId(),
        })),
        createdAt: Date.now(),
        updatedAt: Date.now(),
      };
      setConversations((prev) => [forkedConv, ...prev]);
      setActiveId(forkedId);
      toast.success('Cabang chat baru dibuat');
    },
    [activeConversation]

      model: string;
      sessionSystemInstruction?: string;
      sourceLabel?: string;

                  messages: [...conv.messages, ...payload.messages],

          id: newId,
          title: payload.title,
          model: payload.model || selectedModel,
          sessionSystemInstruction: payload.sessionSystemInstruction,
          importedSource: payload.sourceLabel,
          compactedContext: payload.compactedContext,
          messages: payload.messages,
          createdAt: Date.now(),
          updatedAt: Date.now(),

          `Berhasil impor ${payload.messages.length} pesan (${payload.sourceLabel || 'Konteks'}). Silakan lanjutkan chat!`
        );

    [activeConversation, handleStopStream, isStreamingActive, selectedModel]

                      onClick={() => setImportDialogOpen(true)}

                      Command Palette (Ctrl+K)
                    </DropdownMenuItem>
                    {activeConversation && (

                        <DropdownMenuSeparator />

                          <FileText className="h-3.5 w-3.5" />
                          Export Markdown (.md)

                      Settings & API Key Pool

                </DropdownMenu>

                className="h-full w-full"

                  <div className="relative mx-auto flex min-h-[calc(100dvh-13rem)] w-full max-w-3xl flex-col items-center justify-center px-4 py-8 text-center">
                    <div className="pointer-events-none absolute inset-0 -z-10 bg-dot-matrix opacity-35 [mask-image:radial-gradient(ellipse_at_center,black_30%,transparent_75%)]" />
                    <div className="pointer-events-none absolute top-10 -z-10 h-56 w-56 rounded-full bg-primary/10 blur-3xl" />

                      initial={{ scale: 0.92, opacity: 0 }}

                      transition={{ duration: 0.24 }}

                    <div className="mb-2 inline-flex flex-wrap items-center justify-center gap-2 rounded-full border bg-card/90 px-3.5 py-1 text-xs text-muted-foreground shadow-sm">
                      <Cpu className="h-3.5 w-3.5 text-primary" />
                      <span className="font-mono font-semibold text-foreground">
                        {currentModelId}
                      </span>
                      {activeKeyPoolCount > 0 && (

                          <span>•</span>

                            {activeKeyPoolCount} API Key{activeKeyPoolCount > 1 ? 's Pool' : ''}
                          </span>

                          <span>•</span>
                          <span className="text-primary">App Mode</span>
                        </>

                    </div>

                    <h1 className="mt-2 text-2xl font-bold tracking-tight text-foreground sm:text-4xl">
                      Adam Studio Workspace
                    </h1>
                    <p className="mt-2.5 max-w-lg text-sm leading-relaxed text-muted-foreground">
                      Platform AI dengan Live Thinking, Auto-Rotate Key Pool, Visual Inspector,
                      Per-File Token Counter, & Impor Konteks AI Studio.

                      <Button
                        type="button"
                        variant="outline"
                        size="sm"
                        onClick={() => setImportDialogOpen(true)}
                        className="h-8 gap-1.5 rounded-full border-primary/35 bg-primary/5 text-xs text-primary hover:bg-primary/10"
                      >
                        <FileUp className="h-3.5 w-3.5" />
                        Lanjutkan Chat dari Google AI Studio (.json / .md)

                    </div>

                    {activeKeyPoolCount === 0 && !settings.demoMode && (
                      <div className="mt-4 flex flex-wrap items-center justify-center gap-2 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-2.5 text-xs text-amber-400">
                        <span>API Key belum aktif.</span>

                          type="button"
                          onClick={() => setSettingsDialogOpen(true)}
                          className="font-semibold underline underline-offset-4 hover:text-amber-300"
                        >
                          Isi API Key Pool atau aktifkan Demo SSE
                        </button>

                    {/* Welcome Tabs dengan fix kontras isolate + z-0/z-10 */}
                    <div className="mt-6 inline-flex flex-wrap items-center justify-center gap-1 rounded-full border bg-card/80 p-1">
                      {WELCOME_TABS.map((tab) => {

                              message={effectiveMsg}
                              compact={settings.density === 'compact'}
                              thinkingEnabled={thinkingEnabled}
                              isLastAssistant={msg.id === lastAssistantId}
                              onRegenerate={
                                msg.id === lastAssistantId ? () => handleRegenerate() : undefined
                              }
                              onRetryWithModel={
                                msg.id === lastAssistantId
                                  ? (mId) => handleRegenerate(mId)
                                  : undefined
                              }
                              onEditUserMessage={
                                msg.role === 'user' && !isStreamingActive
                                  ? handleEditUserMessage
                                  : undefined
                              }
                              onForkFromMessage={
                                msg.role === 'assistant' && !isStreamingActive
                                  ? handleForkFromMessage
                                  : undefined
                              }
                              onToggleStar={(msgId) =>
                                handleToggleStar(activeConversation.id, msgId)
                              }
                              onOpenArtifact={setActiveArtifact}
                              onPreviewImage={setLightboxUrl}
                            />
                          </div>
                        );
                      })}
                    </div>
                  </div>
                )}
              </StudioScroll>

              {showScrollBottom && (
                <button
                  type="button"
                  onClick={() => {
                    autoScrollEnabledRef.current = true;
                    scrollToBottom(true);
                  }}
                  className={cn(
                    'absolute bottom-3 left-1/2 z-20 flex -translate-x-1/2 items-center gap-2 rounded-full border px-3.5 py-1.5 text-xs font-medium shadow-xl backdrop-blur-md transition-all hover:scale-105',
                    isStreamingActive
                      ? 'border-primary/40 bg-card/95 text-foreground'
                      : 'bg-secondary/95 text-secondary-foreground'
                  )}
                  aria-label="Scroll ke bawah"
                >
                  {isStreamingActive ? (
                    <>
                      <span className="h-2 w-2 animate-ping rounded-full bg-primary" />
                      <span>
                        {activeStream.statusText
                          ? 'Rotating Key...'
                          : activeStream.tps > 0
                            ? `Streaming ${activeStream.tps} tok/s`
                            : 'Sedang Thinking...'}
                      </span>
                      <ArrowDown className="h-3.5 w-3.5 text-primary" />
                    </>
                  ) : (
                    <ArrowDown className="h-4 w-4" />
                  )}
                </button>
              )}
            </div>

            <ChatComposer
              model={currentModelId}
              isStreaming={isStreamingActive}
              settings={settings}
              hasApiKeyConfigured={activeKeyPoolCount > 0}
              externalDraft={externalDraft}
              visualTarget={visualTarget}
              onClearVisualTarget={() => setVisualTarget(null)}
              onClearExternalDraft={() => setExternalDraft(null)}
              onUpdateSettings={setSettings}
              onSend={handleSend}
              onStop={handleStopStream}
              onPreviewImage={setLightboxUrl}
            />
          </div>

          <AnimatePresence>
            {activeArtifact && (
              <>
                {!artifactFullscreen && (
                  <div
                    onPointerDown={startResizingPane}
                    className="hidden w-1.5 shrink-0 cursor-col-resize items-center justify-center border-l bg-border/40 transition-colors hover:bg-primary/60 lg:flex"
                    title="Tarik untuk mengubah lebar Canvas"
                  />
                )}
                <motion.aside
                  initial={{ opacity: 0, x: 20 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 20 }}
                  transition={{ duration: 0.14 }}
                  className={cn(
                    'fixed inset-0 z-40 border-l lg:static',
                    artifactFullscreen ? 'lg:flex-1' : 'lg:shrink-0'
                  )}
                >
                  <div
                    style={
                      {
                        '--artifact-w': `${artifactWidthPct}vw`,
                      } as CSSProperties
                    }
                    className={cn(
                      'h-full w-full',
                      !artifactFullscreen &&
                        'lg:w-[var(--artifact-w)] lg:min-w-[360px] lg:max-w-[68vw]'
                    )}
                  >
                    <ArtifactPanel
                      artifact={activeArtifact}
                      allArtifacts={conversationArtifacts}
                      isResizing={isResizingPane}
                      isFullscreen={artifactFullscreen}
                      onToggleFullscreen={() => setArtifactFullscreen((f) => !f)}
                      onSelectArtifact={setActiveArtifact}
                      onSelectVisualTarget={(target) => {
                        setVisualTarget(target);
                        if (window.innerWidth < 1024) {
                          setActiveArtifact(null);
                          setArtifactFullscreen(false);
                        }
                      }}
                      onClose={() => {
                        setActiveArtifact(null);
                        setArtifactFullscreen(false);
                      }}
                    />
                  </div>
                </motion.aside>
              </>
            )}
          </AnimatePresence>

          <ModelSelectorDialog
            open={modelDialogOpen}
            onOpenChange={setModelDialogOpen}
            selectedModel={currentModelId}
            onSelectModel={handleSelectModel}
          />

          <SettingsDialog
            open={settingsDialogOpen}
            onOpenChange={setSettingsDialogOpen}
            settings={settings}
            serverHealth={serverHealth}
            onSave={setSettings}
          />

          <ImportChatDialog
            open={importDialogOpen}
            onOpenChange={setImportDialogOpen}
            hasActiveSession={Boolean(activeConversation && activeConversation.messages.length > 0)}
            activeSessionTitle={activeConversation?.title}
            defaultModel={currentModelId}
            onConfirmImport={handleConfirmImport}
          />

          <ContextManagerDialog
            open={contextDialogOpen}
            onOpenChange={setContextDialogOpen}
            conversation={activeConversation}
            apiKey={settings.apiKey}
            hasApiKey={activeKeyPoolCount > 0}
            onApplyCompaction={handleApplyCompaction}
          />

          <CommandPalette
            open={commandOpen}
            onOpenChange={setCommandOpen}
            conversations={conversations}
            activeSkin={settings.skin}
            onNewChat={handleNewChat}
            onImportChat={() => setImportDialogOpen(true)}
            onSelectConversation={(id) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(id);
              setActiveArtifact(null);
              setArtifactFullscreen(false);
            }}
            onOpenModelSelector={() => setModelDialogOpen(true)}
            onOpenSettings={() => setSettingsDialogOpen(true)}
            onOpenBookmarks={() => setBookmarksOpen(true)}
            onToggleZenMode={() => setZenMode((z) => !z)}
            onCycleSkin={handleCycleSkin}
            onExportMarkdown={() => handleExportConversation('md')}
            onInsertTemplate={(tpl) => setExternalDraft(tpl)}
            showInstallOption={!isStandalone}
            onInstallApp={() => void handleInstallAction()}
          />

          <BookmarksDialog
            open={bookmarksOpen}
            onOpenChange={setBookmarksOpen}
            conversations={conversations}
            onJumpToConversation={(convId) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(convId);
            }}
            onToggleStar={handleToggleStar}
          />

          <InstallAppDialog
            open={installDialogOpen}
            onOpenChange={setInstallDialogOpen}
            isIos={isIos}
            canNativeInstall={canNativeInstall}
            onNativeInstall={() => {
              void handleInstallAction().then(() => setInstallDialogOpen(false));
            }}
          />

          <Dialog open={Boolean(lightboxUrl)} onOpenChange={() => setLightboxUrl(null)}>
            <DialogContent className="max-w-4xl border-0 bg-transparent p-2 shadow-none">
              <DialogHeader className="sr-only">
                <DialogTitle>Image Preview</DialogTitle>
              </DialogHeader>
              {lightboxUrl && (
                <img
                  src={lightboxUrl}
                  alt="Attachment Preview"
                  className="max-h-[85dvh] w-full rounded-2xl object-contain shadow-2xl"
                />
              )}
            </DialogContent>
          </Dialog>
        </div>
      </TooltipProvider>
    </MotionConfig>
  );
}

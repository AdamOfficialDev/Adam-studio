export async function getMediaDuration(file: File, kind: 'video' | 'audio'): Promise<number | undefined> {
  return new Promise((resolve) => {
    const url = URL.createObjectURL(file);
    const el = document.createElement(kind);
    el.preload = 'metadata';

    const timer = window.setTimeout(() => {
      URL.revokeObjectURL(url);
      resolve(undefined);
    }, 1800);

    el.onloadedmetadata = () => {
      window.clearTimeout(timer);
      const dur = Number.isFinite(el.duration) && el.duration > 0 ? el.duration : undefined;
      URL.revokeObjectURL(url);
      resolve(dur);
    };

    el.onerror = () => {
      window.clearTimeout(timer);
      URL.revokeObjectURL(url);
      resolve(undefined);
    };

    el.src = url;
  });
}

export async function getImageDimensions(
  dataUrl: string
): Promise<{ width: number; height: number } | undefined> {
  return new Promise((resolve) => {
    const img = new Image();
    img.onload = () => resolve({ width: img.naturalWidth, height: img.naturalHeight });
    img.onerror = () => resolve(undefined);

  });
}

export async function estimatePdfPages(file: File): Promise<number> {
  try {
    const sample = await file.slice(0, 2_000_000).text();
    const matches = sample.match(/\/Type\s*\/Page\b(?!s)/g);
    if (matches && matches.length > 0) return matches.length;
  } catch {
    // Abaikan jika gagal baca
  }
  return Math.max(1, Math.round(file.size / 45_000));
}

export function estimateAttachmentTokensLocal(params: {
  kind: Attachment['kind'];
  size: number;
  textContent?: string;
  width?: number;
  height?: number;
  durationSec?: number;
  pdfPages?: number;
}): number {
  const { kind, size, textContent, width, height, durationSec, pdfPages } = params;

  if (kind === 'text') {
    return Math.max(1, Math.ceil((textContent?.length ?? size) / 4));
  }

  if (kind === 'image') {
    if (width && height) {
      if (width <= 384 && height <= 384) return 258;
      const tiles = Math.max(1, Math.ceil(width / 768) * Math.ceil(height / 768));
      return Math.min(6, tiles) * 258;
    }
    return 258;
  }

  if (kind === 'video') {
    if (durationSec && durationSec > 0) {
      // Gemini rate: ~263 video frames/sec + ~32 audio tokens/sec
      return Math.max(295, Math.round(durationSec * 295));
    }
    return Math.max(500, Math.round((size / 120_000) * 295));
  }

  if (kind === 'audio') {
    if (durationSec && durationSec > 0) {
      return Math.max(32, Math.round(durationSec * 32));
    }
    return Math.max(64, Math.round((size / 16_000) * 32));
  }

  if (kind === 'pdf') {
    const pages = pdfPages || Math.max(1, Math.round(size / 45_000));
    return pages * 258;
  }

  return Math.max(1, Math.ceil(size / 4));
}

export async function countAttachmentTokensExact(params: {
  model: string;
  apiKey: string;
  attachment: Attachment;
}): Promise<number | null> {
  const { model, apiKey, attachment } = params;

  // Hindari mengirim payload base64 raksasa (>3.8MB) ke serverless /countTokens
  const approxPayloadSize = attachment.data?.length ?? attachment.textContent?.length ?? 0;
  if (approxPayloadSize > 3_800_000) {
    return null;
  }

  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
  };

  const normalizedKeys = (apiKey || '')
    .split(/[\r\n,;]+/)
    .map((k) => k.trim())
    .filter(Boolean);
  if (normalizedKeys.length > 0) {
    headers['x-gemini-api-key'] = normalizedKeys.join(',');
  }

  try {
    const res = await fetch('/api/count-tokens', {
      method: 'POST',
      headers,
      body: JSON.stringify({
        model,
        attachment: {
          kind: attachment.kind,
          name: attachment.name,
          mimeType: attachment.mimeType,
          data: attachment.data,
          textContent: attachment.textContent,
        },
      }),
    });

    if (!res.ok) return null;
    const json = (await res.json()) as { totalTokens?: number };
    return typeof json.totalTokens === 'number' ? json.totalTokens : null;
  } catch {
    return null;
  }
}

type UserSegment =
  | { type: 'text'; text: string }
  | { type: 'target'; selector: string }
  | { type: 'code'; language: string; code: string };

function parseUserMessageSegments(content: string): UserSegment[] {
  const segments: UserSegment[] = [];
  const codeRegex = /```([\w-]*)\n([\s\S]*?)```/g;
  let lastIndex = 0;
  let match: RegExpExecArray | null;

  const pushTextWithTarget = (rawChunk: string) => {
    const targetMatch = rawChunk.match(/^\[Visual Target Element:\s*`([^`]+)`\]\s*/);
    if (targetMatch) {
      segments.push({ type: 'target', selector: targetMatch[1] });
      const rest = rawChunk.slice(targetMatch[0].length).trim();
      if (rest) segments.push({ type: 'text', text: rest });
    } else if (rawChunk.trim().length > 0) {
      segments.push({ type: 'text', text: rawChunk.replace(/^\n+|\n+$/g, '') });
    }
  };

  while ((match = codeRegex.exec(content)) !== null) {
    if (match.index > lastIndex) {
      pushTextWithTarget(content.slice(lastIndex, match.index));
    }
    segments.push({
      type: 'code',
      language: match[1] || 'text',
      code: match[2].replace(/\n$/, ''),
    });
    lastIndex = codeRegex.lastIndex;
  }

  if (lastIndex < content.length) {
    pushTextWithTarget(content.slice(lastIndex));
  }

  if (segments.length === 0 && content) {
    segments.push({ type: 'text', text: content });
  }

  return segments;
}

    return SLASH_PROMPTS.filter(
      (p) => p.command.startsWith(cmd) || p.title.toLowerCase().includes(cmd.slice(1))
    );
  }, [input]);

  useEffect(() => {
    setSlashIndex(0);
  }, [slashSuggestions.length]);

  const ingestFiles = async (files: File[]) => {
    const loaded: Attachment[] = [];

    for (const file of files.slice(0, 5)) {
      if (file.size > 18_000_000) {
        toast.error(`File ${file.name} terlalu besar (>18MB).`);
        continue;
      }

      const ext = file.name.split('.').pop()?.toLowerCase() || '';
      const isImage = file.type.startsWith('image/');
      const isVideo = file.type.startsWith('video/') || ['mp4', 'webm', 'mov'].includes(ext);
      const isAudio =
        file.type.startsWith('audio/') || ['mp3', 'wav', 'ogg', 'm4a'].includes(ext);
      const isPdf = file.type === 'application/pdf' || ext === 'pdf';
      const isText = file.type.startsWith('text/') || TEXT_EXTENSIONS.has(ext);

      if (isImage) {
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        if (!base64) continue;
        const dims = await getImageDimensions(dataUrl);
        const localEst = estimateAttachmentTokensLocal({
          kind: 'image',
          size: file.size,
          width: dims?.width,
          height: dims?.height,
        });
        loaded.push({
          id: generateId(),
          kind: 'image',
          name: file.name,
          mimeType: file.type || 'image/png',
          size: file.size,
          data: base64,
          previewUrl: dataUrl,
          tokenCount: localEst,
          tokenStatus: settings.demoMode ? 'estimated' : 'counting',
        });
      } else if (isVideo) {
        const durationSec = await getMediaDuration(file, 'video');
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        if (!base64) continue;
        const localEst = estimateAttachmentTokensLocal({
          kind: 'video',
          size: file.size,
          durationSec,
        });
        loaded.push({
          id: generateId(),
          kind: 'video',
          name: file.name,
          mimeType: file.type || 'video/mp4',
          size: file.size,
          data: base64,
          durationSec,
          tokenCount: localEst,
          tokenStatus: settings.demoMode ? 'estimated' : 'counting',
        });
      } else if (isAudio) {
        const durationSec = await getMediaDuration(file, 'audio');
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        if (!base64) continue;
        const localEst = estimateAttachmentTokensLocal({
          kind: 'audio',
          size: file.size,
          durationSec,
        });
        loaded.push({
          id: generateId(),
          kind: 'audio',
          name: file.name,
          mimeType: file.type || 'audio/mpeg',
          size: file.size,
          data: base64,
          durationSec,
          tokenCount: localEst,
          tokenStatus: settings.demoMode ? 'estimated' : 'counting',
        });
      } else if (isPdf) {
        const pages = await estimatePdfPages(file);
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        if (!base64) continue;
        const localEst = estimateAttachmentTokensLocal({
          kind: 'pdf',
          size: file.size,
          pdfPages: pages,
        });
        loaded.push({
          id: generateId(),
          kind: 'pdf',
          name: file.name,
          mimeType: 'application/pdf',
          size: file.size,
          data: base64,
          tokenCount: localEst,
          tokenStatus: settings.demoMode ? 'estimated' : 'counting',
        });
      } else if (isText) {
        const textContent = (await file.text()).slice(0, 120_000);
        const localEst = estimateAttachmentTokensLocal({
          kind: 'text',
          size: file.size,
          textContent,
        });
        loaded.push({
          id: generateId(),
          kind: 'text',
          name: file.name,
          mimeType: file.type || 'text/plain',
          size: file.size,
          textContent,
          tokenCount: localEst,
          tokenStatus: settings.demoMode ? 'estimated' : 'counting',
        });
      } else {
        toast.error(`Format ${file.name} belum didukung.`);
      }
    }

    if (loaded.length === 0) return;

    setAttachments((prev) => [...prev, ...loaded]);
    if (loaded.some((a) => a.size > 3_500_000)) {
      toast.info('Catatan Vercel: file >3.5MB bisa melebihi batas payload 4.5MB serverless.');
    }

    if (settings.demoMode) return;

    // Hitung token resmi per file lewat Gemini :countTokens API secara paralel
    for (const item of loaded) {
      void countAttachmentTokensExact({
        model,
        apiKey: settings.apiKey,
        attachment: item,
      }).then((exactCount) => {
        setAttachments((prev) =>
          prev.map((att) => {
            if (att.id !== item.id) return att;
            if (exactCount !== null) {
              return { ...att, tokenCount: exactCount, tokenStatus: 'exact' };
            }
            return { ...att, tokenStatus: 'estimated' };
          })
        );
      });
    }
  };

  const handleFileChange = async (e: ChangeEvent<HTMLInputElement>) => {
    const files = Array.from(e.target.files || []);
    if (files.length > 0) await ingestFiles(files);
    e.target.value = '';
  };

  const handlePaste = async (e: ClipboardEvent<HTMLTextAreaElement>) => {
    const files = Array.from(e.clipboardData.files || []);
    if (files.length > 0) {
      e.preventDefault();
      await ingestFiles(files);
    }
  };

  const handleDrop = async (e: DragEvent<HTMLDivElement>) => {
    e.preventDefault();
    setIsDragging(false);
    const files = Array.from(e.dataTransfer.files || []);
    if (files.length > 0) await ingestFiles(files);
  };

  const toggleVoiceInput = () => {
    if (!speechSupported) return;
    if (isListening) {
      recognitionRef.current?.stop();
      setIsListening(false);
      return;
    }

    const w = window as unknown as {
      SpeechRecognition?: SpeechRecognitionCtor;
      webkitSpeechRecognition?: SpeechRecognitionCtor;
    };
    const Ctor = w.SpeechRecognition || w.webkitSpeechRecognition;
    if (!Ctor) return;

    const recognition = new Ctor();
    recognition.lang = 'id-ID';
    recognition.interimResults = false;
    recognition.continuous = false;

    recognition.onresult = (event) => {
      const transcript = Array.from(event.results)
        .map((res) => res[0]?.transcript || '')
        .join(' ')
        .trim();
      if (transcript) {
        setInput((prev) => (prev ? `${prev} ${transcript}` : transcript));
        setTimeout(adjustHeight, 10);
      }
    };
    recognition.onerror = () => setIsListening(false);
    recognition.onend = () => setIsListening(false);

    recognitionRef.current = recognition;
    setIsListening(true);
    recognition.start();
  };

  const handleEnhancePrompt = () => {
    const trimmed = input.trim();
    if (!trimmed) return;
    const enhanced = [
      `Tolong kerjakan permintaan berikut secara terstruktur dan menyeluruh:`,
      ``,
      `**Tujuan Utama:** ${trimmed}`,
      ``,
      `**Kriteria Output:**`,
      `- Berikan analisis arsitektur / langkah jelas`,
      `- Sertakan contoh implementasi lengkap & siap jalan`,
      `- Jelaskan edge-case dan optimasi performa`,
    ].join('\n');
    setInput(enhanced);
    setTimeout(adjustHeight, 10);
    toast.success('Prompt diperkaya');
  };

  const applySlashTemplate = (template: string) => {
    setInput(template);
    setTimeout(() => {
      textareaRef.current?.focus();
      adjustHeight();
    }, 10);
  };

  const submit = () => {
    if (isStreaming) return;
    const trimmed = input.trim();
    if (!trimmed && attachments.length === 0) return;

    onSend(trimmed, attachments);
    setInput('');
    setAttachments([]);
    if (textareaRef.current) {
      textareaRef.current.style.height = 'auto';
    }
  };

  const handleKeyDown = (e: KeyboardEvent<HTMLTextAreaElement>) => {
    if (slashSuggestions.length > 0) {
      if (e.key === 'ArrowDown') {
        e.preventDefault();
        setSlashIndex((i) => (i + 1) % slashSuggestions.length);
        return;
      }
      if (e.key === 'ArrowUp') {
        e.preventDefault();
        setSlashIndex((i) => (i - 1 + slashSuggestions.length) % slashSuggestions.length);
        return;
      }
      if (e.key === 'Enter' || e.key === 'Tab') {
        e.preventDefault();
        const chosen = slashSuggestions[slashIndex];
        if (chosen) applySlashTemplate(chosen.template);
        return;
      }
    }

    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      submit();
    }
  };

  const activePersona = PERSONA_PRESETS.find((p) => p.id === settings.activePersonaId);
  const draftTokens = estimateTokens(input);
  const attachmentsTokens = attachments.reduce((sum, a) => sum + (a.tokenCount ?? 0), 0);
  const totalComposerTokens = draftTokens + attachmentsTokens;
  const thinkingOn = settings.includeThoughts && settings.thinkingBudget !== 0;

  return (
    <div
      className="relative mx-auto w-full max-w-3xl px-3 pb-3 pt-1.5 sm:px-4 sm:pb-4"
      onDragOver={(e) => {
        e.preventDefault();
        setIsDragging(true);
      }}
      onDragLeave={() => setIsDragging(false)}
      onDrop={handleDrop}
    >
      <AnimatePresence>
        {slashSuggestions.length > 0 && (
          <motion.div
            initial={{ opacity: 0, y: 6 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: 6 }}
            transition={{ duration: 0.14 }}
            className="mb-2 overflow-hidden rounded-2xl border bg-popover/95 p-1.5 shadow-2xl backdrop-blur-xl"
          >
            <div className="flex items-center justify-between px-2.5 py-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
              <span>Slash Prompt Templates</span>
              <span className="hidden sm:inline">↑↓ pilih • Enter sisipkan</span>
            </div>
            <StudioScroll className="max-h-[210px]" autoHide="scroll">
              <div className="space-y-0.5">
                {slashSuggestions.map((item, idx) => (
                  <button
                    key={item.command}
                    type="button"
                    onClick={() => applySlashTemplate(item.template)}
                    className={cn(
                      'flex w-full items-center justify-between rounded-xl px-2.5 py-2 text-left text-xs transition-colors',
                      idx === slashIndex ? 'bg-accent text-accent-foreground' : 'hover:bg-accent/60'
                    )}
                  >
                    <span className="flex min-w-0 items-center gap-2.5">
                      <Terminal className="h-3.5 w-3.5 shrink-0 text-primary" />
                      <span className="truncate">
                        <span className="font-semibold text-foreground">{item.title}</span>
                        <span className="ml-2 hidden text-muted-foreground sm:inline">
                          {item.description}
                        </span>
                      </span>
                    </span>
                    <code className="ml-2 shrink-0 rounded bg-muted px-1.5 py-0.5 font-mono text-[10px]">
                      {item.command}
                    </code>
                  </button>
                ))}
              </div>
            </StudioScroll>
          </motion.div>
        )}
      </AnimatePresence>

      <div
        className={cn(
          'rounded-2xl border border-border/90 bg-card/95 p-2.5 shadow-[0_12px_40px_-12px_rgba(0,0,0,0.45)] transition-all focus-within:border-primary/60 focus-within:ring-2 focus-within:ring-primary/20 sm:p-3',
          isDragging && 'border-primary ring-2 ring-primary/30'
        )}
      >
        {isDragging && (
          <div className="mb-2 rounded-xl border border-dashed border-primary bg-primary/10 py-3 text-center text-xs font-medium text-primary">
            Lepaskan gambar, video, audio, PDF, atau file kode di sini
          </div>
        )}

        {visualTarget && (
          <div className="mb-2 flex items-center justify-between gap-2 rounded-xl border border-primary/40 bg-primary/10 px-3 py-2 text-xs">
            <div className="flex min-w-0 items-center gap-2">
              <Crosshair className="h-3.5 w-3.5 shrink-0 text-primary" />
              <span className="truncate font-mono text-[11px] font-semibold text-primary">
                &lt;{visualTarget.selector}&gt;
              </span>
              {visualTarget.textSnippet && (
                <span className="hidden truncate text-muted-foreground sm:inline">
                  &ldquo;{visualTarget.textSnippet}&rdquo;
                </span>
              )}
            </div>
            {onClearVisualTarget && (
              <button
                type="button"
                onClick={onClearVisualTarget}
                className="rounded-md p-0.5 text-muted-foreground hover:bg-background hover:text-foreground"
                aria-label="Hapus target elemen"
              >
                <X className="h-3.5 w-3.5" />
              </button>
            )}
          </div>
        )}

        {attachments.length > 0 && (
          <div className="mb-2.5 flex flex-wrap gap-2 px-1">
            {attachments.map((att) => (
              <div
                key={att.id}
                className="group relative flex items-center gap-2 rounded-xl border bg-muted/50 p-1.5 pr-2.5 text-xs"
              >
                {att.kind === 'image' && att.previewUrl ? (
                  <button
                    type="button"
                    onClick={() => onPreviewImage(att.previewUrl!)}
                    className="overflow-hidden rounded-lg border"
                  >
                    <img
                      src={att.previewUrl}
                      alt={att.name}
                      className="h-10 w-10 object-cover transition-transform hover:scale-105"
                    />
                  </button>
                ) : (
                  <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg bg-primary/15 text-primary">
                    {att.kind === 'video' ? (
                      <Film className="h-4 w-4" />
                    ) : att.kind === 'audio' ? (
                      <FileAudio className="h-4 w-4" />
                    ) : att.kind === 'pdf' ? (
                      <FileText className="h-4 w-4 text-rose-400" />
                    ) : (
                      <FileCode2 className="h-4 w-4" />
                    )}
                  </div>
                )}

                <div className="min-w-0">
                  <div className="max-w-[140px] truncate font-mono text-[11px] font-medium text-foreground">
                    {att.name}
                  </div>
                  <div className="mt-0.5 flex items-center gap-1.5 text-[10px]">
                    <span className="text-muted-foreground">
                      {formatBytes(att.size)}
                      {att.durationSec ? ` • ${Math.round(att.durationSec)}s` : ''}
                    </span>
                    <span
                      className={cn(
                        'inline-flex items-center gap-1 rounded px-1.5 py-0 font-mono font-semibold',
                        att.tokenStatus === 'exact'
                          ? 'bg-emerald-500/15 text-emerald-400'
                          : 'bg-primary/15 text-primary'
                      )}
                      title={
                        att.tokenStatus === 'exact'
                          ? 'Jumlah token resmi dari Gemini countTokens API'
                          : 'Estimasi lokal berdasarkan resolusi/durasi/teks'
                      }
                    >
                      {att.tokenStatus === 'counting' && (
                        <Loader2 className="h-2.5 w-2.5 animate-spin" />
                      )}
                      {att.tokenStatus !== 'exact' && '~'}
                      {(att.tokenCount ?? 0).toLocaleString()} tok
                    </span>
                  </div>
                </div>

                <button
                  type="button"
                  onClick={() =>
                    setAttachments((prev) => prev.filter((item) => item.id !== att.id))
                  }
                  className="ml-1 rounded-full p-1 text-muted-foreground hover:bg-destructive hover:text-destructive-foreground"
                  aria-label="Hapus lampiran"
                >
                  <X className="h-3 w-3" />
                </button>
              </div>
            ))}
          </div>
        )}

        <StudioScroll ref={inputScrollRef} className="max-h-[185px] w-full" autoHide="scroll">
          <Textarea
            ref={textareaRef}
            rows={1}
            value={input}
            onChange={(e) => {
              setInput(e.target.value);
              adjustHeight();
            }}
            onKeyDown={handleKeyDown}
            onPaste={handlePaste}
            placeholder={
              visualTarget
                ? `Instruksi edit untuk <${visualTarget.selector}>...`
                : `Tanya ${model}...`
            }
            className="min-h-[42px] w-full resize-none overflow-hidden border-0 bg-transparent px-2 py-2 text-[15px] leading-relaxed shadow-none focus-visible:ring-0 focus-visible:ring-offset-0"
          />
        </StudioScroll>

        <div className="mt-1.5 flex items-center justify-between gap-1.5 border-t border-border/40 pt-2">
          <div className="flex flex-wrap items-center gap-1">
            <input
              ref={fileInputRef}
              type="file"
              accept="image/*,video/*,audio/*,application/pdf,.txt,.md,.ts,.tsx,.js,.jsx,.json,.py,.css,.html,.sql,.yaml,.yml,.sh,.csv"
              multiple
              className="hidden"
              onChange={handleFileChange}
            />

            <Button
              type="button"
              variant="ghost"
              size="icon"
              onClick={() => fileInputRef.current?.click()}
              disabled={isStreaming}
              className="h-8 w-8 rounded-xl text-muted-foreground hover:text-foreground sm:w-auto sm:px-2.5"
              title="Upload Gambar, Video, Audio, PDF, atau Kode"
            >
              <ImagePlus className="h-4 w-4" />
              <span className="hidden text-xs sm:inline">Lampiran</span>
            </Button>

            <Button
              type="button"
              variant={thinkingOn ? 'secondary' : 'ghost'}
              size="sm"
              disabled={isStreaming}
              onClick={() => {
                const next = !thinkingOn;
                onUpdateSettings({
                  ...settings,
                  includeThoughts: next,
                  thinkingBudget: next && settings.thinkingBudget === 0 ? -1 : settings.thinkingBudget,
                });
                toast.success(next ? 'Thinking Process: Aktif' : 'Thinking Process: Nonaktif');
              }}
              className={cn(
                'h-8 gap-1 rounded-xl px-2 text-xs',
                thinkingOn
                  ? 'border border-primary/35 bg-primary/10 text-primary hover:bg-primary/20'
                  : 'text-muted-foreground hover:text-foreground'
              )}
              title="Tampilkan proses thinking AI (ala Google AI Studio)"
            >
              <Brain className="h-3.5 w-3.5" />
              <span className="hidden sm:inline">Think</span>
            </Button>

            <Button
              type="button"
              variant="ghost"
              size="icon"
              onClick={() => {
                setInput('/');
                textareaRef.current?.focus();
              }}
              disabled={isStreaming}
              className="h-8 w-8 rounded-xl text-muted-foreground hover:text-foreground"
              title="Slash Templates (/)"
            >
              <Slash className="h-3.5 w-3.5" />
            </Button>

            {input.trim().length > 4 && !isStreaming && (
              <Button
                type="button"
                variant="ghost"
                size="sm"
                onClick={handleEnhancePrompt}
                className="h-8 gap-1 rounded-xl px-2 text-xs text-primary hover:bg-primary/10"
                title="Perkaya struktur prompt"
              >
                <Wand2 className="h-3.5 w-3.5" />
                <span className="hidden sm:inline">Enhance</span>
              </Button>
            )}

            {speechSupported && (
              <Button
                type="button"
                variant={isListening ? 'destructive' : 'ghost'}
                size="icon"
                onClick={toggleVoiceInput}
                disabled={isStreaming}
                className="h-8 w-8 rounded-xl"
                title={isListening ? 'Hentikan Voice Input' : 'Voice Input'}
              >
                {isListening ? (
                  <MicOff className="h-4 w-4 animate-pulse" />
                ) : (
                  <Mic className="h-4 w-4 text-muted-foreground" />
                )}
              </Button>
            )}

            <Popover>
              <PopoverTrigger asChild>
                <Button
                  type="button"
                  variant="ghost"
                  size="sm"
                  className="h-8 gap-1.5 rounded-xl px-2 text-xs text-muted-foreground hover:text-foreground"
                  title="Quick Tuning"
                >
                  <SlidersHorizontal className="h-3.5 w-3.5" />
                  <span className="hidden max-w-[95px] truncate sm:inline">
                    {activePersona?.name || 'Tuning'}
                  </span>
                </Button>
              </PopoverTrigger>
              <PopoverContent align="start" className="w-[calc(100vw-2rem)] max-w-xs space-y-4">
                <div className="space-y-1.5">
                  <div className="text-xs font-semibold">Quick Persona</div>
                  <div className="grid grid-cols-2 gap-1.5">
                    {PERSONA_PRESETS.map((p) => (
                      <button
                        key={p.id}
                        type="button"
                        onClick={() =>
                          onUpdateSettings({
                            ...settings,
                            activePersonaId: p.id,
                            systemInstruction: p.systemInstruction,
                            temperature: p.temperature,
                          })
                        }
                        className={cn(
                          'rounded-lg border px-2.5 py-1.5 text-left text-xs transition-colors',
                          settings.activePersonaId === p.id
                            ? 'border-primary bg-primary/10 font-medium text-primary'
                            : 'hover:bg-accent'
                        )}
                      >
                        {p.name}
                      </button>
                    ))}
                  </div>
                </div>

                <div className="space-y-2">
                  <div className="flex items-center justify-between text-xs">
                    <span className="flex items-center gap-1.5 font-medium">
                      <Brain className="h-3.5 w-3.5 text-primary" />
                      Thinking Budget
                    </span>
                    <Switch
                      checked={thinkingOn}
                      onCheckedChange={(checked) =>
                        onUpdateSettings({
                          ...settings,
                          includeThoughts: checked,
                          thinkingBudget: checked && settings.thinkingBudget === 0 ? -1 : settings.thinkingBudget,
                        })
                      }
                    />
                  </div>
                  {thinkingOn && (
                    <div className="grid grid-cols-5 gap-1">
                      {THINKING_BUDGET_OPTIONS.map((opt) => (
                        <button
                          key={opt.value}
                          type="button"
                          onClick={() =>
                            onUpdateSettings({
                              ...settings,
                              includeThoughts: true,
                              thinkingBudget: opt.value,
                            })
                          }
                          className={cn(
                            'rounded-md border py-1 font-mono text-[10px] transition-colors',
                            settings.thinkingBudget === opt.value
                              ? 'border-primary bg-primary/15 font-semibold text-primary'
                              : 'hover:bg-accent'
                          )}
                        >
                          {opt.label}
                        </button>
                      ))}
                    </div>
                  )}
                </div>

                <div className="space-y-2">
                  <div className="flex justify-between text-xs">
                    <span className="font-medium">Temperature</span>
                    <span className="font-mono text-primary">{settings.temperature.toFixed(1)}</span>
                  </div>
                  <Slider
                    min={0}
                    max={2}
                    step={0.1}
                    value={[settings.temperature]}
                    onValueChange={([val]) => onUpdateSettings({ ...settings, temperature: val })}
                  />
                </div>

                <div className="flex items-center justify-between rounded-lg border bg-muted/40 px-3 py-2">
                  <span className="text-xs font-medium">Demo SSE Mode</span>
                  <Switch
                    checked={settings.demoMode}
                    onCheckedChange={(checked) =>
                      onUpdateSettings({ ...settings, demoMode: checked })
                    }
                  />
                </div>
              </PopoverContent>
            </Popover>
          </div>

          <div className="flex items-center gap-2">
            {totalComposerTokens > 0 && (
              <span className="font-mono text-[10px] text-muted-foreground">
                {totalComposerTokens.toLocaleString()} tok
              </span>
            )}

            {isStreaming ? (
              <Button
                type="button"
                size="sm"
                variant="destructive"
                onClick={onStop}
                className="h-8 gap-1.5 rounded-xl px-3"
              >
                <Square className="h-3.5 w-3.5 fill-current" />
                <span>Stop</span>
              </Button>
            ) : (
              <Button
                type="button"
                size="icon"
                onClick={submit}
                disabled={!input.trim() && attachments.length === 0}
                className="h-8 w-8 rounded-xl shadow-md"
                aria-label="Kirim pesan"
              >
                <ArrowUp className="h-4 w-4" />
              </Button>
            )}
          </div>

  const pool = [];

function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) {
      reject(new DOMException('Aborted', 'AbortError'));
      return;
    }
    const timer = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();
    }, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

function buildOrderedCandidateKeys(keys) {
  const now = Date.now();
  const start = roundRobinCursor % keys.length;
  roundRobinCursor = (roundRobinCursor + 1) % keys.length;

  const rotated = [...keys.slice(start), ...keys.slice(0, start)];
  const ready = [];
  const cooling = [];
  for (const item of rotated) {
    const origIdx = keys.indexOf(item);
    const coolUntil = keyCooldownMap.get(item) || 0;
    if (coolUntil > now) {
      cooling.push({ key: item, idx: origIdx });
    } else {
      ready.push({ key: item, idx: origIdx });
    }
  }
  return [...ready, ...cooling];
}

async function fetchGeminiWithRotationAndRetry({
  targetModel,
  apiKeys,
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  signal,
  onStatus,
  onInfo,
}) {
  const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(
    targetModel
  )}:streamGenerateContent?alt=sse`;

  let wantsThinking = Boolean(includeThoughts) && Number(thinkingBudget) !== 0;
  const orderedKeys = buildOrderedCandidateKeys(apiKeys);
  const maxAttempts = Math.min(Math.max(orderedKeys.length * 2, 4), 10);

  let keyPointer = 0;
  let lastStatus = 503;
  let lastErrorMessage = 'Semua API Key sedang sibuk atau terkena limit.';

  for (let attempt = 0; attempt < maxAttempts; attempt++) {
    const current = orderedKeys[keyPointer % orderedKeys.length];
    const keyLabel = maskKeyLabel(current.key, current.idx);

    const upstream = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-goog-api-key': current.key,
      },
      body: JSON.stringify(
        buildGeminiPayload({
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          withThinkingConfig: wantsThinking,
        })
      ),
      signal,
    });

    if (upstream.ok) {
      if (attempt > 0 && orderedKeys.length > 1) {
        onInfo(`Berhasil terhubung menggunakan ${keyLabel}`);
      }
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    const errBody = await upstream.text();
    lastStatus = upstream.status;
    lastErrorMessage = extractGeminiError(errBody, upstream.status);

    if (upstream.status === 400 && wantsThinking && /thinking/i.test(errBody)) {
      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    const isBadKey =
      upstream.status === 401 ||
      upstream.status === 403 ||
      (upstream.status === 400 && /api.?key|expired|invalid/i.test(lastErrorMessage));

    const canRotateToAnotherKey = orderedKeys.length > 1;
    const isRetryable = RETRYABLE_STATUSES.has(upstream.status) || (isBadKey && canRotateToAnotherKey);

    if (isRetryable && attempt + 1 < maxAttempts) {
      const coolMs = isBadKey ? 120_000 : upstream.status === 429 ? 60_000 : 12_000;
      keyCooldownMap.set(current.key, Date.now() + coolMs);

      if (canRotateToAnotherKey) {
        keyPointer += 1;
        const nextCandidate = orderedKeys[keyPointer % orderedKeys.length];
        const nextLabel = maskKeyLabel(nextCandidate.key, nextCandidate.idx);
        const fastRotateDelay = ROTATE_IMMEDIATELY_STATUSES.has(upstream.status) || isBadKey ? 280 : 900;

        onStatus(
          `${keyLabel} terkena (${upstream.status}). Merotasi otomatis ke ${nextLabel} (${attempt + 1}/${maxAttempts})...`
        );
        await sleep(fastRotateDelay, signal);
        continue;
      }

      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${maxAttempts - 1}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: lastStatus,
      message: lastErrorMessage,
    };
  }

  return { ok: false, status: lastStatus, message: lastErrorMessage };
}

        content: '',
        createdAt: Date.now(),
        model: targetModel,
        isStreaming: true,
      };

      setConversations((prev) =>
        prev.map((conv) =>
          conv.id !== activeConversation.id
            ? conv
            : {
                ...conv,
                model: targetModel,
                updatedAt: Date.now(),
                messages: [...baseMessages, assistantPlaceholder],
              }
        )
      );

      void runStreamForConversation(
        activeConversation.id,
        targetModel,
        baseMessages,
        assistantMessageId
      );
    },
    [activeConversation, isStreamingActive, runStreamForConversation]
  );

  const handleEditUserMessage = useCallback(
    (messageId: string, newContent: string) => {
      if (!activeConversation || isStreamingActive) return;
      const idx = activeConversation.messages.findIndex((m) => m.id === messageId);
      if (idx === -1) return;

      const updatedMsg: ChatMessageType = {
        ...activeConversation.messages[idx],
        content: newContent,
      };
      const baseHistory = [...activeConversation.messages.slice(0, idx), updatedMsg];

      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: activeConversation.model,
        isStreaming: true,
      };

      setConversations((prev) =>
        prev.map((conv) =>
          conv.id !== activeConversation.id
            ? conv
            : {
                ...conv,
                updatedAt: Date.now(),
                messages: [...baseHistory, assistantPlaceholder],
              }
        )
      );

      void runStreamForConversation(
        activeConversation.id,
        activeConversation.model,
        baseHistory,
        assistantMessageId
      );
    },
    [activeConversation, isStreamingActive, runStreamForConversation]
  );

  const handleForkFromMessage = useCallback(
    (messageId: string) => {
      if (!activeConversation) return;
      const idx = activeConversation.messages.findIndex((m) => m.id === messageId);
      if (idx === -1) return;

      const forkedId = generateId();
      const forkedConv: Conversation = {
        id: forkedId,
        title: `Branch: ${activeConversation.title}`.slice(0, 52),
        model: activeConversation.model,
        messages: activeConversation.messages.slice(0, idx + 1).map((m) => ({
          ...m,
          id: generateId(),
        })),
        createdAt: Date.now(),
        updatedAt: Date.now(),
      };
      setConversations((prev) => [forkedConv, ...prev]);
      setActiveId(forkedId);
      toast.success('Cabang chat baru dibuat');
    },
    [activeConversation]
  );

  const handleToggleStar = useCallback((convId: string, msgId: string) => {
    setConversations((prev) =>
      prev.map((conv) =>
        conv.id !== convId
          ? conv
          : {
              ...conv,
              messages: conv.messages.map((m) =>
                m.id === msgId ? { ...m, starred: !m.starred } : m
              ),
            }
      )
    );
  }, []);

  const handleSelectModel = (modelId: string) => {
    setSelectedModel(modelId);
    if (activeConversation) {
      setConversations((prev) =>
        prev.map((c) => (c.id === activeConversation.id ? { ...c, model: modelId } : c))
      );
    }
    toast.success(`Model aktif: ${modelId}`);
  };

  const handleExportConversation = useCallback(
    (format: 'md' | 'json') => {
      if (!activeConversation) {
        toast.error('Tidak ada percakapan aktif untuk di-export');
        return;
      }
      let content = '';
      let mime = 'text/markdown;charset=utf-8';
      if (format === 'json') {
        content = JSON.stringify(activeConversation, null, 2);
        mime = 'application/json;charset=utf-8';
      } else {
        content = [
          `# ${activeConversation.title}`,
          `Model: \`${activeConversation.model}\`\n`,
          ...activeConversation.messages.map(
            (m) => `### ${m.role === 'user' ? 'User' : 'Assistant'}\n\n${m.content}\n`
          ),
        ].join('\n');
      }
      const blob = new Blob([content], { type: mime });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `chat-${activeConversation.id.slice(0, 8)}.${format}`;
      a.click();
      URL.revokeObjectURL(url);
      toast.success(`Exported .${format}`);
    },
    [activeConversation]
  );

  const handleDeleteConversation = useCallback(
    (id: string) => {
      if (activeStreamRef.current?.convId === id) handleStopStream();
      setConversations((prev) => {
        const next = prev.filter((c) => c.id !== id);
        if (activeId === id) setActiveId(next[0]?.id ?? null);
        return next;
      });
      toast.success('Percakapan dihapus');
    },
    [activeId, handleStopStream]
  );

  const handleRenameConversation = useCallback((id: string, title: string) => {
    setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title } : c)));
  }, []);

  const handleTogglePinConversation = useCallback((id: string) => {
    setConversations((prev) =>
      prev.map((c) => (c.id === id ? { ...c, pinned: !c.pinned } : c))
    );
  }, []);

  const lastAssistantId = useMemo(() => {
    const msgs = activeConversation?.messages ?? [];
    for (let i = msgs.length - 1; i >= 0; i--) {
      if (msgs[i].role === 'assistant') return msgs[i].id;
    }
    return null;
  }, [activeConversation]);

  const filteredStarterCards = useMemo(() => {
    if (welcomeTab === 'All') return STARTER_CARDS;
    return STARTER_CARDS.filter((c) => c.category === welcomeTab);
  }, [welcomeTab]);

  return (
    <MotionConfig reducedMotion="user">
      <TooltipProvider>
        <Toaster
          richColors
          position="bottom-right"
          theme={settings.skin === 'paper' ? 'light' : 'dark'}
        />

        <div className="flex h-[100dvh] w-full max-w-full overflow-hidden bg-background text-foreground">
          {!zenMode && (
            <Sidebar
              open={sidebarOpen}
              mobileOpen={mobileSidebarOpen}
              onToggleDesktop={() => setSidebarOpen((prev) => !prev)}
              onCloseMobile={() => setMobileSidebarOpen(false)}
              conversations={conversations}
              activeId={activeId}
              onNewChat={handleNewChat}
              onSelectConversation={(id) => {
                if (isStreamingActive) handleStopStream();
                setActiveId(id);
                setActiveArtifact(null);
                setArtifactFullscreen(false);
                setVisualTarget(null);
              }}
              onRenameConversation={handleRenameConversation}
              onTogglePinConversation={handleTogglePinConversation}
              onDeleteConversation={handleDeleteConversation}
              onOpenSettings={() => setSettingsDialogOpen(true)}
              onOpenCommandPalette={() => setCommandOpen(true)}
              onOpenBookmarks={() => setBookmarksOpen(true)}
              onCycleSkin={handleCycleSkin}
              showInstallButton={!isStandalone}
              onInstallApp={() => void handleInstallAction()}
            />
          )}

          <div
            className={cn(
              'flex min-w-0 max-w-full flex-1 flex-col overflow-hidden',
              activeArtifact && artifactFullscreen && 'hidden lg:hidden'
            )}
          >
            <header className="flex h-14 shrink-0 items-center justify-between gap-2 border-b bg-background/95 px-3 sm:px-4">
              <div className="flex min-w-0 items-center gap-1.5 sm:gap-2">
                {!zenMode && (
                  <Button
                    type="button"
                    variant="ghost"
                    size="icon"
                    onClick={() => setMobileSidebarOpen(true)}
                    className="shrink-0 md:hidden"
                    aria-label="Menu"
                  >
                    <Menu className="h-5 w-5" />
                  </Button>
                )}

                <Button
                  type="button"
                  variant="ghost"
                  onClick={() => setModelDialogOpen(true)}
                  className="h-9 min-w-0 gap-1.5 rounded-xl border border-border/50 bg-card/60 px-2.5 font-medium shadow-sm sm:gap-2 sm:px-3"
                >
                  <Sparkles className="h-4 w-4 shrink-0 text-primary" />
                  <span className="truncate text-xs sm:text-sm">{currentModelInfo.name}</span>
                  {currentModelInfo.isNew && (
                    <Badge
                      variant="secondary"
                      className="hidden bg-emerald-500/15 text-[10px] text-emerald-500 sm:inline-flex"
                    >
                      New
                    </Badge>
                  )}
                  <ChevronDown className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
                </Button>

                {isStreamingActive && (
                  <Badge variant="outline" className="hidden font-mono text-[10px] sm:inline-flex">
                    {activeStream.statusText
                      ? 'Rotating Key...'
                      : activeStream.tps > 0
                        ? `${activeStream.tps} tok/s`
                        : 'Thinking...'}
                  </Badge>
                )}
              </div>

              <div className="flex shrink-0 items-center gap-1">
                {!isOnline && (
                  <Badge variant="destructive" className="gap-1 px-2 py-0.5 text-[10px]">
                    <WifiOff className="h-3 w-3" />
                    <span className="hidden sm:inline">Offline</span>
                  </Badge>
                )}

                <Tooltip>
                  <TooltipTrigger asChild>
                    <div className="hidden items-center gap-2 rounded-full border bg-card/60 px-2.5 py-1 text-[11px] text-muted-foreground lg:flex">
                      <span className="font-mono">
                        {(contextTokenEstimate / 1000).toFixed(1)}k ctx
                      </span>
                      <div className="h-1.5 w-12 overflow-hidden rounded-full bg-muted">
                        <div
                          className="h-full rounded-full bg-primary transition-all"
                          style={{ width: `${contextUsagePct}%` }}
                        />
                      </div>
                    </div>
                  </TooltipTrigger>
                  <TooltipContent>
                    Estimasi context (~{contextTokenEstimate.toLocaleString()} /{' '}
                    {currentModelInfo.contextLimit.toLocaleString()} tokens)
                  </TooltipContent>
                </Tooltip>

                {conversationArtifacts.length > 0 && !activeArtifact && (
                  <Button
                    type="button"
                    variant="outline"
                    size="sm"
                    onClick={() =>
                      setActiveArtifact(conversationArtifacts[conversationArtifacts.length - 1])
                    }
                    className="h-8 gap-1 rounded-full px-2.5 text-xs"
                  >
                    <Eye className="h-3.5 w-3.5 text-primary" />
                    <span className="hidden sm:inline">
                      Canvas ({conversationArtifacts.length})
                    </span>
                  </Button>
                )}

                <Tooltip>
                  <TooltipTrigger asChild>
                    <button
                      type="button"
                      onClick={() => setSettingsDialogOpen(true)}
                      className="inline-flex items-center gap-1.5 rounded-full border bg-card/70 px-2 py-1 text-xs text-muted-foreground transition-colors hover:bg-accent hover:text-foreground sm:px-2.5"
                    >
                      <Radio
                        className={
                          settings.demoMode
                            ? 'h-3.5 w-3.5 text-amber-500'
                            : 'h-3.5 w-3.5 text-emerald-500'
                        }
                      />
                      <span className="hidden sm:inline">
                        {settings.demoMode
                          ? 'Demo SSE'
                          : activeKeyPoolCount > 1
                            ? `${activeKeyPoolCount} Keys`
                            : 'Live SSE'}
                      </span>
                    </button>
                  </TooltipTrigger>
                  <TooltipContent>
                    {settings.demoMode
                      ? 'Mode simulasi SSE lokal aktif'
                      : activeKeyPoolCount > 1
                        ? `Auto-Rotation Pool aktif (${activeKeyPoolCount} API keys)`
                        : 'Streaming SSE ke /api/chat'}
                  </TooltipContent>
                </Tooltip>

                <Tooltip>
                  <TooltipTrigger asChild>
                    <Button
                      type="button"
                      variant={zenMode ? 'secondary' : 'ghost'}
                      size="icon"
                      onClick={() => setZenMode((z) => !z)}
                      className="hidden sm:inline-flex"
                    >
                      {zenMode ? (
                        <Minimize2 className="h-4 w-4" />
                      ) : (
                        <Maximize2 className="h-4 w-4" />
                      )}
                    </Button>
                  </TooltipTrigger>
                  <TooltipContent>Zen Focus Mode (Ctrl+\)</TooltipContent>
                </Tooltip>

                <DropdownMenu>
                  <DropdownMenuTrigger asChild>
                    <Button type="button" variant="ghost" size="icon" aria-label="Menu studio">
                      <MoreVertical className="h-4 w-4" />
                    </Button>
                  </DropdownMenuTrigger>
                  <DropdownMenuContent align="end" className="w-52">
                    {!isStandalone && (
                      <>
                        <DropdownMenuItem onClick={() => void handleInstallAction()}>
                          <MonitorDown className="h-3.5 w-3.5 text-primary" />
                          Install Adam Studio App
                        </DropdownMenuItem>
                        <DropdownMenuSeparator />
                      </>
                    )}
                    <DropdownMenuItem onClick={handleCycleSkin}>
                      <Palette className="h-3.5 w-3.5" />
                      Skin: {settings.skin}
                    </DropdownMenuItem>
                    <DropdownMenuItem onClick={() => setCommandOpen(true)}>
                      <Download className="h-3.5 w-3.5" />
                      Command Palette (Ctrl+K)
                    </DropdownMenuItem>
                    {activeConversation && (
                      <>
                        <DropdownMenuSeparator />
                        <DropdownMenuItem onClick={() => handleExportConversation('md')}>
                          <FileText className="h-3.5 w-3.5" />
                          Export Markdown (.md)
                        </DropdownMenuItem>
                        <DropdownMenuItem onClick={() => handleExportConversation('json')}>
                          <FileJson className="h-3.5 w-3.5" />
                          Export JSON (.json)
                        </DropdownMenuItem>
                      </>
                    )}
                    <DropdownMenuSeparator />
                    <DropdownMenuItem onClick={() => setSettingsDialogOpen(true)}>
                      <Settings className="h-3.5 w-3.5" />
                      Settings & API Key Pool
                    </DropdownMenuItem>
                  </DropdownMenuContent>
                </DropdownMenu>
              </div>
            </header>

            <div className="relative min-h-0 flex-1 overflow-hidden">
              <StudioScroll
                ref={chatScrollRef}
                onScroll={handleViewportScroll}
                className="h-full w-full"
                autoHide="scroll"
              >
                {!activeConversation || activeConversation.messages.length === 0 ? (
                  <div className="relative mx-auto flex min-h-[calc(100dvh-13rem)] w-full max-w-3xl flex-col items-center justify-center px-4 py-8 text-center">
                    <div className="pointer-events-none absolute inset-0 -z-10 bg-dot-matrix opacity-35 [mask-image:radial-gradient(ellipse_at_center,black_30%,transparent_75%)]" />
                    <div className="pointer-events-none absolute top-10 -z-10 h-56 w-56 rounded-full bg-primary/10 blur-3xl" />

                    <motion.div
                      initial={{ scale: 0.92, opacity: 0 }}
                      animate={{ scale: 1, opacity: 1 }}
                      transition={{ duration: 0.24 }}
                      className="mb-4"
                    >
                      <BrandLogo size="lg" className="shadow-xl" />
                    </motion.div>

                    <div className="mb-2 inline-flex flex-wrap items-center justify-center gap-2 rounded-full border bg-card/90 px-3.5 py-1 text-xs text-muted-foreground shadow-sm">
                      <Cpu className="h-3.5 w-3.5 text-primary" />
                      <span className="font-mono font-semibold text-foreground">
                        {currentModelId}
                      </span>
                      {activeKeyPoolCount > 0 && (
                        <>
                          <span>•</span>
                          <span className="text-emerald-400">
                            {activeKeyPoolCount} API Key{activeKeyPoolCount > 1 ? 's Pool' : ''}
                          </span>
                        </>
                      )}
                      {isStandalone && (
                        <>
                          <span>•</span>
                          <span className="text-primary">App Mode</span>
                        </>
                      )}
                    </div>

                    <h1 className="mt-2 text-2xl font-bold tracking-tight text-foreground sm:text-4xl">
                      Adam Studio Workspace
                    </h1>
                    <p className="mt-2.5 max-w-lg text-sm leading-relaxed text-muted-foreground">
                      Platform AI dengan Live Thinking, Auto-Rotate Key Pool, Visual Element
                      Inspector, Token Counter per file, & Artifact Time-Machine Diff.
                    </p>

                    {activeKeyPoolCount === 0 && !settings.demoMode && (
                      <div className="mt-5 flex flex-wrap items-center justify-center gap-2 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-2.5 text-xs text-amber-400">
                        <span>API Key belum aktif.</span>
                        <button
                          type="button"
                          onClick={() => setSettingsDialogOpen(true)}
                          className="font-semibold underline underline-offset-4 hover:text-amber-300"
                        >
                          Isi API Key Pool atau aktifkan Demo SSE
                        </button>
                      </div>
                    )}

                    <div className="mt-6 inline-flex flex-wrap items-center justify-center gap-1 rounded-full border bg-card/80 p-1">
                      {WELCOME_TABS.map((tab) => (
                        <button
                          key={tab}
                          type="button"
                          onClick={() => setWelcomeTab(tab)}
                          className={cn(
                            'relative rounded-full px-3 py-1 text-xs font-medium transition-colors sm:px-3.5',
                            welcomeTab === tab
                              ? 'text-primary-foreground'
                              : 'text-muted-foreground hover:text-foreground'
                          )}
                        >
                          {welcomeTab === tab && (
                            <motion.span
                              layoutId="welcome-tab-pill"
                              transition={{ type: 'spring', stiffness: 380, damping: 30 }}
                              className="absolute inset-0 -z-10 rounded-full bg-primary shadow-sm"
                            />
                          )}
                          {tab}
                        </button>
                      ))}
                    </div>

                    <div className="mt-5 grid w-full grid-cols-1 gap-3 text-left sm:grid-cols-2">
                      {filteredStarterCards.map((item) => {
                        const Icon = item.icon;
                        return (
                          <button
                            key={item.title}
                            type="button"
                            onClick={() => handleSend(item.prompt, [])}
                            className="group rounded-2xl border bg-card/80 p-4 transition-all hover:-translate-y-0.5 hover:border-primary/50 hover:bg-accent/40 hover:shadow-lg"
                          >
                            <div className="flex items-center justify-between">
                              <div className="flex items-center gap-2 text-xs font-semibold text-foreground">
                                <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-primary/15 text-primary">
                                  <Icon className="h-3.5 w-3.5" />
                                </div>
                                <span>{item.title}</span>
                              </div>
                              <Badge variant="outline" className="text-[10px] font-normal">
                                {item.category}
                              </Badge>
                            </div>
                            <p className="mt-2.5 text-xs leading-relaxed text-muted-foreground">
                              {item.subtitle}
                            </p>
                          </button>
                        );
                      })}
                    </div>
                  </div>
                ) : (
                  <div
                    className={cn(
                      'mx-auto w-full min-w-0 max-w-3xl px-3 py-5 sm:px-5 sm:py-6',
                      settings.density === 'compact' ? 'space-y-4' : 'space-y-6'
                    )}
                  >
                    {activeConversation.messages.map((msg) => {
                      const effectiveMsg =
                        activeStream &&
                        activeStream.convId === activeConversation.id &&
                        activeStream.msgId === msg.id
                          ? {
                              ...msg,
                              content: activeStream.content,
                              thought: activeStream.thought || msg.thought,
                              statusText: activeStream.statusText,
                              metrics: activeStream.thinkingDurationMs
                                ? {
                                    ...msg.metrics,
                                    thinkingDurationMs: activeStream.thinkingDurationMs,
                                  }
                                : msg.metrics,
                              isStreaming: true,
                            }
                          : msg;

                      return (
                        <ChatMessage
                          key={msg.id}
                          message={effectiveMsg}
                          compact={settings.density === 'compact'}
                          thinkingEnabled={thinkingEnabled}
                          isLastAssistant={msg.id === lastAssistantId}
                          onRegenerate={
                            msg.id === lastAssistantId ? () => handleRegenerate() : undefined
                          }
                          onRetryWithModel={
                            msg.id === lastAssistantId
                              ? (mId) => handleRegenerate(mId)
                              : undefined
                          }
                          onEditUserMessage={
                            msg.role === 'user' && !isStreamingActive
                              ? handleEditUserMessage
                              : undefined
                          }
                          onForkFromMessage={
                            msg.role === 'assistant' && !isStreamingActive
                              ? handleForkFromMessage
                              : undefined
                          }
                          onToggleStar={(msgId) =>
                            handleToggleStar(activeConversation.id, msgId)
                          }
                          onOpenArtifact={setActiveArtifact}
                          onPreviewImage={setLightboxUrl}
                        />
                      );
                    })}
                  </div>
                )}
              </StudioScroll>

              {showScrollBottom && (
                <button
                  type="button"
                  onClick={() => {
                    autoScrollEnabledRef.current = true;
                    scrollToBottom(true);
                  }}
                  className={cn(
                    'absolute bottom-3 left-1/2 z-20 flex -translate-x-1/2 items-center gap-2 rounded-full border px-3.5 py-1.5 text-xs font-medium shadow-xl backdrop-blur-md transition-all hover:scale-105',
                    isStreamingActive
                      ? 'border-primary/40 bg-card/95 text-foreground'
                      : 'bg-secondary/95 text-secondary-foreground'
                  )}
                  aria-label="Scroll ke bawah"
                >
                  {isStreamingActive ? (
                    <>
                      <span className="h-2 w-2 animate-ping rounded-full bg-primary" />
                      <span>
                        {activeStream.statusText
                          ? 'Rotating Key...'
                          : activeStream.tps > 0
                            ? `Streaming ${activeStream.tps} tok/s`
                            : 'Sedang Thinking...'}
                      </span>
                      <ArrowDown className="h-3.5 w-3.5 text-primary" />
                    </>
                  ) : (
                    <ArrowDown className="h-4 w-4" />
                  )}
                </button>
              )}
            </div>

            <ChatComposer
              model={currentModelId}
              isStreaming={isStreamingActive}
              settings={settings}
              externalDraft={externalDraft}
              visualTarget={visualTarget}
              onClearVisualTarget={() => setVisualTarget(null)}
              onClearExternalDraft={() => setExternalDraft(null)}
              onUpdateSettings={setSettings}
              onSend={handleSend}
              onStop={handleStopStream}
              onPreviewImage={setLightboxUrl}
            />
          </div>

          <AnimatePresence>
            {activeArtifact && (
              <>
                {!artifactFullscreen && (
                  <div
                    onPointerDown={startResizingPane}
                    className="hidden w-1.5 shrink-0 cursor-col-resize items-center justify-center border-l bg-border/40 transition-colors hover:bg-primary/60 lg:flex"
                    title="Tarik untuk mengubah lebar Canvas"
                  />
                )}
                <motion.aside
                  initial={{ opacity: 0, x: 20 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 20 }}
                  transition={{ duration: 0.16 }}
                  className={cn(
                    'fixed inset-0 z-40 border-l lg:static',
                    artifactFullscreen ? 'lg:flex-1' : 'lg:shrink-0'
                  )}
                >
                  <div
                    style={
                      {
                        '--artifact-w': `${artifactWidthPct}vw`,
                      } as CSSProperties
                    }
                    className={cn(
                      'h-full w-full',
                      !artifactFullscreen &&
                        'lg:w-[var(--artifact-w)] lg:min-w-[360px] lg:max-w-[68vw]'
                    )}
                  >
                    <ArtifactPanel
                      artifact={activeArtifact}
                      allArtifacts={conversationArtifacts}
                      isResizing={isResizingPane}
                      isFullscreen={artifactFullscreen}
                      onToggleFullscreen={() => setArtifactFullscreen((f) => !f)}
                      onSelectArtifact={setActiveArtifact}
                      onSelectVisualTarget={(target) => {
                        setVisualTarget(target);
                        if (window.innerWidth < 1024) {
                          setActiveArtifact(null);
                          setArtifactFullscreen(false);
                        }
                      }}
                      onClose={() => {
                        setActiveArtifact(null);
                        setArtifactFullscreen(false);
                      }}
                    />
                  </div>
                </motion.aside>
              </>
            )}
          </AnimatePresence>

          <ModelSelectorDialog
            open={modelDialogOpen}
            onOpenChange={setModelDialogOpen}
            selectedModel={currentModelId}
            onSelectModel={handleSelectModel}
          />

          <SettingsDialog
            open={settingsDialogOpen}
            onOpenChange={setSettingsDialogOpen}
            settings={settings}
            serverHealth={serverHealth}
            onSave={setSettings}
          />

          <CommandPalette
            open={commandOpen}
            onOpenChange={setCommandOpen}
            conversations={conversations}
            activeSkin={settings.skin}
            onNewChat={handleNewChat}
            onSelectConversation={(id) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(id);
              setActiveArtifact(null);
              setArtifactFullscreen(false);
            }}
            onOpenModelSelector={() => setModelDialogOpen(true)}
            onOpenSettings={() => setSettingsDialogOpen(true)}
            onOpenBookmarks={() => setBookmarksOpen(true)}
            onToggleZenMode={() => setZenMode((z) => !z)}
            onCycleSkin={handleCycleSkin}
            onExportMarkdown={() => handleExportConversation('md')}
            onInsertTemplate={(tpl) => setExternalDraft(tpl)}
            showInstallOption={!isStandalone}
            onInstallApp={() => void handleInstallAction()}
          />

          <BookmarksDialog
            open={bookmarksOpen}
            onOpenChange={setBookmarksOpen}
            conversations={conversations}
            onJumpToConversation={(convId) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(convId);
            }}
            onToggleStar={handleToggleStar}
          />

          <InstallAppDialog
            open={installDialogOpen}
            onOpenChange={setInstallDialogOpen}
            isIos={isIos}
            canNativeInstall={canNativeInstall}
            onNativeInstall={() => {
              void handleInstallAction().then(() => setInstallDialogOpen(false));
            }}
          />

          <Dialog open={Boolean(lightboxUrl)} onOpenChange={() => setLightboxUrl(null)}>
            <DialogContent className="max-w-4xl border-0 bg-transparent p-2 shadow-none">
              <DialogHeader className="sr-only">
                <DialogTitle>Image Preview</DialogTitle>
              </DialogHeader>
              {lightboxUrl && (
                <img
                  src={lightboxUrl}
                  alt="Attachment Preview"
                  className="max-h-[85dvh] w-full rounded-2xl object-contain shadow-2xl"
                />
              )}
            </DialogContent>
          </Dialog>
        </div>
      </TooltipProvider>
    </MotionConfig>
  );
}

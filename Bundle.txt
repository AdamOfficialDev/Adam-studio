export interface DiffLine {
  type: 'same' | 'add' | 'remove';
  line: string;
  oldLineNum?: number;
  newLineNum?: number;
}

export function computeLineDiff(oldText: string, newText: string): DiffLine[] {
  const oldLines = oldText.split('\n');
  const newLines = newText.split('\n');

  // Guard untuk file super besar agar UI tetap instan
  if (oldLines.length * newLines.length > 450_000) {
    const result: DiffLine[] = [];
    const maxLen = Math.max(oldLines.length, newLines.length);
    for (let i = 0; i < maxLen; i++) {
      if (i < oldLines.length && i < newLines.length) {
        if (oldLines[i] === newLines[i]) {
          result.push({ type: 'same', line: newLines[i], oldLineNum: i + 1, newLineNum: i + 1 });
        } else {
          result.push({ type: 'remove', line: oldLines[i], oldLineNum: i + 1 });
          result.push({ type: 'add', line: newLines[i], newLineNum: i + 1 });
        }
      } else if (i < oldLines.length) {
        result.push({ type: 'remove', line: oldLines[i], oldLineNum: i + 1 });
      } else if (i < newLines.length) {
        result.push({ type: 'add', line: newLines[i], newLineNum: i + 1 });
      }
    }
    return result;
  }

  const m = oldLines.length;
  const n = newLines.length;
  const dp: number[][] = Array.from({ length: m + 1 }, () => new Array<number>(n + 1).fill(0));

  for (let i = m - 1; i >= 0; i--) {
    for (let j = n - 1; j >= 0; j--) {
      if (oldLines[i] === newLines[j]) {
        dp[i][j] = dp[i + 1][j + 1] + 1;
      } else {
        dp[i][j] = dp[i + 1][j] >= dp[i][j + 1] ? dp[i + 1][j] : dp[i][j + 1];
      }
    }
  }

  const diff: DiffLine[] = [];
  let i = 0;
  let j = 0;
  while (i < m && j < n) {
    if (oldLines[i] === newLines[j]) {
      diff.push({ type: 'same', line: oldLines[i], oldLineNum: i + 1, newLineNum: j + 1 });
      i += 1;
      j += 1;
    } else if (dp[i + 1][j] >= dp[i][j + 1]) {
      diff.push({ type: 'remove', line: oldLines[i], oldLineNum: i + 1 });
      i += 1;
    } else {
      diff.push({ type: 'add', line: newLines[j], newLineNum: j + 1 });
      j += 1;
    }
  }

  while (i < m) {
    diff.push({ type: 'remove', line: oldLines[i], oldLineNum: i + 1 });
    i += 1;
  }
  while (j < n) {
    diff.push({ type: 'add', line: newLines[j], newLineNum: j + 1 });
    j += 1;
  }

  return diff;
}

export function playCompletionChime(): void {
  try {
    const AudioCtx =
      window.AudioContext ||
      (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    if (!AudioCtx) return;

    const ctx = new AudioCtx();
    const now = ctx.currentTime;

    const playNote = (freq: number, startOffset: number, duration: number) => {
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      osc.type = 'sine';
      osc.frequency.setValueAtTime(freq, now + startOffset);

      gain.gain.setValueAtTime(0.0001, now + startOffset);
      gain.gain.exponentialRampToValueAtTime(0.06, now + startOffset + 0.02);
      gain.gain.exponentialRampToValueAtTime(0.0001, now + startOffset + duration);

      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start(now + startOffset);
      osc.stop(now + startOffset + duration);
    };

    // Chime harmonik dua nada lembut (C6 -> G6)
    playNote(1046.5, 0, 0.22);
    playNote(1567.98, 0.11, 0.32);

    window.setTimeout(() => {
      void ctx.close().catch(() => {});
    }, 600);
  } catch {
    // Abaikan jika browser memblokir AudioContext
  }
}

export interface StreamHandlers {
  signal: AbortSignal;
  onThought: (text: string) => void;
  onDelta: (text: string) => void;
  onUsage: (usage: TokenUsage) => void;
  onStatus?: (message: string) => void;
  onInfo?: (message: string) => void;
  onDone: (meta: { model: string; finishReason?: string; isDemo?: boolean }) => void;
  onError: (message: string) => void;
}

export async function streamChatCompletion(
  payload: StreamChatPayload,
  handlers: StreamHandlers
): Promise<void> {
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
  };

  // Normalisasi multi-line API key menjadi satu baris koma-separated agar valid sebagai HTTP header
  const normalizedKeys = (payload.apiKey || '')
    .split(/[\r\n,;]+/)
    .map((k) => k.trim())
    .filter(Boolean);

  if (normalizedKeys.length > 0) {
    headers['x-gemini-api-key'] = normalizedKeys.join(',');
  }

  const response = await fetch('/api/chat', {
    method: 'POST',
    headers,
    body: JSON.stringify({
      model: payload.model,
      messages: payload.messages.map((m) => ({
        role: m.role,
        content: m.content,
        attachments: m.attachments?.map((a) => ({
          kind: a.kind,
          name: a.name,
          mimeType: a.mimeType,
          data: a.data,
          textContent: a.textContent,
        })),
      })),
      systemInstruction: payload.systemInstruction,
      temperature: payload.temperature,
      maxOutputTokens: payload.maxOutputTokens,
      includeThoughts: payload.includeThoughts,
      thinkingBudget: payload.thinkingBudget,
      autoFallbackOnOverload: payload.autoFallbackOnOverload,
      fallbackModel: payload.fallbackModel,
      demoMode: payload.demoMode,
    }),
    signal: handlers.signal,
  });

  if (!response.ok) {
    const errText = await response.text();
    handlers.onError(errText || `HTTP Error ${response.status}`);
    return;
  }

  if (!response.body) {
    handlers.onError('Browser tidak menerima ReadableStream dari server.');
    return;
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  while (true) {
    const { value, done } = await reader.read();
    if (done) break;

    buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');

    let boundaryIndex: number;
    while ((boundaryIndex = buffer.indexOf('\n\n')) !== -1) {
      const rawBlock = buffer.slice(0, boundaryIndex);
      buffer = buffer.slice(boundaryIndex + 2);

      if (!rawBlock.trim() || rawBlock.startsWith(':')) continue;

      let eventName = 'message';
      const dataLines: string[] = [];

      for (const line of rawBlock.split('\n')) {
        if (line.startsWith('event:')) {
          eventName = line.slice(6).trim();
        } else if (line.startsWith('data:')) {
          dataLines.push(line.slice(5).trimStart());
        }
      }

      if (dataLines.length === 0) continue;

      let parsed: Record<string, unknown>;
      try {
        parsed = JSON.parse(dataLines.join('\n'));
      } catch {
        continue;
      }

      if (eventName === 'thought' && typeof parsed.text === 'string') {
        handlers.onThought(parsed.text);
      } else if (eventName === 'delta' && typeof parsed.text === 'string') {
        handlers.onDelta(parsed.text);
      } else if (eventName === 'usage') {
        handlers.onUsage(parsed as TokenUsage);
      } else if (eventName === 'status' && typeof parsed.message === 'string') {
        handlers.onStatus?.(parsed.message);
      } else if (eventName === 'info' && typeof parsed.message === 'string') {
        handlers.onInfo?.(parsed.message);
      } else if (eventName === 'done') {
        handlers.onDone({
          model: String(parsed.model ?? payload.model),
          finishReason: typeof parsed.finishReason === 'string' ? parsed.finishReason : undefined,
          isDemo: Boolean(parsed.isDemo),
        });
      } else if (eventName === 'error') {
        handlers.onError(
          typeof parsed.message === 'string'
            ? parsed.message
            : 'Terjadi kesalahan pada stream SSE.'
        );
      }
    }
  }
}

const PREVIEWABLE_LANGS = new Set(['html', 'htm', 'svg', 'markdown', 'md', 'mermaid']);

export function detectArtifactKind(language: string): ArtifactItem['kind'] {
  const norm = language.toLowerCase();
  if (norm === 'html' || norm === 'htm') return 'html';
  if (norm === 'svg') return 'svg';
  if (norm === 'markdown' || norm === 'md') return 'markdown';
  if (norm === 'mermaid') return 'mermaid';
  return 'code';
}

export function isArtifactCandidate(language: string, code: string): boolean {
  const norm = language.toLowerCase();
  if (PREVIEWABLE_LANGS.has(norm) && code.trim().length > 20) return true;
  return code.split('\n').length >= 8;
}

export function extractArtifactsFromMarkdown(messageId: string, markdown: string): ArtifactItem[] {
  const regex = /```([\w-]*)\n([\s\S]*?)```/g;
  const items: ArtifactItem[] = [];
  let match: RegExpExecArray | null;
  let index = 0;

  while ((match = regex.exec(markdown)) !== null) {
    const language = (match[1] || 'text').toLowerCase();
    const content = match[2].replace(/\n$/, '');
    if (!isArtifactCandidate(language, content)) continue;

    index += 1;
    const kind = detectArtifactKind(language);
    const titleMap: Record<ArtifactItem['kind'], string> = {
      html: `HTML App #${index}`,
      svg: `SVG Graphic #${index}`,
      markdown: `Document #${index}`,
      mermaid: `Diagram #${index}`,
      code: `${language.toUpperCase()} Module #${index}`,
    };

    items.push({
      id: `${messageId}-artifact-${index}`,
      title: titleMap[kind],
      language,
      content,
      kind,
    });
  }

  return items;
}

      }
    }
  }
  return pool;

function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) {
      reject(new DOMException('Aborted', 'AbortError'));
      return;
    }
    const timer = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();

    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

function buildGeminiContents(messages) {
  return messages.map((msg) => {
    const parts = [];
    if (msg.content && msg.content.trim().length > 0) {
      parts.push({ text: msg.content });
    }
    if (Array.isArray(msg.attachments)) {
      for (const att of msg.attachments) {
        if (att?.kind === 'text' && typeof att?.textContent === 'string') {
          parts.push({
            text: `\n\n[Attached File: ${att.name}]\n\`\`\`\n${att.textContent}\n\`\`\``,
          });
        } else if (att?.mimeType && att?.data) {
          parts.push({
            inlineData: {
              mimeType: att.mimeType,
              data: att.data,
            },
          });
        }
      }
    }
    if (parts.length === 0) {
      parts.push({ text: ' ' });
    }
    return {
      role: msg.role === 'assistant' ? 'model' : 'user',
      parts,
    };
  });
}

function buildGeminiPayload({
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,

  const generationConfig = {

function extractGeminiError(rawText, status) {
  let msg = `Gemini API error (${status})`;

function buildOrderedCandidateKeys(keys) {
  const now = Date.now();
  const start = roundRobinCursor % keys.length;
  roundRobinCursor = (roundRobinCursor + 1) % keys.length;

  const rotated = [...keys.slice(start), ...keys.slice(0, start)];
  const ready = [];
  const cooling = [];
  for (const item of rotated) {
    const origIdx = keys.indexOf(item);
    const coolUntil = keyCooldownMap.get(item) || 0;
    if (coolUntil > now) {
      cooling.push({ key: item, idx: origIdx });
    } else {
      ready.push({ key: item, idx: origIdx });
    }
  }
  return [...ready, ...cooling];
}

async function fetchGeminiWithRotationAndRetry({
  targetModel,
  apiKeys,
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  signal,
  onStatus,
  onInfo,
}) {
  const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(
    targetModel
  )}:streamGenerateContent?alt=sse`;

  let wantsThinking = Boolean(includeThoughts) && Number(thinkingBudget) !== 0;
  const orderedKeys = buildOrderedCandidateKeys(apiKeys);
  const maxAttempts = Math.min(Math.max(orderedKeys.length * 2, 4), 10);

  let keyPointer = 0;
  let lastStatus = 503;
  let lastErrorMessage = 'Semua API Key sedang sibuk atau terkena limit.';

  for (let attempt = 0; attempt < maxAttempts; attempt++) {
    const current = orderedKeys[keyPointer % orderedKeys.length];
    const keyLabel = maskKeyLabel(current.key, current.idx);

    const upstream = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-goog-api-key': current.key,
      },
      body: JSON.stringify(
        buildGeminiPayload({
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          withThinkingConfig: wantsThinking,
        })
      ),
      signal,
    });

    if (upstream.ok) {
      if (attempt > 0 && orderedKeys.length > 1) {
        onInfo(`Berhasil terhubung menggunakan ${keyLabel}`);
      }
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    const errBody = await upstream.text();
    lastStatus = upstream.status;
    lastErrorMessage = extractGeminiError(errBody, upstream.status);

    // Fallback jika model menolak parameter thinkingConfig
    if (upstream.status === 400 && wantsThinking && /thinking/i.test(errBody)) {
      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    const isBadKey =
      upstream.status === 401 ||
      upstream.status === 403 ||
      (upstream.status === 400 && /api.?key|expired|invalid/i.test(lastErrorMessage));

    const canRotateToAnotherKey = orderedKeys.length > 1;
    const isRetryable = RETRYABLE_STATUSES.has(upstream.status) || (isBadKey && canRotateToAnotherKey);

    if (isRetryable && attempt + 1 < maxAttempts) {
      // Tandai cooldown untuk key yang baru gagal
      const coolMs = isBadKey ? 120_000 : upstream.status === 429 ? 60_000 : 12_000;
      keyCooldownMap.set(current.key, Date.now() + coolMs);

      if (canRotateToAnotherKey) {
        keyPointer += 1;
        const nextCandidate = orderedKeys[keyPointer % orderedKeys.length];
        const nextLabel = maskKeyLabel(nextCandidate.key, nextCandidate.idx);
        const fastRotateDelay = ROTATE_IMMEDIATELY_STATUSES.has(upstream.status) || isBadKey ? 280 : 900;

        onStatus(
          `${keyLabel} terkena (${upstream.status}). Merotasi otomatis ke ${nextLabel} (${attempt + 1}/${maxAttempts})...`
        );
        await sleep(fastRotateDelay, signal);
        continue;
      }

      // Single-key exponential backoff
      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${maxAttempts - 1}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: lastStatus,
      message: lastErrorMessage,
    };
  }

  return { ok: false, status: lastStatus, message: lastErrorMessage };
}

  const orderedKeys = buildOrderedCandidateKeys(apiKeys);
  const maxAttempts = Math.min(Math.max(orderedKeys.length * 2, 4), 10);

  let keyPointer = 0;
  let lastStatus = 503;
  let lastErrorMessage = 'Semua API Key sedang sibuk atau terkena limit.';

  for (let attempt = 0; attempt < maxAttempts; attempt++) {
    const current = orderedKeys[keyPointer % orderedKeys.length];
    const keyLabel = maskKeyLabel(current.key, current.idx);

    const upstream = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-goog-api-key': current.key,
      },
      body: JSON.stringify(
        buildGeminiPayload({
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          withThinkingConfig: wantsThinking,
        })
      ),
      signal,
    });

    if (upstream.ok) {
      if (attempt > 0 && orderedKeys.length > 1) {
        onInfo(`Berhasil terhubung menggunakan ${keyLabel}`);
      }
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    const errBody = await upstream.text();
    lastStatus = upstream.status;
    lastErrorMessage = extractGeminiError(errBody, upstream.status);

    if (upstream.status === 400 && wantsThinking && /thinking/i.test(errBody)) {
      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    const isBadKey =
      upstream.status === 401 ||
      upstream.status === 403 ||
      (upstream.status === 400 && /api.?key|expired|invalid/i.test(lastErrorMessage));

    const canRotateToAnotherKey = orderedKeys.length > 1;
    const isRetryable = RETRYABLE_STATUSES.has(upstream.status) || (isBadKey && canRotateToAnotherKey);

    if (isRetryable && attempt + 1 < maxAttempts) {
      const coolMs = isBadKey ? 120_000 : upstream.status === 429 ? 60_000 : 12_000;
      keyCooldownMap.set(current.key, Date.now() + coolMs);

      if (canRotateToAnotherKey) {
        keyPointer += 1;
        const nextCandidate = orderedKeys[keyPointer % orderedKeys.length];
        const nextLabel = maskKeyLabel(nextCandidate.key, nextCandidate.idx);
        const fastRotateDelay = ROTATE_IMMEDIATELY_STATUSES.has(upstream.status) || isBadKey ? 280 : 900;

        onStatus(
          `${keyLabel} terkena (${upstream.status}). Merotasi otomatis ke ${nextLabel} (${attempt + 1}/${maxAttempts})...`
        );
        await sleep(fastRotateDelay, signal);
        continue;
      }

      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${maxAttempts - 1}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: lastStatus,
      message: lastErrorMessage,
    };
  }

  return { ok: false, status: lastStatus, message: lastErrorMessage };
}

              // Abaikan JSON parsial
            }
          }
        }

          finishReason: lastFinishReason || 'STOP',
          isDemo: false,
        });

          message: err instanceof Error ? err.message : 'Internal Edge Function Error',
        });
      } finally {
        try {
          controller.close();
        } catch {
          // Abaikan
        }
      }
    },
  });

  return new Response(stream, {

      'Content-Type': 'text/event-stream; charset=utf-8',
      'Cache-Control': 'no-cache, no-transform',
      'X-Accel-Buffering': 'no',
    },
  });

              toast.info(infoMsg);
            },
            onDone: ({ model: resolvedModel, isDemo }) => {
              if (flushTimerRef.current !== null) {
                window.clearTimeout(flushTimerRef.current);
                flushTimerRef.current = null;
              }
              activeStreamRef.current = null;
              setActiveStream(null);

              const finishedAt = performance.now();
              const durationMs = Math.max(1, Math.round(finishedAt - startedAt));
              const firstAny = firstAnswerAt ?? firstThoughtAt;
              const ttftMs = firstAny !== null ? Math.round(firstAny - startedAt) : durationMs;

              if (thinkingDurationMs === undefined && accThought) {
                thinkingDurationMs = Math.max(
                  100,
                  Math.round(finishedAt - (firstThoughtAt ?? startedAt))
                );
              }

              const answerWindow = Math.max(1, finishedAt - (firstAnswerAt ?? startedAt));
              const estTokens = Math.max(chunkCount, Math.round(accContent.length / 4));
              const tokensPerSec = Number(((estTokens / answerWindow) * 1000).toFixed(1));

              if (settings.soundChime && durationMs >= 2500) {
                playCompletionChime();
              }

              setConversations((prev) =>
                prev.map((conv) =>
                  conv.id !== convId
                    ? conv
                    : {
                        ...conv,
                        updatedAt: Date.now(),
                        messages: conv.messages.map((m) =>
                          m.id === assistantMessageId
                            ? {
                                ...m,
                                model: resolvedModel || m.model,
                                content: accContent,
                                thought: accThought || undefined,
                                statusText: undefined,
                                usage: accUsage,
                                isStreaming: false,
                                isDemo,
                                metrics: {
                                  ttftMs,
                                  thinkingDurationMs,
                                  durationMs,
                                  tokensPerSec,
                                },
                              }
                            : m
                        ),
                      }
                )
              );

              if (settings.autoOpenArtifacts && accContent) {
                const detected = extractArtifactsFromMarkdown(assistantMessageId, accContent);
                const previewable = detected.find(
                  (a) => a.kind === 'html' || a.kind === 'svg' || a.kind === 'mermaid'
                );
                if (previewable) setActiveArtifact(previewable);
              }
            },
            onError: (errorMessage) => {
              if (flushTimerRef.current !== null) {
                window.clearTimeout(flushTimerRef.current);
                flushTimerRef.current = null;
              }
              activeStreamRef.current = null;
              setActiveStream(null);
              toast.error(errorMessage);
              setConversations((prev) =>
                prev.map((conv) =>
                  conv.id !== convId
                    ? conv
                    : {
                        ...conv,
                        messages: conv.messages.map((m) =>
                          m.id === assistantMessageId
                            ? {
                                ...m,
                                content: accContent,
                                thought: accThought || undefined,
                                statusText: undefined,
                                isStreaming: false,
                                error: errorMessage,
                              }
                            : m
                        ),
                      }
                )
              );
            },
          }
        );
      } catch (err) {
        if (err instanceof Error && err.name === 'AbortError') return;
        const message = err instanceof Error ? err.message : 'Gagal terhubung ke server SSE.';
        toast.error(message);
        activeStreamRef.current = null;
        setActiveStream(null);
        setConversations((prev) =>
          prev.map((conv) =>
            conv.id !== convId
              ? conv
              : {
                  ...conv,
                  messages: conv.messages.map((m) =>
                    m.id === assistantMessageId
                      ? {
                          ...m,
                          content: accContent,
                          statusText: undefined,
                          isStreaming: false,
                          error: message,
                        }
                      : m
                  ),
                }
          )
        );
      } finally {
        if (abortControllerRef.current === controller) {
          abortControllerRef.current = null;
          if (activeStreamRef.current?.msgId === assistantMessageId) {
            activeStreamRef.current = null;
            setActiveStream(null);
            setConversations((prev) =>
              prev.map((conv) =>
                conv.id !== convId
                  ? conv
                  : {
                      ...conv,
                      messages: conv.messages.map((m) =>
                        m.id === assistantMessageId && m.isStreaming
                          ? {
                              ...m,
                              content: accContent,
                              thought: accThought || undefined,
                              statusText: undefined,
                              isStreaming: false,
                            }
                          : m
                      ),
                    }
              )
            );
          }
        }
      }
    },
    [settings]
  );

    (text: string, attachments: Attachment[]) => {
      if (isStreamingActive) return;

      const composedText = visualTarget
        ? `[Visual Target Element: \`${visualTarget.selector}\`]\n\`\`\`html\n${visualTarget.outerHtml}\n\`\`\`\n\n${text}`
        : text;

      const userMessage: ChatMessageType = {
        id: generateId(),
        role: 'user',
        content: composedText,

          prev.map((conv) =>
            conv.id !== activeConversation.id
              ? conv
              : {
                  ...conv,
                  updatedAt: Date.now(),
                  messages: [...conv.messages, userMessage, assistantPlaceholder],
                }
          )
        );
        void runStreamForConversation(
          activeConversation.id,
          currentModelId,
          historyForModel,
          assistantMessageId
        );
      }
    },
    [activeConversation, currentModelId, isStreamingActive, runStreamForConversation, visualTarget]
  );

        targetModel,

        assistantMessageId

    [activeConversation, isStreamingActive, runStreamForConversation]

                            <div className="flex items-center justify-between">
                              <div className="flex items-center gap-2 text-xs font-semibold text-foreground">
                                <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-primary/15 text-primary">
                                  <Icon className="h-3.5 w-3.5" />
                                </div>
                                <span>{item.title}</span>
                              </div>
                              <Badge variant="outline" className="text-[10px] font-normal">
                                {item.category}
                              </Badge>
                            </div>
                            <p className="mt-2.5 text-xs leading-relaxed text-muted-foreground">
                              {item.subtitle}
                            </p>
                          </button>
                        );
                      })}
                    </div>
                  </div>
                ) : (
                  <div
                    className={cn(
                      'mx-auto w-full min-w-0 max-w-3xl px-3 py-5 sm:px-5 sm:py-6',
                      settings.density === 'compact' ? 'space-y-4' : 'space-y-6'
                    )}
                  >
                    {activeConversation.messages.map((msg) => {
                      const effectiveMsg =
                        activeStream &&
                        activeStream.convId === activeConversation.id &&
                        activeStream.msgId === msg.id
                          ? {
                              ...msg,
                              content: activeStream.content,
                              thought: activeStream.thought || msg.thought,
                              statusText: activeStream.statusText,
                              metrics: activeStream.thinkingDurationMs
                                ? {
                                    ...msg.metrics,
                                    thinkingDurationMs: activeStream.thinkingDurationMs,
                                  }
                                : msg.metrics,
                              isStreaming: true,
                            }
                          : msg;

                      return (
                        <ChatMessage
                          key={msg.id}
                          message={effectiveMsg}
                          compact={settings.density === 'compact'}
                          thinkingEnabled={thinkingEnabled}
                          isLastAssistant={msg.id === lastAssistantId}
                          onRegenerate={
                            msg.id === lastAssistantId ? () => handleRegenerate() : undefined
                          }
                          onRetryWithModel={
                            msg.id === lastAssistantId
                              ? (mId) => handleRegenerate(mId)
                              : undefined
                          }
                          onEditUserMessage={
                            msg.role === 'user' && !isStreamingActive
                              ? handleEditUserMessage
                              : undefined
                          }
                          onForkFromMessage={
                            msg.role === 'assistant' && !isStreamingActive
                              ? handleForkFromMessage
                              : undefined
                          }
                          onToggleStar={(msgId) =>
                            handleToggleStar(activeConversation.id, msgId)
                          }
                          onOpenArtifact={setActiveArtifact}
                          onPreviewImage={setLightboxUrl}
                        />
                      );
                    })}
                  </div>
                )}
              </StudioScroll>

              {/* Dynamic Live Pill saat scroll ke atas, atau ArrowDown saat idle */}
              {showScrollBottom && (
                <button
                  type="button"
                  onClick={() => {
                    autoScrollEnabledRef.current = true;
                    scrollToBottom(true);
                  }}
                  className={cn(
                    'absolute bottom-3 left-1/2 z-20 flex -translate-x-1/2 items-center gap-2 rounded-full border px-3.5 py-1.5 text-xs font-medium shadow-xl backdrop-blur-md transition-all hover:scale-105',
                    isStreamingActive
                      ? 'border-primary/40 bg-card/95 text-foreground'
                      : 'bg-secondary/95 text-secondary-foreground'
                  )}
                  aria-label="Scroll ke bawah"
                >
                  {isStreamingActive ? (
                    <>
                      <span className="h-2 w-2 animate-ping rounded-full bg-primary" />
                      <span>
                        {activeStream.statusText
                          ? 'Rotating Key...'
                          : activeStream.tps > 0
                            ? `Streaming ${activeStream.tps} tok/s`
                            : 'Sedang Thinking...'}
                      </span>
                      <ArrowDown className="h-3.5 w-3.5 text-primary" />
                    </>
                  ) : (
                    <ArrowDown className="h-4 w-4" />
                  )}
                </button>
              )}
            </div>

            <ChatComposer
              model={currentModelId}
              isStreaming={isStreamingActive}
              settings={settings}
              externalDraft={externalDraft}
              visualTarget={visualTarget}
              onClearVisualTarget={() => setVisualTarget(null)}
              onClearExternalDraft={() => setExternalDraft(null)}
              onUpdateSettings={setSettings}
              onSend={handleSend}
              onStop={handleStopStream}
              onPreviewImage={setLightboxUrl}
            />
          </div>

          <AnimatePresence>
            {activeArtifact && (
              <>
                {!artifactFullscreen && (
                  <div
                    onPointerDown={startResizingPane}
                    className="hidden w-1.5 shrink-0 cursor-col-resize items-center justify-center border-l bg-border/40 transition-colors hover:bg-primary/60 lg:flex"
                    title="Tarik untuk mengubah lebar Canvas"
                  />
                )}
                <motion.aside
                  initial={{ opacity: 0, x: 20 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 20 }}
                  transition={{ duration: 0.16 }}
                  className={cn(
                    'fixed inset-0 z-40 border-l lg:static',
                    artifactFullscreen ? 'lg:flex-1' : 'lg:shrink-0'
                  )}
                >
                  <div
                    style={
                      {
                        '--artifact-w': `${artifactWidthPct}vw`,
                      } as CSSProperties
                    }
                    className={cn(
                      'h-full w-full',
                      !artifactFullscreen &&
                        'lg:w-[var(--artifact-w)] lg:min-w-[360px] lg:max-w-[68vw]'
                    )}
                  >
                    <ArtifactPanel
                      artifact={activeArtifact}
                      allArtifacts={conversationArtifacts}
                      isResizing={isResizingPane}
                      isFullscreen={artifactFullscreen}
                      onToggleFullscreen={() => setArtifactFullscreen((f) => !f)}
                      onSelectArtifact={setActiveArtifact}
                      onSelectVisualTarget={(target) => {
                        setVisualTarget(target);
                        if (window.innerWidth < 1024) {
                          setActiveArtifact(null);
                          setArtifactFullscreen(false);
                        }
                      }}
                      onClose={() => {
                        setActiveArtifact(null);
                        setArtifactFullscreen(false);
                      }}
                    />
                  </div>
                </motion.aside>
              </>
            )}
          </AnimatePresence>

          <ModelSelectorDialog
            open={modelDialogOpen}
            onOpenChange={setModelDialogOpen}
            selectedModel={currentModelId}
            onSelectModel={handleSelectModel}
          />

          <SettingsDialog
            open={settingsDialogOpen}
            onOpenChange={setSettingsDialogOpen}
            settings={settings}
            serverHealth={serverHealth}
            onSave={setSettings}
          />

          <CommandPalette
            open={commandOpen}
            onOpenChange={setCommandOpen}
            conversations={conversations}
            activeSkin={settings.skin}
            onNewChat={handleNewChat}
            onSelectConversation={(id) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(id);
              setActiveArtifact(null);
              setArtifactFullscreen(false);
            }}
            onOpenModelSelector={() => setModelDialogOpen(true)}
            onOpenSettings={() => setSettingsDialogOpen(true)}
            onOpenBookmarks={() => setBookmarksOpen(true)}
            onToggleZenMode={() => setZenMode((z) => !z)}
            onCycleSkin={handleCycleSkin}
            onExportMarkdown={() => handleExportConversation('md')}
            onInsertTemplate={(tpl) => setExternalDraft(tpl)}
            showInstallOption={!isStandalone}
            onInstallApp={() => void handleInstallAction()}
          />

          <BookmarksDialog
            open={bookmarksOpen}
            onOpenChange={setBookmarksOpen}
            conversations={conversations}
            onJumpToConversation={(convId) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(convId);
            }}
            onToggleStar={handleToggleStar}
          />

          <InstallAppDialog
            open={installDialogOpen}
            onOpenChange={setInstallDialogOpen}
            isIos={isIos}
            canNativeInstall={canNativeInstall}
            onNativeInstall={() => {
              void handleInstallAction().then(() => setInstallDialogOpen(false));
            }}
          />

          <Dialog open={Boolean(lightboxUrl)} onOpenChange={() => setLightboxUrl(null)}>
            <DialogContent className="max-w-4xl border-0 bg-transparent p-2 shadow-none">
              <DialogHeader className="sr-only">
                <DialogTitle>Image Preview</DialogTitle>
              </DialogHeader>
              {lightboxUrl && (
                <img
                  src={lightboxUrl}
                  alt="Attachment Preview"
                  className="max-h-[85dvh] w-full rounded-2xl object-contain shadow-2xl"
                />
              )}
            </DialogContent>
          </Dialog>
        </div>
      </TooltipProvider>
    </MotionConfig>
  );
}

import type { Attachment, TokenUsage } from '@/types/chat';

export interface StreamChatPayload {
  model: string;
  messages: Array<{
    role: 'user' | 'assistant';
    content: string;
    attachments?: Attachment[];
  }>;
  systemInstruction: string;
  temperature: number;
  maxOutputTokens: number;
  includeThoughts: boolean;
  thinkingBudget: number;
  autoFallbackOnOverload: boolean;
  fallbackModel: string;
  demoMode: boolean;
  apiKey?: string;
}

export interface StreamHandlers {
  signal: AbortSignal;
  onThought: (text: string) => void;
  onDelta: (text: string) => void;
  onUsage: (usage: TokenUsage) => void;
  onStatus?: (message: string) => void;
  onInfo?: (message: string) => void;
  onDone: (meta: { model: string; finishReason?: string; isDemo?: boolean }) => void;
  onError: (message: string) => void;
}

export async function streamChatCompletion(
  payload: StreamChatPayload,
  handlers: StreamHandlers
): Promise<void> {
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
  };

  if (payload.apiKey?.trim()) {
    headers['x-gemini-api-key'] = payload.apiKey.trim();
  }

  const response = await fetch('/api/chat', {
    method: 'POST',
    headers,
    body: JSON.stringify({
      model: payload.model,
      messages: payload.messages.map((m) => ({
        role: m.role,
        content: m.content,
        attachments: m.attachments?.map((a) => ({
          kind: a.kind,
          name: a.name,
          mimeType: a.mimeType,
          data: a.data,
          textContent: a.textContent,
        })),
      })),
      systemInstruction: payload.systemInstruction,
      temperature: payload.temperature,
      maxOutputTokens: payload.maxOutputTokens,
      includeThoughts: payload.includeThoughts,
      thinkingBudget: payload.thinkingBudget,
      autoFallbackOnOverload: payload.autoFallbackOnOverload,
      fallbackModel: payload.fallbackModel,
      demoMode: payload.demoMode,
    }),
    signal: handlers.signal,
  });

  if (!response.ok) {
    const errText = await response.text();
    handlers.onError(errText || `HTTP Error ${response.status}`);
    return;
  }

  if (!response.body) {
    handlers.onError('Browser tidak menerima ReadableStream dari server.');
    return;
  }

  const reader = response.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';

  while (true) {
    const { value, done } = await reader.read();
    if (done) break;

    buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');

    let boundaryIndex: number;
    while ((boundaryIndex = buffer.indexOf('\n\n')) !== -1) {
      const rawBlock = buffer.slice(0, boundaryIndex);
      buffer = buffer.slice(boundaryIndex + 2);

      if (!rawBlock.trim() || rawBlock.startsWith(':')) continue;

      let eventName = 'message';
      const dataLines: string[] = [];

      for (const line of rawBlock.split('\n')) {
        if (line.startsWith('event:')) {
          eventName = line.slice(6).trim();
        } else if (line.startsWith('data:')) {
          dataLines.push(line.slice(5).trimStart());
        }
      }

      if (dataLines.length === 0) continue;

      let parsed: Record<string, unknown>;
      try {
        parsed = JSON.parse(dataLines.join('\n'));
      } catch {
        continue;
      }

      if (eventName === 'thought' && typeof parsed.text === 'string') {
        handlers.onThought(parsed.text);
      } else if (eventName === 'delta' && typeof parsed.text === 'string') {
        handlers.onDelta(parsed.text);
      } else if (eventName === 'usage') {
        handlers.onUsage(parsed as TokenUsage);
      } else if (eventName === 'status' && typeof parsed.message === 'string') {
        handlers.onStatus?.(parsed.message);
      } else if (eventName === 'info' && typeof parsed.message === 'string') {
        handlers.onInfo?.(parsed.message);
      } else if (eventName === 'done') {
        handlers.onDone({
          model: String(parsed.model ?? payload.model),
          finishReason: typeof parsed.finishReason === 'string' ? parsed.finishReason : undefined,
          isDemo: Boolean(parsed.isDemo),
        });
      } else if (eventName === 'error') {
        handlers.onError(
          typeof parsed.message === 'string'
            ? parsed.message
            : 'Terjadi kesalahan pada stream SSE.'
        );
      }
    }
  }
}

function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) {
      reject(new DOMException('Aborted', 'AbortError'));
      return;
    }
    const timer = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();
    }, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

function buildGeminiContents(messages) {
  return messages.map((msg) => {
    const parts = [];
    if (msg.content && msg.content.trim().length > 0) {
      parts.push({ text: msg.content });
    }
    if (Array.isArray(msg.attachments)) {
      for (const att of msg.attachments) {
        if (att?.kind === 'text' && typeof att?.textContent === 'string') {
          parts.push({
            text: `\n\n[Attached File: ${att.name}]\n\`\`\`\n${att.textContent}\n\`\`\``,
          });
        } else if (att?.mimeType && att?.data) {
          parts.push({
            inlineData: {
              mimeType: att.mimeType,
              data: att.data,
            },
          });
        }
      }
    }
    if (parts.length === 0) {
      parts.push({ text: ' ' });
    }
    return {
      role: msg.role === 'assistant' ? 'model' : 'user',
      parts,
    };
  });
}

function buildGeminiPayload({
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  withThinkingConfig,
}) {
  const generationConfig = {
    temperature: Number.isFinite(Number(temperature)) ? Number(temperature) : 0.7,
    maxOutputTokens: Number.isFinite(Number(maxOutputTokens))
      ? Number(maxOutputTokens)
      : 8192,
  };

  if (withThinkingConfig && includeThoughts && Number(thinkingBudget) !== 0) {
    const thinkingConfig = { includeThoughts: true };
    if (Number(thinkingBudget) > 0) {
      thinkingConfig.thinkingBudget = Number(thinkingBudget);
    }
    generationConfig.thinkingConfig = thinkingConfig;
  }

  const payload = {
    contents: buildGeminiContents(messages),
    generationConfig,
  };

  if (typeof systemInstruction === 'string' && systemInstruction.trim().length > 0) {
    payload.systemInstruction = {
      parts: [{ text: systemInstruction.trim() }],
    };
  }

  return payload;
}

function extractGeminiError(rawText, status) {
  let msg = `Gemini API error (${status})`;
  try {
    const parsed = JSON.parse(rawText);
    if (parsed?.error?.message) msg = parsed.error.message;
  } catch {
    if (rawText) msg = rawText;
  }
  return msg;
}

async function fetchGeminiWithRetry({
  targetModel,
  apiKey,
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  signal,
  onStatus,
  onInfo,
}) {
  const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(
    targetModel
  )}:streamGenerateContent?alt=sse`;

  let wantsThinking = Boolean(includeThoughts) && Number(thinkingBudget) !== 0;

  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
    const upstream = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-goog-api-key': apiKey,
      },
      body: JSON.stringify(
        buildGeminiPayload({
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          withThinkingConfig: wantsThinking,
        })
      ),
      signal,
    });

    if (upstream.ok) {
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    const errBody = await upstream.text();

    // Fallback jika model menolak parameter thinkingConfig
    if (upstream.status === 400 && wantsThinking && /thinking/i.test(errBody)) {
      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    if (RETRYABLE_STATUSES.has(upstream.status) && attempt < MAX_RETRIES) {
      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${MAX_RETRIES}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: upstream.status,
      message: extractGeminiError(errBody, upstream.status),
    };
  }

  return { ok: false, status: 503, message: 'Server Gemini sedang sibuk.' };
}

    }
    const timer = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();
    }, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

function buildGeminiContents(messages) {
  return messages.map((msg) => {
    const parts = [];
    if (msg.content && msg.content.trim().length > 0) {
      parts.push({ text: msg.content });
    }
    if (Array.isArray(msg.attachments)) {
      for (const att of msg.attachments) {
        if (att?.kind === 'text' && typeof att?.textContent === 'string') {
          parts.push({
            text: `\n\n[Attached File: ${att.name}]\n\`\`\`\n${att.textContent}\n\`\`\``,
          });
        } else if (att?.mimeType && att?.data) {
          parts.push({
            inlineData: {
              mimeType: att.mimeType,
              data: att.data,
            },
          });
        }
      }
    }
    if (parts.length === 0) {
      parts.push({ text: ' ' });
    }
    return {
      role: msg.role === 'assistant' ? 'model' : 'user',
      parts,
    };
  });
}

function buildGeminiPayload({
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  withThinkingConfig,
}) {
  const generationConfig = {
    temperature: Number.isFinite(Number(temperature)) ? Number(temperature) : 0.7,
    maxOutputTokens: Number.isFinite(Number(maxOutputTokens))
      ? Number(maxOutputTokens)
      : 8192,
  };

  if (withThinkingConfig && includeThoughts && Number(thinkingBudget) !== 0) {
    const thinkingConfig = { includeThoughts: true };
    if (Number(thinkingBudget) > 0) {
      thinkingConfig.thinkingBudget = Number(thinkingBudget);
    }
    generationConfig.thinkingConfig = thinkingConfig;
  }

  const payload = {
    contents: buildGeminiContents(messages),
    generationConfig,
  };

  if (typeof systemInstruction === 'string' && systemInstruction.trim().length > 0) {
    payload.systemInstruction = {
      parts: [{ text: systemInstruction.trim() }],
    };
  }

  return payload;
}

function extractGeminiError(rawText, status) {
  let msg = `Gemini API error (${status})`;
  try {
    const parsed = JSON.parse(rawText);
    if (parsed?.error?.message) msg = parsed.error.message;
  } catch {
    if (rawText) msg = rawText;
  }
  return msg;
}

async function fetchGeminiWithRetry({
  targetModel,
  apiKey,
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  signal,
  onStatus,
  onInfo,
}) {
  const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${encodeURIComponent(
    targetModel
  )}:streamGenerateContent?alt=sse`;

  let wantsThinking = Boolean(includeThoughts) && Number(thinkingBudget) !== 0;

  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
    const upstream = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-goog-api-key': apiKey,
      },
      body: JSON.stringify(
        buildGeminiPayload({
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          withThinkingConfig: wantsThinking,
        })
      ),
      signal,
    });

    if (upstream.ok) {
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    const errBody = await upstream.text();

    if (upstream.status === 400 && wantsThinking && /thinking/i.test(errBody)) {
      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    if (RETRYABLE_STATUSES.has(upstream.status) && attempt < MAX_RETRIES) {
      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${MAX_RETRIES}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: upstream.status,
      message: extractGeminiError(errBody, upstream.status),
    };
  }

  return { ok: false, status: 503, message: 'Server Gemini sedang sibuk.' };
}

export default async function handler(request) {
  if (request.method !== 'POST') {
    return new Response('Method Not Allowed', { status: 405 });
  }

  let body;
  try {
    body = await request.json();
  } catch {
    return new Response('Invalid JSON body', { status: 400 });
  }

    model = DEFAULT_MODEL,
    messages = [],
    systemInstruction = '',
    temperature = 0.7,
    maxOutputTokens = 8192,
    includeThoughts = true,
    thinkingBudget = -1,

    demoMode = false,

          const userText = lastUserMsg?.content?.trim() || 'Demo Vercel SSE';

              sendEvent('thought', { text: chunk });

            `> **Prompt:** ${userText.replace(/\n/g, ' ')}\n`,
          ].join('');

          const tokens = demoReply.match(/\S+\s*/g) || [demoReply];
          for (const token of tokens) {
            await sleep(16, request.signal);
            sendEvent('delta', { text: token });
          }

          const promptTokens = Math.max(24, Math.ceil(userText.length / 3));
          sendEvent('usage', {
            promptTokenCount: promptTokens,
            candidatesTokenCount: tokens.length,
            thoughtsTokenCount: showThoughts ? 40 : 0,
            totalTokenCount: promptTokens + tokens.length + (showThoughts ? 40 : 0),
          });
          sendEvent('done', { model, finishReason: 'STOP', isDemo: true });
          return;
        }

        if (!apiKey) {
          sendEvent('error', {
            status: 401,
            message:
              'GEMINI_API_KEY belum diset di Environment Variables Vercel. Tambahkan di Project Settings Vercel, masukkan lewat Settings UI, atau aktifkan Demo SSE Mode.',
          });
          return;
        }

        let result = await fetchGeminiWithRetry({
          targetModel: model,
          apiKey,
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          signal: request.signal,
          onStatus: (message) => sendEvent('status', { message }),
          onInfo: (message) => sendEvent('info', { message }),
        });

        if (
          !result.ok &&
          RETRYABLE_STATUSES.has(result.status) &&
          autoFallbackOnOverload &&
          fallbackModel &&
          fallbackModel !== model
        ) {
          sendEvent('status', {
            message: `${model} masih penuh (${result.status}). Beralih otomatis ke ${fallbackModel}...`,
          });
          sendEvent('info', {
            message: `Fallback otomatis ke ${fallbackModel} karena ${model} mengalami high demand.`,
          });
          result = await fetchGeminiWithRetry({
            targetModel: fallbackModel,
            apiKey,
            messages,
            systemInstruction,
            temperature,
            maxOutputTokens,
            includeThoughts,
            thinkingBudget,
            signal: request.signal,
            onStatus: (message) => sendEvent('status', { message }),
            onInfo: (message) => sendEvent('info', { message }),
          });
        }

        if (!result.ok) {
          sendEvent('error', { status: result.status, message: result.message });
          return;
        }

        const upstream = result.response;
        const activeModel = result.usedModel;

        if (!upstream.body) {
          sendEvent('error', { status: 502, message: 'Stream respons Gemini kosong.' });
          return;
        }

        const reader = upstream.body.getReader();
        const decoder = new TextDecoder();
        let buffer = '';
        let lastFinishReason = undefined;

        while (true) {
          const { value, done } = await reader.read();
          if (done) break;

          buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');

          let boundaryIndex;
          while ((boundaryIndex = buffer.indexOf('\n\n')) !== -1) {
            const rawEvent = buffer.slice(0, boundaryIndex);
            buffer = buffer.slice(boundaryIndex + 2);

            const dataLines = rawEvent
              .split('\n')
              .filter((line) => line.startsWith('data:'))
              .map((line) => line.slice(5).trimStart());

            if (dataLines.length === 0) continue;
            const jsonString = dataLines.join('\n').trim();
            if (!jsonString || jsonString === '[DONE]') continue;

            try {
              const chunk = JSON.parse(jsonString);
              const candidate = chunk?.candidates?.[0];
              const parts = candidate?.content?.parts;

              if (Array.isArray(parts)) {
                for (const part of parts) {
                  if (part?.thought && typeof part?.text === 'string' && part.text.length > 0) {
                    sendEvent('thought', { text: part.text });
                  } else if (typeof part?.text === 'string' && part.text.length > 0) {
                    sendEvent('delta', { text: part.text });
                  }
                }
              }

              if (candidate?.finishReason) lastFinishReason = candidate.finishReason;
              if (chunk?.usageMetadata) sendEvent('usage', chunk.usageMetadata);
            } catch {
              // Abaikan JSON parsial
            }
          }
        }

        sendEvent('done', {
          model: activeModel,
          finishReason: lastFinishReason || 'STOP',
          isDemo: false,
        });
      } catch (err) {
        if (err instanceof Error && err.name === 'AbortError') return;
        sendEvent('error', {
          status: 500,
          message: err instanceof Error ? err.message : 'Internal Edge Function Error',
        });
      } finally {
        try {
          controller.close();
        } catch {
          // Abaikan
        }
      }
    },
  });

  return new Response(stream, {
    status: 200,
    headers: {
      'Content-Type': 'text/event-stream; charset=utf-8',
      'Cache-Control': 'no-cache, no-transform',
      'X-Accel-Buffering': 'no',
    },
  });
}

                const detected = extractArtifactsFromMarkdown(assistantMessageId, accContent);
                const previewable = detected.find(
                  (a) => a.kind === 'html' || a.kind === 'svg' || a.kind === 'mermaid'
                );
                if (previewable) setActiveArtifact(previewable);
              }
            },
            onError: (errorMessage) => {
              if (flushTimerRef.current !== null) {
                window.clearTimeout(flushTimerRef.current);
                flushTimerRef.current = null;
              }
              activeStreamRef.current = null;
              setActiveStream(null);
              toast.error(errorMessage);
              setConversations((prev) =>
                prev.map((conv) =>
                  conv.id !== convId
                    ? conv
                    : {
                        ...conv,
                        messages: conv.messages.map((m) =>
                          m.id === assistantMessageId
                            ? {
                                ...m,
                                content: accContent,
                                thought: accThought || undefined,
                                statusText: undefined,
                                isStreaming: false,
                                error: errorMessage,
                              }
                            : m
                        ),
                      }
                )
              );
            },
          }
        );
      } catch (err) {
        if (err instanceof Error && err.name === 'AbortError') return;
        const message = err instanceof Error ? err.message : 'Gagal terhubung ke server SSE.';
        toast.error(message);
        activeStreamRef.current = null;
        setActiveStream(null);
        setConversations((prev) =>
          prev.map((conv) =>
            conv.id !== convId
              ? conv
              : {
                  ...conv,
                  messages: conv.messages.map((m) =>
                    m.id === assistantMessageId
                      ? {
                          ...m,
                          content: accContent,
                          statusText: undefined,
                          isStreaming: false,
                          error: message,
                        }
                      : m
                  ),
                }
          )
        );
      } finally {
        if (abortControllerRef.current === controller) {
          abortControllerRef.current = null;
          if (activeStreamRef.current?.msgId === assistantMessageId) {
            activeStreamRef.current = null;
            setActiveStream(null);
            setConversations((prev) =>
              prev.map((conv) =>
                conv.id !== convId
                  ? conv
                  : {
                      ...conv,
                      messages: conv.messages.map((m) =>
                        m.id === assistantMessageId && m.isStreaming
                          ? {
                              ...m,
                              content: accContent,
                              thought: accThought || undefined,
                              statusText: undefined,
                              isStreaming: false,
                            }
                          : m
                      ),
                    }
              )
            );
          }
        }
      }
    },
    [settings]
  );

  const handleSend = useCallback(
    (text: string, attachments: Attachment[]) => {
      if (isStreamingActive) return;

      const userMessage: ChatMessageType = {
        id: generateId(),
        role: 'user',
        content: text,
        attachments: attachments.length > 0 ? attachments : undefined,
        createdAt: Date.now(),
      };

      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: currentModelId,
        isStreaming: true,
      };

      if (!activeConversation) {
        const newId = generateId();
        const title = (text || attachments[0]?.name || 'New Chat').slice(0, 50);
        const newConv: Conversation = {
          id: newId,
          title,
          model: currentModelId,
          messages: [userMessage, assistantPlaceholder],
          createdAt: Date.now(),
          updatedAt: Date.now(),
        };

        setConversations((prev) => [newConv, ...prev]);
        setActiveId(newId);
        void runStreamForConversation(newId, currentModelId, [userMessage], assistantMessageId);
      } else {
        const historyForModel = [...activeConversation.messages, userMessage];
        setConversations((prev) =>
          prev.map((conv) =>
            conv.id !== activeConversation.id
              ? conv
              : {
                  ...conv,
                  updatedAt: Date.now(),
                  messages: [...conv.messages, userMessage, assistantPlaceholder],
                }
          )
        );
        void runStreamForConversation(
          activeConversation.id,
          currentModelId,
          historyForModel,
          assistantMessageId
        );
      }
    },
    [activeConversation, currentModelId, isStreamingActive, runStreamForConversation]
  );

  const handleRegenerate = useCallback(
    (overrideModel?: string) => {
      if (!activeConversation || isStreamingActive) return;
      const msgs = activeConversation.messages;
      if (msgs.length === 0) return;

      const lastMsg = msgs[msgs.length - 1];
      const baseMessages = lastMsg.role === 'assistant' ? msgs.slice(0, -1) : msgs;
      if (baseMessages.length === 0) return;

      const targetModel = overrideModel || activeConversation.model;
      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: targetModel,
        isStreaming: true,
      };

      setConversations((prev) =>
        prev.map((conv) =>
          conv.id !== activeConversation.id
            ? conv
            : {
                ...conv,
                model: targetModel,
                updatedAt: Date.now(),
                messages: [...baseMessages, assistantPlaceholder],
              }
        )
      );

      void runStreamForConversation(
        activeConversation.id,
        targetModel,
        baseMessages,
        assistantMessageId
      );
    },
    [activeConversation, isStreamingActive, runStreamForConversation]
  );

  const handleEditUserMessage = useCallback(
    (messageId: string, newContent: string) => {
      if (!activeConversation || isStreamingActive) return;
      const idx = activeConversation.messages.findIndex((m) => m.id === messageId);
      if (idx === -1) return;

      const updatedMsg: ChatMessageType = {
        ...activeConversation.messages[idx],
        content: newContent,
      };
      const baseHistory = [...activeConversation.messages.slice(0, idx), updatedMsg];

      const assistantMessageId = generateId();
      const assistantPlaceholder: ChatMessageType = {
        id: assistantMessageId,
        role: 'assistant',
        content: '',
        createdAt: Date.now(),
        model: activeConversation.model,
        isStreaming: true,
      };

      setConversations((prev) =>
        prev.map((conv) =>
          conv.id !== activeConversation.id
            ? conv
            : {
                ...conv,
                updatedAt: Date.now(),
                messages: [...baseHistory, assistantPlaceholder],
              }
        )
      );

      void runStreamForConversation(
        activeConversation.id,
        activeConversation.model,
        baseHistory,
        assistantMessageId
      );
    },
    [activeConversation, isStreamingActive, runStreamForConversation]
  );

  const handleForkFromMessage = useCallback(
    (messageId: string) => {
      if (!activeConversation) return;
      const idx = activeConversation.messages.findIndex((m) => m.id === messageId);
      if (idx === -1) return;

      const forkedId = generateId();
      const forkedConv: Conversation = {
        id: forkedId,
        title: `Branch: ${activeConversation.title}`.slice(0, 52),
        model: activeConversation.model,
        messages: activeConversation.messages.slice(0, idx + 1).map((m) => ({
          ...m,
          id: generateId(),
        })),
        createdAt: Date.now(),
        updatedAt: Date.now(),
      };
      setConversations((prev) => [forkedConv, ...prev]);
      setActiveId(forkedId);
      toast.success('Cabang chat baru dibuat');
    },
    [activeConversation]
  );

  const handleToggleStar = useCallback((convId: string, msgId: string) => {
    setConversations((prev) =>
      prev.map((conv) =>
        conv.id !== convId
          ? conv
          : {
              ...conv,
              messages: conv.messages.map((m) =>
                m.id === msgId ? { ...m, starred: !m.starred } : m
              ),
            }
      )
    );
  }, []);

  const handleSelectModel = (modelId: string) => {
    setSelectedModel(modelId);
    if (activeConversation) {
      setConversations((prev) =>
        prev.map((c) => (c.id === activeConversation.id ? { ...c, model: modelId } : c))
      );
    }
    toast.success(`Model aktif: ${modelId}`);
  };

  const handleExportConversation = useCallback(
    (format: 'md' | 'json') => {
      if (!activeConversation) {
        toast.error('Tidak ada percakapan aktif untuk di-export');
        return;
      }
      let content = '';
      let mime = 'text/markdown;charset=utf-8';
      if (format === 'json') {
        content = JSON.stringify(activeConversation, null, 2);
        mime = 'application/json;charset=utf-8';
      } else {
        content = [
          `# ${activeConversation.title}`,
          `Model: \`${activeConversation.model}\`\n`,
          ...activeConversation.messages.map(
            (m) => `### ${m.role === 'user' ? 'User' : 'Assistant'}\n\n${m.content}\n`
          ),
        ].join('\n');
      }
      const blob = new Blob([content], { type: mime });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `chat-${activeConversation.id.slice(0, 8)}.${format}`;
      a.click();
      URL.revokeObjectURL(url);
      toast.success(`Exported .${format}`);
    },
    [activeConversation]
  );

  const handleDeleteConversation = useCallback(
    (id: string) => {
      if (activeStreamRef.current?.convId === id) handleStopStream();
      setConversations((prev) => {
        const next = prev.filter((c) => c.id !== id);
        if (activeId === id) setActiveId(next[0]?.id ?? null);
        return next;
      });
      toast.success('Percakapan dihapus');
    },
    [activeId, handleStopStream]
  );

  const handleRenameConversation = useCallback((id: string, title: string) => {
    setConversations((prev) => prev.map((c) => (c.id === id ? { ...c, title } : c)));
  }, []);

  const handleTogglePinConversation = useCallback((id: string) => {
    setConversations((prev) =>
      prev.map((c) => (c.id === id ? { ...c, pinned: !c.pinned } : c))
    );
  }, []);

    const msgs = activeConversation?.messages ?? [];
    for (let i = msgs.length - 1; i >= 0; i--) {

          {!zenMode && (

              mobileOpen={mobileSidebarOpen}

              conversations={conversations}
              activeId={activeId}
              onNewChat={handleNewChat}
              onSelectConversation={(id) => {
                if (isStreamingActive) handleStopStream();
                setActiveId(id);
                setActiveArtifact(null);
                setArtifactFullscreen(false);
              }}
              onRenameConversation={handleRenameConversation}
              onTogglePinConversation={handleTogglePinConversation}
              onDeleteConversation={handleDeleteConversation}
              onOpenSettings={() => setSettingsDialogOpen(true)}

              onOpenBookmarks={() => setBookmarksOpen(true)}
              onCycleSkin={handleCycleSkin}
              showInstallButton={!isStandalone}
              onInstallApp={() => void handleInstallAction()}
            />

          <div
            className={cn(
              'flex min-w-0 max-w-full flex-1 flex-col overflow-hidden',
              activeArtifact && artifactFullscreen && 'hidden lg:hidden'
            )}
          >
            <header className="flex h-14 shrink-0 items-center justify-between gap-2 border-b bg-background/95 px-3 sm:px-4">
              <div className="flex min-w-0 items-center gap-1.5 sm:gap-2">
                {!zenMode && (
                  <Button
                    type="button"
                    variant="ghost"
                    size="icon"
                    onClick={() => setMobileSidebarOpen(true)}
                    className="shrink-0 md:hidden"
                    aria-label="Menu"
                  >
                    <Menu className="h-5 w-5" />
                  </Button>

                <Button
                  type="button"
                  variant="ghost"
                  onClick={() => setModelDialogOpen(true)}
                  className="h-9 min-w-0 gap-1.5 rounded-xl border border-border/50 bg-card/60 px-2.5 font-medium shadow-sm sm:gap-2 sm:px-3"
                >
                  <Sparkles className="h-4 w-4 shrink-0 text-primary" />
                  <span className="truncate text-xs sm:text-sm">{currentModelInfo.name}</span>
                  {currentModelInfo.isNew && (
                    <Badge
                      variant="secondary"
                      className="hidden bg-emerald-500/15 text-[10px] text-emerald-500 sm:inline-flex"
                    >
                      New
                    </Badge>
                  )}
                  <ChevronDown className="h-3.5 w-3.5 shrink-0 text-muted-foreground" />
                </Button>

                {isStreamingActive && (
                  <Badge variant="outline" className="hidden font-mono text-[10px] sm:inline-flex">
                    {activeStream.statusText
                      ? 'Retrying...'
                      : activeStream.tps > 0
                        ? `${activeStream.tps} tok/s`
                        : 'Thinking...'}
                  </Badge>
                )}
              </div>

              <div className="flex shrink-0 items-center gap-1">
                {!isOnline && (
                  <Badge variant="destructive" className="gap-1 px-2 py-0.5 text-[10px]">
                    <WifiOff className="h-3 w-3" />
                    <span className="hidden sm:inline">Offline</span>
                  </Badge>
                )}

                <Tooltip>
                  <TooltipTrigger asChild>
                    <div className="hidden items-center gap-2 rounded-full border bg-card/60 px-2.5 py-1 text-[11px] text-muted-foreground lg:flex">
                      <span className="font-mono">
                        {(contextTokenEstimate / 1000).toFixed(1)}k ctx
                      </span>

                          className="h-full rounded-full bg-primary transition-all"
                          style={{ width: `${contextUsagePct}%` }}

                    Estimasi context (~{contextTokenEstimate.toLocaleString()} /{' '}
                    {currentModelInfo.contextLimit.toLocaleString()} tokens)

                  </TooltipContent>
                </Tooltip>

                <Tooltip>
                  <TooltipTrigger asChild>
                    <Button
                      type="button"
                      variant={zenMode ? 'secondary' : 'ghost'}
                      size="icon"
                      onClick={() => setZenMode((z) => !z)}
                      className="hidden sm:inline-flex"

                      Skin: {settings.skin}

                      Command Palette (Ctrl+K)

                      <>

                        <DropdownMenuItem onClick={() => handleExportConversation('md')}>
                          <FileText className="h-3.5 w-3.5" />
                          Export Markdown (.md)
                        </DropdownMenuItem>
                        <DropdownMenuItem onClick={() => handleExportConversation('json')}>
                          <FileJson className="h-3.5 w-3.5" />
                          Export JSON (.json)
                        </DropdownMenuItem>

                    <DropdownMenuSeparator />
                    <DropdownMenuItem onClick={() => setSettingsDialogOpen(true)}>
                      <Settings className="h-3.5 w-3.5" />
                      Settings & API Key
                    </DropdownMenuItem>

              <StudioScroll
                ref={chatScrollRef}
                onScroll={handleViewportScroll}
                className="h-full w-full"
                autoHide="scroll"
              >
                {!activeConversation || activeConversation.messages.length === 0 ? (
                  <div className="relative mx-auto flex min-h-[calc(100dvh-13rem)] w-full max-w-3xl flex-col items-center justify-center px-4 py-8 text-center">

                    <div className="pointer-events-none absolute top-10 -z-10 h-56 w-56 rounded-full bg-primary/10 blur-3xl" />

                      initial={{ scale: 0.92, opacity: 0 }}
                      animate={{ scale: 1, opacity: 1 }}
                      transition={{ duration: 0.24 }}
                      className="mb-4"
                    >
                      <BrandLogo size="lg" className="shadow-xl" />
                    </motion.div>

                    <div className="mb-2 inline-flex flex-wrap items-center justify-center gap-2 rounded-full border bg-card/90 px-3.5 py-1 text-xs text-muted-foreground shadow-sm">
                      <Cpu className="h-3.5 w-3.5 text-primary" />
                      <span className="font-mono font-semibold text-foreground">
                        {currentModelId}
                      </span>
                      <span className="hidden sm:inline">•</span>
                      <span className="hidden sm:inline">
                        ${currentModelInfo.inputPricePerMillion.toFixed(2)} / $
                        {currentModelInfo.outputPricePerMillion.toFixed(2)} per 1M
                      </span>
                      {isStandalone && (
                        <>
                          <span>•</span>
                          <span className="text-emerald-400">App Mode</span>
                        </>
                      )}
                    </div>

                    <h1 className="mt-2 text-2xl font-bold tracking-tight text-foreground sm:text-4xl">
                      Adam Studio Workspace
                    </h1>
                    <p className="mt-2.5 max-w-lg text-sm leading-relaxed text-muted-foreground">
                      Platform AI real-time SSE dengan Live Thinking ala Google AI Studio,
                      Auto-Retry 503/429, Pro Code Blocks, dan Artifact IDE Canvas.
                    </p>

                    {!settings.apiKey &&
                      !serverHealth?.hasServerApiKey &&
                      !settings.demoMode && (
                        <div className="mt-5 flex flex-wrap items-center justify-center gap-2 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-2.5 text-xs text-amber-400">
                          <span>API Key belum aktif.</span>
                          <button
                            type="button"
                            onClick={() => setSettingsDialogOpen(true)}
                            className="font-semibold underline underline-offset-4 hover:text-amber-300"
                          >
                            Isi API Key atau aktifkan Demo SSE Showcase
                          </button>
                        </div>
                      )}

                    <div className="mt-6 inline-flex flex-wrap items-center justify-center gap-1 rounded-full border bg-card/80 p-1">
                      {WELCOME_TABS.map((tab) => (
                        <button
                          key={tab}
                          type="button"
                          onClick={() => setWelcomeTab(tab)}
                          className={cn(
                            'relative rounded-full px-3 py-1 text-xs font-medium transition-colors sm:px-3.5',
                            welcomeTab === tab
                              ? 'text-primary-foreground'
                              : 'text-muted-foreground hover:text-foreground'
                          )}
                        >
                          {welcomeTab === tab && (
                            <motion.span
                              layoutId="welcome-tab-pill"
                              transition={{ type: 'spring', stiffness: 380, damping: 30 }}
                              className="absolute inset-0 -z-10 rounded-full bg-primary shadow-sm"
                            />
                          )}
                          {tab}
                        </button>
                      ))}
                    </div>

                    <div className="mt-5 grid w-full grid-cols-1 gap-3 text-left sm:grid-cols-2">
                      {filteredStarterCards.map((item) => {
                        const Icon = item.icon;
                        return (
                          <button
                            key={item.title}
                            type="button"
                            onClick={() => handleSend(item.prompt, [])}
                            className="group rounded-2xl border bg-card/80 p-4 transition-all hover:-translate-y-0.5 hover:border-primary/50 hover:bg-accent/40 hover:shadow-lg"
                          >
                            <div className="flex items-center justify-between">
                              <div className="flex items-center gap-2 text-xs font-semibold text-foreground">
                                <div className="flex h-7 w-7 items-center justify-center rounded-lg bg-primary/15 text-primary">
                                  <Icon className="h-3.5 w-3.5" />
                                </div>
                                <span>{item.title}</span>
                              </div>
                              <Badge variant="outline" className="text-[10px] font-normal">
                                {item.category}
                              </Badge>
                            </div>
                            <p className="mt-2.5 text-xs leading-relaxed text-muted-foreground">
                              {item.subtitle}
                            </p>
                          </button>
                        );
                      })}
                    </div>
                  </div>
                ) : (
                  <div
                    className={cn(
                      'mx-auto w-full min-w-0 max-w-3xl px-3 py-5 sm:px-5 sm:py-6',
                      settings.density === 'compact' ? 'space-y-4' : 'space-y-6'
                    )}
                  >
                    {activeConversation.messages.map((msg) => {
                      const effectiveMsg =
                        activeStream &&
                        activeStream.convId === activeConversation.id &&
                        activeStream.msgId === msg.id
                          ? {
                              ...msg,
                              content: activeStream.content,
                              thought: activeStream.thought || msg.thought,
                              statusText: activeStream.statusText,
                              metrics: activeStream.thinkingDurationMs
                                ? {
                                    ...msg.metrics,
                                    thinkingDurationMs: activeStream.thinkingDurationMs,

                              isStreaming: true,
                            }
                          : msg;

                      return (
                        <ChatMessage
                          key={msg.id}
                          message={effectiveMsg}
                          compact={settings.density === 'compact'}
                          thinkingEnabled={thinkingEnabled}
                          isLastAssistant={msg.id === lastAssistantId}
                          onRegenerate={
                            msg.id === lastAssistantId ? () => handleRegenerate() : undefined
                          }
                          onRetryWithModel={
                            msg.id === lastAssistantId
                              ? (mId) => handleRegenerate(mId)
                              : undefined
                          }
                          onEditUserMessage={
                            msg.role === 'user' && !isStreamingActive
                              ? handleEditUserMessage
                              : undefined
                          }
                          onForkFromMessage={
                            msg.role === 'assistant' && !isStreamingActive
                              ? handleForkFromMessage
                              : undefined
                          }
                          onToggleStar={(msgId) =>
                            handleToggleStar(activeConversation.id, msgId)
                          }
                          onOpenArtifact={setActiveArtifact}
                          onPreviewImage={setLightboxUrl}
                        />
                      );
                    })}
                  </div>
                )}
              </StudioScroll>

              {showScrollBottom && (
                <Button
                  type="button"
                  size="icon"
                  variant="secondary"
                  onClick={() => {
                    autoScrollEnabledRef.current = true;
                    scrollToBottom(true);
                  }}
                  className="absolute bottom-3 left-1/2 z-20 -translate-x-1/2 rounded-full border shadow-xl"
                  aria-label="Scroll ke bawah"
                >
                  <ArrowDown className="h-4 w-4" />
                </Button>
              )}
            </div>

            <ChatComposer
              model={currentModelId}
              isStreaming={isStreamingActive}
              settings={settings}
              externalDraft={externalDraft}
              onClearExternalDraft={() => setExternalDraft(null)}
              onUpdateSettings={setSettings}
              onSend={handleSend}
              onStop={handleStopStream}
              onPreviewImage={setLightboxUrl}
            />
          </div>

          <AnimatePresence>
            {activeArtifact && (
              <>
                {!artifactFullscreen && (
                  <div
                    onPointerDown={startResizingPane}
                    className="hidden w-1.5 shrink-0 cursor-col-resize items-center justify-center border-l bg-border/40 transition-colors hover:bg-primary/60 lg:flex"
                    title="Tarik untuk mengubah lebar Canvas"
                  />
                )}
                <motion.aside
                  initial={{ opacity: 0, x: 20 }}
                  animate={{ opacity: 1, x: 0 }}
                  exit={{ opacity: 0, x: 20 }}
                  transition={{ duration: 0.16 }}
                  className={cn(
                    'fixed inset-0 z-40 border-l lg:static',
                    artifactFullscreen ? 'lg:flex-1' : 'lg:shrink-0'
                  )}
                >
                  <div
                    style={
                      {
                        '--artifact-w': `${artifactWidthPct}vw`,
                      } as CSSProperties
                    }
                    className={cn(
                      'h-full w-full',
                      !artifactFullscreen &&
                        'lg:w-[var(--artifact-w)] lg:min-w-[360px] lg:max-w-[68vw]'
                    )}
                  >
                    <ArtifactPanel
                      artifact={activeArtifact}
                      allArtifacts={conversationArtifacts}
                      isResizing={isResizingPane}
                      isFullscreen={artifactFullscreen}
                      onToggleFullscreen={() => setArtifactFullscreen((f) => !f)}
                      onSelectArtifact={setActiveArtifact}
                      onClose={() => {
                        setActiveArtifact(null);
                        setArtifactFullscreen(false);
                      }}
                    />
                  </div>
                </motion.aside>
              </>
            )}
          </AnimatePresence>

          <ModelSelectorDialog
            open={modelDialogOpen}
            onOpenChange={setModelDialogOpen}
            selectedModel={currentModelId}
            onSelectModel={handleSelectModel}
          />

          <SettingsDialog
            open={settingsDialogOpen}
            onOpenChange={setSettingsDialogOpen}
            settings={settings}
            serverHealth={serverHealth}
            onSave={setSettings}
          />

          <CommandPalette
            open={commandOpen}
            onOpenChange={setCommandOpen}
            conversations={conversations}
            activeSkin={settings.skin}
            onNewChat={handleNewChat}
            onSelectConversation={(id) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(id);
              setActiveArtifact(null);
              setArtifactFullscreen(false);
            }}
            onOpenModelSelector={() => setModelDialogOpen(true)}
            onOpenSettings={() => setSettingsDialogOpen(true)}
            onOpenBookmarks={() => setBookmarksOpen(true)}
            onToggleZenMode={() => setZenMode((z) => !z)}
            onCycleSkin={handleCycleSkin}
            onExportMarkdown={() => handleExportConversation('md')}
            onInsertTemplate={(tpl) => setExternalDraft(tpl)}
            showInstallOption={!isStandalone}
            onInstallApp={() => void handleInstallAction()}
          />

          <BookmarksDialog
            open={bookmarksOpen}
            onOpenChange={setBookmarksOpen}
            conversations={conversations}
            onJumpToConversation={(convId) => {
              if (isStreamingActive) handleStopStream();
              setActiveId(convId);
            }}
            onToggleStar={handleToggleStar}
          />

          <InstallAppDialog
            open={installDialogOpen}
            onOpenChange={setInstallDialogOpen}
            isIos={isIos}
            canNativeInstall={canNativeInstall}
            onNativeInstall={() => {
              void handleInstallAction().then(() => setInstallDialogOpen(false));
            }}
          />

          <Dialog open={Boolean(lightboxUrl)} onOpenChange={() => setLightboxUrl(null)}>
            <DialogContent className="max-w-4xl border-0 bg-transparent p-2 shadow-none">
              <DialogHeader className="sr-only">
                <DialogTitle>Image Preview</DialogTitle>
              </DialogHeader>
              {lightboxUrl && (
                <img
                  src={lightboxUrl}
                  alt="Attachment Preview"
                  className="max-h-[85dvh] w-full rounded-2xl object-contain shadow-2xl"
                />
              )}
            </DialogContent>
          </Dialog>
        </div>
      </TooltipProvider>
    </MotionConfig>
  );
}

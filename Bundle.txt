let roundRobinCursor = Math.floor(Math.random() * 1000);
const keyCooldownMap = new Map();

function parseKeyPool(...sources) {
  const pool = [];
  const seen = new Set();
  for (const src of sources) {
    if (!src || typeof src !== 'string') continue;
    for (const part of src.split(/[\r\n,;]+/)) {
      const k = part.trim();
      if (k && !seen.has(k)) {
        seen.add(k);
        pool.push(k);
      }
    }
  }
  return pool;
}

function buildOrderedKeys(keys) {
  const now = Date.now();
  const start = roundRobinCursor % keys.length;
  roundRobinCursor = (roundRobinCursor + 1) % keys.length;

  const rotated = [...keys.slice(start), ...keys.slice(0, start)];
  const ready = [];
  const cooling = [];
  for (let i = 0; i < rotated.length; i++) {
    const k = rotated[i];
    const origIdx = keys.indexOf(k);
    if ((keyCooldownMap.get(k) || 0) > now) cooling.push({ key: k, idx: origIdx });
    else ready.push({ key: k, idx: origIdx });
  }
  return [...ready, ...cooling];
}

function maskKeyLabel(key, idx) {
  const tail = key.length >= 4 ? key.slice(-4) : '****';
  return `Key #${idx + 1} (•••${tail})`;
}

function extractGeminiError(rawText, status) {
  let msg = `Gemini API error (${status})`;
  try {
    const parsed = JSON.parse(rawText);
    if (parsed?.error?.message) msg = parsed.error.message;
  } catch {
    if (rawText) msg = rawText;
  }
  return msg;
}

async function fetchWithTimeout(url, options, parentSignal, timeoutMs = 10000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);
  const onParentAbort = () => controller.abort();
  parentSignal?.addEventListener('abort', onParentAbort, { once: true });

  try {
    return await fetch(url, { ...options, signal: controller.signal });
  } finally {
    clearTimeout(timer);
    parentSignal?.removeEventListener('abort', onParentAbort);
  }
}

function sleep(ms, signal) {
  return new Promise((resolve, reject) => {
    if (signal?.aborted) {
      reject(new DOMException('Aborted', 'AbortError'));
      return;
    }
    const timer = setTimeout(() => {
      signal?.removeEventListener('abort', onAbort);
      resolve();
    }, ms);
    const onAbort = () => {
      clearTimeout(timer);
      reject(new DOMException('Aborted', 'AbortError'));
    };
    signal?.addEventListener('abort', onAbort, { once: true });
  });
}

function buildGeminiContents(messages) {
  return messages.map((msg) => {
    const parts = [];
    if (msg.content && msg.content.trim().length > 0) {
      parts.push({ text: msg.content });
    }
    if (Array.isArray(msg.attachments)) {
      for (const att of msg.attachments) {
        parts.push(...buildAttachmentParts(att));
      }
    }
    if (parts.length === 0) {
      parts.push({ text: ' ' });
    }
    return {
      role: msg.role === 'assistant' ? 'model' : 'user',
      parts,
    };
  });
}

function buildGeminiPayload({
  messages,
  systemInstruction,
  temperature,
  maxOutputTokens,
  includeThoughts,
  thinkingBudget,
  withThinkingConfig,
}) {
  const generationConfig = {
    temperature: Number.isFinite(Number(temperature)) ? Number(temperature) : 0.7,
    maxOutputTokens: Number.isFinite(Number(maxOutputTokens))
      ? Number(maxOutputTokens)
      : 8192,
  };

  if (withThinkingConfig && includeThoughts && Number(thinkingBudget) !== 0) {
    const thinkingConfig = { includeThoughts: true };
    if (Number(thinkingBudget) > 0) {
      thinkingConfig.thinkingBudget = Number(thinkingBudget);
    }
    generationConfig.thinkingConfig = thinkingConfig;
  }

  const payload = {
    contents: buildGeminiContents(messages),
    generationConfig,
  };

  if (typeof systemInstruction === 'string' && systemInstruction.trim().length > 0) {
    payload.systemInstruction = {
      parts: [{ text: systemInstruction.trim() }],
    };
  }

  return payload;
}

function extractGeminiError(rawText, status) {

}

function buildOrderedCandidateKeys(keys) {
  const now = Date.now();
  const start = roundRobinCursor % keys.length;
  roundRobinCursor = (roundRobinCursor + 1) % keys.length;

  const rotated = [...keys.slice(start), ...keys.slice(0, start)];
  const ready = [];
  const cooling = [];
  for (const item of rotated) {
    const origIdx = keys.indexOf(item);
    const coolUntil = keyCooldownMap.get(item) || 0;
    if (coolUntil > now) {
      cooling.push({ key: item, idx: origIdx });
    } else {
      ready.push({ key: item, idx: origIdx });
    }
  }
  return [...ready, ...cooling];
}

  const orderedKeys = buildOrderedCandidateKeys(apiKeys);
  const maxAttempts = orderedKeys.length > 1 ? orderedKeys.length : 4;

  let keyPointer = 0;
  let lastStatus = 503;
  let lastErrorMessage = 'Semua API Key sedang sibuk atau terkena limit.';

  for (let attempt = 0; attempt < maxAttempts; attempt++) {

    const keyLabel = maskKeyLabel(current.key, current.idx);

    const upstream = await fetch(endpoint, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-goog-api-key': current.key,
      },
      body: JSON.stringify(
        buildGeminiPayload({
          messages,
          systemInstruction,
          temperature,
          maxOutputTokens,
          includeThoughts,
          thinkingBudget,
          withThinkingConfig: wantsThinking,
        })
      ),
      signal,
    });

    if (upstream.ok) {
      if (attempt > 0 && orderedKeys.length > 1) {
        onInfo(`Berhasil terhubung menggunakan ${keyLabel} (pool ${orderedKeys.length} keys)`);
      }
      return { ok: true, response: upstream, usedModel: targetModel };
    }

    lastErrorMessage = extractGeminiError(errBody, upstream.status);

      wantsThinking = false;
      onInfo(`Model ${targetModel} tidak mendukung thinkingConfig; melanjutkan tanpa thinking summaries.`);
      attempt -= 1;
      continue;
    }

    const isBadKey =
      upstream.status === 401 ||
      upstream.status === 403 ||
      (upstream.status === 400 && /api.?key|expired|invalid/i.test(lastErrorMessage));

    const canRotateToAnotherKey = orderedKeys.length > 1;
    const isRetryable = RETRYABLE_STATUSES.has(upstream.status) || (isBadKey && canRotateToAnotherKey);

    if (isRetryable && attempt + 1 < maxAttempts) {
      const coolMs = isBadKey ? 120_000 : upstream.status === 429 ? 60_000 : 12_000;
      keyCooldownMap.set(current.key, Date.now() + coolMs);

      if (canRotateToAnotherKey) {
        keyPointer += 1;
        const nextCandidate = orderedKeys[keyPointer % orderedKeys.length];
        const nextLabel = maskKeyLabel(nextCandidate.key, nextCandidate.idx);
        const fastRotateDelay = ROTATE_IMMEDIATELY_STATUSES.has(upstream.status) || isBadKey ? 200 : 450;

        onStatus(
          `${keyLabel} terkena (${upstream.status}). Merotasi ke ${nextLabel} (${attempt + 2}/${maxAttempts})...`
        );
        await sleep(fastRotateDelay, signal);
        continue;
      }

      const retryAfterSec = Number(upstream.headers.get('retry-after'));
      const baseDelay =
        Number.isFinite(retryAfterSec) && retryAfterSec > 0 && retryAfterSec <= 15
          ? retryAfterSec * 1000
          : 1200 * Math.pow(2, attempt) + Math.round(Math.random() * 450);

      onStatus(
        `Server ${targetModel} sedang padat (${upstream.status}). Mencoba ulang otomatis (${attempt + 1}/${maxAttempts - 1}) dalam ${(
          baseDelay / 1000
        ).toFixed(1)}s...`
      );
      await sleep(baseDelay, signal);
      continue;
    }

    return {
      ok: false,
      status: lastStatus,
      message: lastErrorMessage,
    };
  }

  return { ok: false, status: lastStatus, message: lastErrorMessage };
}

function encodeWav16kHz(chunks: Float32Array[], inputSampleRate: number): Blob {
  let totalLength = 0;
  for (const c of chunks) totalLength += c.length;

  const merged = new Float32Array(totalLength);
  let pos = 0;
  for (const c of chunks) {
    merged.set(c, pos);
    pos += c.length;
  }

  const targetRate = 16000;
  const ratio = Math.max(1, inputSampleRate / targetRate);
  const outLength = Math.max(1, Math.floor(merged.length / ratio));

  const buffer = new ArrayBuffer(44 + outLength * 2);
  const view = new DataView(buffer);

  writeAscii(view, 0, 'RIFF');
  view.setUint32(4, 36 + outLength * 2, true);
  writeAscii(view, 8, 'WAVE');
  writeAscii(view, 12, 'fmt ');
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, 1, true);
  view.setUint32(24, targetRate, true);
  view.setUint32(28, targetRate * 2, true);
  view.setUint16(32, 2, true);
  view.setUint16(34, 16, true);
  writeAscii(view, 36, 'data');
  view.setUint32(40, outLength * 2, true);

  for (let i = 0; i < outLength; i++) {
    const startIdx = Math.floor(i * ratio);
    const endIdx = Math.min(merged.length, Math.floor((i + 1) * ratio));
    let sum = 0;
    let count = 0;
    for (let j = startIdx; j < endIdx; j++) {
      sum += merged[j];
      count += 1;
    }
    const sample = count > 0 ? sum / count : (merged[startIdx] ?? 0);
    const clamped = Math.max(-1, Math.min(1, sample));
    view.setInt16(44 + i * 2, clamped < 0 ? clamped * 0x8000 : clamped * 0x7fff, true);
  }

  return new Blob([buffer], { type: 'audio/wav' });
}

export class WavAudioRecorder {
  private stream: MediaStream | null = null;
  private audioCtx: AudioContext | null = null;
  private source: MediaStreamAudioSourceNode | null = null;
  private processor: ScriptProcessorNode | null = null;
  private silentGain: GainNode | null = null;
  private chunks: Float32Array[] = [];
  private sampleRate = 48000;

  async start(): Promise<void> {
    this.stream = await navigator.mediaDevices.getUserMedia({
      audio: {
        echoCancellation: true,
        noiseSuppression: true,
        autoGainControl: true,
      },
    });

    const AudioCtx =
      window.AudioContext ||
      (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;

    this.audioCtx = new AudioCtx();
    if (this.audioCtx.state === 'suspended') {
      await this.audioCtx.resume();
    }
    this.sampleRate = this.audioCtx.sampleRate || 48000;
    this.chunks = [];

    this.source = this.audioCtx.createMediaStreamSource(this.stream);
    this.processor = this.audioCtx.createScriptProcessor(4096, 1, 1);
    this.silentGain = this.audioCtx.createGain();
    this.silentGain.gain.value = 0;

    this.processor.onaudioprocess = (event) => {
      const channelData = event.inputBuffer.getChannelData(0);
      this.chunks.push(new Float32Array(channelData));
    };

    this.source.connect(this.processor);
    this.processor.connect(this.silentGain);
    this.silentGain.connect(this.audioCtx.destination);
  }

  stop(): Blob {
    const recorded = this.chunks;
    const rate = this.sampleRate;
    this.cleanup();
    if (recorded.length === 0) {
      return new Blob([], { type: 'audio/wav' });
    }
    return encodeWav16kHz(recorded, rate);
  }

  cancel(): void {
    this.cleanup();
  }

  private cleanup(): void {
    try {
      this.processor?.disconnect();
      this.source?.disconnect();
      this.silentGain?.disconnect();
    } catch {
      // Abaikan
    }
    this.stream?.getTracks().forEach((t) => t.stop());
    if (this.audioCtx && this.audioCtx.state !== 'closed') {
      void this.audioCtx.close().catch(() => {});
    }
    this.processor = null;
    this.source = null;
    this.silentGain = null;
    this.stream = null;
    this.audioCtx = null;
    this.chunks = [];
  }
}

  if (!base64) {

  }

    .split(/[\r\n,;]+/)

      data: base64,
    }),

  if (contentType.includes('text/event-stream') && res.body) {
    const reader = res.body.getReader();
    const decoder = new TextDecoder();
    let buffer = '';
    let finalText = '';
    let errorMsg = '';

    while (true) {
      const { value, done } = await reader.read();
      if (done) break;

      buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, '\n');

      let boundaryIndex: number;
      while ((boundaryIndex = buffer.indexOf('\n\n')) !== -1) {
        const rawBlock = buffer.slice(0, boundaryIndex);
        buffer = buffer.slice(boundaryIndex + 2);
        if (!rawBlock.trim() || rawBlock.startsWith(':')) continue;

        let eventName = 'message';
        const dataLines: string[] = [];
        for (const line of rawBlock.split('\n')) {
          if (line.startsWith('event:')) eventName = line.slice(6).trim();
          else if (line.startsWith('data:')) dataLines.push(line.slice(5).trimStart());
        }

        if (dataLines.length === 0) continue;
        let parsed: Record<string, unknown>;
        try {
          parsed = JSON.parse(dataLines.join('\n'));
        } catch {
          continue;
        }

        if (eventName === 'status' && typeof parsed.message === 'string') {
          params.onStatus?.(parsed.message);
        } else if (eventName === 'result' && typeof parsed.text === 'string') {
          finalText = parsed.text.trim();
        } else if (eventName === 'error' && typeof parsed.message === 'string') {
          errorMsg = parsed.message;
        }
      }
    }

    if (finalText) return { ok: true, text: finalText };
    return { ok: false, error: errorMsg || 'Transkripsi AI Audio gagal.' };
  }

  const data = (await res.json().catch(() => ({}))) as {
    ok?: boolean;
    text?: string;
    message?: string;
  };

  if (!res.ok || !data.ok || !data.text) {
    return { ok: false, error: data.message || `Gagal transkripsi (${res.status})` };
  }

  return { ok: true, text: data.text.trim() };
}

    const el = document.createElement(kind);
    el.preload = 'metadata';
    el.onloadedmetadata = () => {
      const dur = Number.isFinite(el.duration) ? el.duration : undefined;
      URL.revokeObjectURL(url);
      resolve(dur);
    };
    el.onerror = () => {
      URL.revokeObjectURL(url);
      resolve(undefined);
    };
    el.src = url;
  });
}

export function ChatComposer({
  model,
  isStreaming,
  settings,
  hasApiKeyConfigured,
  externalDraft,
  visualTarget,
  onClearVisualTarget,
  onClearExternalDraft,
  onUpdateSettings,
  onSend,
  onStop,
  onPreviewImage,
}: ChatComposerProps) {
  const [input, setInput] = useState('');
  const [attachments, setAttachments] = useState<Attachment[]>([]);
  const [isDragging, setIsDragging] = useState(false);

  const [isRecording, setIsRecording] = useState(false);
  const [isTranscribing, setIsTranscribing] = useState(false);
  const [transcribeStatus, setTranscribeStatus] = useState('');
  const [recordSeconds, setRecordSeconds] = useState(0);
  const [interimText, setInterimText] = useState('');
  const [slashIndex, setSlashIndex] = useState(0);

  const fileInputRef = useRef<HTMLInputElement | null>(null);
  const textareaRef = useRef<HTMLTextAreaElement | null>(null);
  const inputScrollRef = useRef<StudioScrollHandle | null>(null);

  const recognitionRef = useRef<SpeechRecognitionInstance | null>(null);
  const wavRecorderRef = useRef<WavAudioRecorder | null>(null);

  const adjustHeight = () => {
    const el = textareaRef.current;
    if (!el) return;
    el.style.height = 'auto';
    el.style.height = `${el.scrollHeight}px`;

    const vp = inputScrollRef.current?.getViewport();
    if (vp && el.selectionStart === el.value.length) {
      vp.scrollTop = vp.scrollHeight;
    }
  };

  useEffect(() => {
    if (externalDraft) {
      setInput(externalDraft);
      onClearExternalDraft?.();
      setTimeout(() => {
        textareaRef.current?.focus();
        adjustHeight();
      }, 20);
    }
  }, [externalDraft, onClearExternalDraft]);

  useEffect(() => {
    if (visualTarget) {
      setTimeout(() => textareaRef.current?.focus(), 30);
    }
  }, [visualTarget]);

  useEffect(() => {
    if (!isRecording) {
      setRecordSeconds(0);
      return;
    }
    const timer = window.setInterval(() => setRecordSeconds((s) => s + 1), 1000);
    return () => window.clearInterval(timer);
  }, [isRecording]);

  useEffect(() => {
    return () => {
      recognitionRef.current?.stop();
      wavRecorderRef.current?.cancel();
    };
  }, []);

  const browserSpeechSupported = useMemo(() => {
    if (typeof window === 'undefined') return false;
    const w = window as unknown as {
      SpeechRecognition?: SpeechRecognitionCtor;
      webkitSpeechRecognition?: SpeechRecognitionCtor;
    };
    return Boolean(w.SpeechRecognition || w.webkitSpeechRecognition);
  }, []);

  const aiAudioSupported =
    typeof navigator !== 'undefined' &&
    Boolean(navigator.mediaDevices?.getUserMedia) &&
    typeof AudioContext !== 'undefined';

  const useAiStt =
    settings.sttMode === 'ai' && hasApiKeyConfigured && !settings.demoMode && aiAudioSupported;

  const micAvailable = useAiStt || browserSpeechSupported;

  const slashSuggestions = useMemo(() => {
    if (!input.startsWith('/') || input.includes('\n')) return [];
    const cmd = input.split(/\s/)[0].toLowerCase();
    return SLASH_PROMPTS.filter(
      (p) => p.command.startsWith(cmd) || p.title.toLowerCase().includes(cmd.slice(1))
    );
  }, [input]);

  useEffect(() => {
    setSlashIndex(0);
  }, [slashSuggestions.length]);

  const verifyExactTokensAsync = async (item: Attachment) => {
    if (settings.demoMode) return;
    if (item.size > 3_200_000 && item.kind !== 'text') return;

    setAttachments((prev) =>
      prev.map((a) => (a.id === item.id ? { ...a, countingTokens: true } : a))
    );

    const exactCount = await fetchExactAttachmentTokens({
      model,
      attachment: item,
      apiKey: settings.apiKey,
    });

    setAttachments((prev) =>
      prev.map((a) =>
        a.id !== item.id
          ? a
          : {
              ...a,
              countingTokens: false,
              tokenCount: exactCount !== null ? exactCount : a.tokenCount,
              tokenExact: exactCount !== null,
            }
      )
    );
  };

  const ingestFiles = async (files: File[]) => {
    const loaded: Attachment[] = [];

    for (const file of files.slice(0, 6)) {
      const ext = file.name.split('.').pop()?.toLowerCase() || '';

      if (file.type.startsWith('image/')) {
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        if (!base64) continue;
        const dims = await readImageDimensions(dataUrl);
        const est = estimateAttachmentTokensLocal({
          kind: 'image',
          size: file.size,
          width: dims.width,
          height: dims.height,
        });
        loaded.push({
          id: generateId(),
          kind: 'image',
          name: file.name,
          mimeType: file.type,
          size: file.size,
          data: base64,
          previewUrl: dataUrl,
          tokenCount: est,
          tokenExact: false,
        });
      } else if (file.type.startsWith('video/') || ['mp4', 'webm', 'mov', 'm4v'].includes(ext)) {
        const durationSec = await readMediaDuration(file, 'video');
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        const est = estimateAttachmentTokensLocal({
          kind: 'video',
          size: file.size,
          durationSec,
        });
        loaded.push({
          id: generateId(),
          kind: 'video',
          name: file.name,
          mimeType: file.type || 'video/mp4',
          size: file.size,
          data: base64,
          durationSec,
          tokenCount: est,
          tokenExact: false,
        });
      } else if (file.type.startsWith('audio/') || ['mp3', 'wav', 'ogg', 'm4a'].includes(ext)) {
        const durationSec = await readMediaDuration(file, 'audio');
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        const est = estimateAttachmentTokensLocal({
          kind: 'audio',
          size: file.size,
          durationSec,
        });
        loaded.push({
          id: generateId(),
          kind: 'audio',
          name: file.name,
          mimeType: file.type || 'audio/mpeg',
          size: file.size,
          data: base64,
          durationSec,
          tokenCount: est,
          tokenExact: false,
        });
      } else if (file.type === 'application/pdf' || ext === 'pdf') {
        const dataUrl = await readFileAsDataUrl(file);
        const base64 = dataUrl.split(',')[1] || '';
        const est = estimateAttachmentTokensLocal({
          kind: 'pdf',
          size: file.size,
        });
        loaded.push({
          id: generateId(),
          kind: 'pdf',
          name: file.name,
          mimeType: 'application/pdf',
          size: file.size,
          data: base64,
          tokenCount: est,
          tokenExact: false,
        });
      } else if (file.type.startsWith('text/') || TEXT_EXTENSIONS.has(ext)) {
        const textContent = (await file.text()).slice(0, 120000);
        const est = estimateAttachmentTokensLocal({
          kind: 'text',
          size: file.size,
          textContent,
        });
        loaded.push({
          id: generateId(),
          kind: 'text',
          name: file.name,
          mimeType: file.type || 'text/plain',
          size: file.size,
          textContent,
          tokenCount: est,
          tokenExact: false,
        });
      }
    }

    if (loaded.length > 0) {
      setAttachments((prev) => [...prev, ...loaded]);
      toast.success(`${loaded.length} file ditambahkan`);
      for (const item of loaded) {
        void verifyExactTokensAsync(item);
      }
    }
  };

  const handleFileChange = async (e: ChangeEvent<HTMLInputElement>) => {
    const files = Array.from(e.target.files || []);
    if (files.length > 0) await ingestFiles(files);
    e.target.value = '';
  };

  const handlePaste = async (e: ClipboardEvent<HTMLTextAreaElement>) => {
    const files = Array.from(e.clipboardData.files || []);
    if (files.length > 0) {
      e.preventDefault();
      await ingestFiles(files);
    }
  };

  const handleDrop = async (e: DragEvent<HTMLDivElement>) => {
    e.preventDefault();
    setIsDragging(false);
    const files = Array.from(e.dataTransfer.files || []);
    if (files.length > 0) await ingestFiles(files);
  };

  const startAiAudioRecording = async () => {
    try {
      const recorder = new WavAudioRecorder();
      await recorder.start();
      wavRecorderRef.current = recorder;
      setIsRecording(true);
    } catch {
      toast.error('Izin mikrofon ditolak atau tidak tersedia');
      setIsRecording(false);
    }
  };

  const stopAndTranscribeAiAudio = async () => {
    const recorder = wavRecorderRef.current;
    wavRecorderRef.current = null;
    setIsRecording(false);
    if (!recorder) return;

    const wavBlob = recorder.stop();
    if (wavBlob.size < 1200) {
      toast.error('Rekaman suara terlalu singkat.');
      return;
    }

    setIsTranscribing(true);
    setTranscribeStatus('Mengirim audio ke Gemini AI...');
    const result = await transcribeAudioWithAI({
      audioBlob: wavBlob,
      model,
      apiKey: settings.apiKey,
      onStatus: (msg) => setTranscribeStatus(msg),
    });
    setIsTranscribing(false);
    setTranscribeStatus('');

    if (result.ok && result.text) {
      setInput((prev) => (prev ? `${prev} ${result.text}` : result.text!));
      setTimeout(() => {
        textareaRef.current?.focus();
        adjustHeight();
      }, 20);
    } else {
      toast.error(result.error || 'Gagal mentranskrip suara');
    }
  };

  const startBrowserSpeech = () => {
    const w = window as unknown as {
      SpeechRecognition?: SpeechRecognitionCtor;
      webkitSpeechRecognition?: SpeechRecognitionCtor;
    };
    const Ctor = w.SpeechRecognition || w.webkitSpeechRecognition;
    if (!Ctor) return;

    const recognition = new Ctor();
    recognition.lang = 'id-ID';
    recognition.interimResults = true;
    recognition.continuous = true;

    recognition.onresult = (event) => {
      let finalChunk = '';
      let interimChunk = '';
      for (let i = event.resultIndex; i < event.results.length; i++) {
        const res = event.results[i];
        const transcript = res[0]?.transcript || '';
        if (res.isFinal) {
          finalChunk += transcript + ' ';
        } else {
          interimChunk += transcript;
        }
      }

      setInterimText(interimChunk);

      if (finalChunk.trim()) {
        const cleaned = normalizeTechTranscript(finalChunk);
        setInput((prev) => (prev ? `${prev} ${cleaned}` : cleaned));
        setTimeout(adjustHeight, 10);
      }
    };

    recognition.onerror = () => {
      setIsRecording(false);
      setInterimText('');
    };
    recognition.onend = () => {
      setIsRecording(false);
      setInterimText('');
    };

    recognitionRef.current = recognition;
    setIsRecording(true);
    recognition.start();
  };

  const handleToggleMic = () => {
    if (!micAvailable || isTranscribing) return;

    if (isRecording) {
      if (wavRecorderRef.current) {
        void stopAndTranscribeAiAudio();
      } else {
        recognitionRef.current?.stop();
        setIsRecording(false);
        setInterimText('');
      }
      return;
    }

    if (useAiStt) {
      void startAiAudioRecording();
    } else {
      startBrowserSpeech();
    }
  };

  const handleCancelRecording = () => {
    if (wavRecorderRef.current) {
      wavRecorderRef.current.cancel();
      wavRecorderRef.current = null;
      setIsRecording(false);
    } else {
      recognitionRef.current?.stop();
      setIsRecording(false);
      setInterimText('');
    }
  };

  const handleEnhancePrompt = () => {
    const trimmed = input.trim();
    if (!trimmed) return;
    const enhanced = [
      `Tolong kerjakan permintaan berikut secara terstruktur dan menyeluruh:`,
      ``,
      `**Tujuan Utama:** ${trimmed}`,
      ``,
      `**Kriteria Output:**`,
      `- Berikan analisis arsitektur / langkah jelas`,
      `- Sertakan contoh implementasi lengkap & siap jalan`,
      `- Jelaskan edge-case dan optimasi performa`,
    ].join('\n');
    setInput(enhanced);
    setTimeout(adjustHeight, 10);
    toast.success('Prompt diperkaya');
  };

  const applySlashTemplate = (template: string) => {
    setInput(template);
    setTimeout(() => {
      textareaRef.current?.focus();
      adjustHeight();
    }, 10);
  };

  const submit = () => {
    if (isStreaming || isRecording || isTranscribing) return;
    const trimmed = input.trim();
    if (!trimmed && attachments.length === 0) return;

    onSend(trimmed, attachments);
    setInput('');
    setAttachments([]);
    if (textareaRef.current) {
      textareaRef.current.style.height = 'auto';
    }
  };

  const handleKeyDown = (e: KeyboardEvent<HTMLTextAreaElement>) => {
    if (slashSuggestions.length > 0) {
      if (e.key === 'ArrowDown') {
        e.preventDefault();
        setSlashIndex((i) => (i + 1) % slashSuggestions.length);
        return;
      }
      if (e.key === 'ArrowUp') {
        e.preventDefault();
        setSlashIndex((i) => (i - 1 + slashSuggestions.length) % slashSuggestions.length);
        return;
      }
      if (e.key === 'Enter' || e.key === 'Tab') {
        e.preventDefault();
        const chosen = slashSuggestions[slashIndex];
        if (chosen) applySlashTemplate(chosen.template);
        return;
      }
    }

    if (e.key === 'Enter' && !e.shiftKey) {
      e.preventDefault();
      submit();
    }
  };

  const activePersona = PERSONA_PRESETS.find((p) => p.id === settings.activePersonaId);
  const textDraftTokens = estimateTokens(input);
  const filesTotalTokens = attachments.reduce((acc, a) => acc + (a.tokenCount ?? 0), 0);
  const totalDraftTokens = textDraftTokens + filesTotalTokens;
  const thinkingOn = settings.includeThoughts && settings.thinkingBudget !== 0;

  return (
    <div
      className="relative mx-auto w-full max-w-3xl px-3 pb-3 pt-1.5 sm:px-4 sm:pb-4"
      onDragOver={(e) => {
        e.preventDefault();
        setIsDragging(true);
      }}
      onDragLeave={() => setIsDragging(false)}
      onDrop={handleDrop}
    >
      <AnimatePresence>
        {slashSuggestions.length > 0 && (
          <motion.div
            initial={{ opacity: 0, y: 6 }}
            animate={{ opacity: 1, y: 0 }}
            exit={{ opacity: 0, y: 6 }}
            transition={{ duration: 0.14 }}
            className="mb-2 overflow-hidden rounded-2xl border bg-popover/95 p-1.5 shadow-2xl backdrop-blur-xl"
          >
            <div className="flex items-center justify-between px-2.5 py-1 text-[10px] font-semibold uppercase tracking-wider text-muted-foreground">
              <span>Slash Prompt Templates</span>
              <span className="hidden sm:inline">↑↓ pilih • Enter sisipkan</span>
            </div>
            <StudioScroll className="max-h-[210px]" autoHide="scroll">
              <div className="space-y-0.5">
                {slashSuggestions.map((item, idx) => (
                  <button
                    key={item.command}
                    type="button"
                    onClick={() => applySlashTemplate(item.template)}
                    className={cn(
                      'flex w-full items-center justify-between rounded-xl px-2.5 py-2 text-left text-xs transition-colors',
                      idx === slashIndex ? 'bg-accent text-accent-foreground' : 'hover:bg-accent/60'
                    )}
                  >
                    <span className="flex min-w-0 items-center gap-2.5">
                      <Terminal className="h-3.5 w-3.5 shrink-0 text-primary" />
                      <span className="truncate">
                        <span className="font-semibold text-foreground">{item.title}</span>
                        <span className="ml-2 hidden text-muted-foreground sm:inline">
                          {item.description}
                        </span>
                      </span>
                    </span>
                    <code className="ml-2 shrink-0 rounded bg-muted px-1.5 py-0.5 font-mono text-[10px]">
                      {item.command}
                    </code>
                  </button>
                ))}
              </div>
            </StudioScroll>
          </motion.div>
        )}
      </AnimatePresence>

      <div
        className={cn(
          'rounded-2xl border border-border/90 bg-card/95 p-2.5 shadow-[0_12px_40px_-12px_rgba(0,0,0,0.45)] transition-all focus-within:border-primary/60 focus-within:ring-2 focus-within:ring-primary/20 sm:p-3',
          isDragging && 'border-primary ring-2 ring-primary/30'
        )}
      >
        {isDragging && (
          <div className="mb-2 rounded-xl border border-dashed border-primary bg-primary/10 py-3 text-center text-xs font-medium text-primary">
            Lepaskan gambar, video, audio, PDF, atau file kode di sini
          </div>
        )}

        {(isRecording || isTranscribing) && (
          <div className="mb-2 flex items-center justify-between gap-2 rounded-xl border border-primary/40 bg-primary/10 px-3 py-2 text-xs">
            <div className="flex min-w-0 items-center gap-2">
              {isTranscribing ? (
                <Loader2 className="h-3.5 w-3.5 shrink-0 animate-spin text-primary" />
              ) : (
                <span className="h-2.5 w-2.5 shrink-0 animate-ping rounded-full bg-rose-500" />
              )}
              <span className="truncate font-medium text-foreground">
                {isTranscribing
                  ? transcribeStatus || 'Mentranskrip suara dengan Gemini AI...'
                  : useAiStt
                    ? `Merekam WAV (${formatDuration(recordSeconds)}) — sebutkan istilah teknik apa saja`
                    : interimText
                      ? `Live: "${interimText}"`
                      : `Mendengarkan (${formatDuration(recordSeconds)})...`}
              </span>
            </div>

            {isRecording && (
              <div className="flex shrink-0 items-center gap-1.5">
                <Button
                  type="button"
                  size="sm"
                  variant="ghost"
                  onClick={handleCancelRecording}
                  className="h-6 px-2 text-[11px] text-muted-foreground hover:text-destructive"
                >
                  Batal
                </Button>
                <Button
                  type="button"
                  size="sm"
                  onClick={handleToggleMic}
                  className="h-6 gap-1 rounded-lg px-2.5 text-[11px]"
                >
                  <Check className="h-3 w-3" />
                  Selesai
                </Button>
              </div>
            )}
          </div>
        )}

        {visualTarget && (
          <div className="mb-2 flex items-center justify-between gap-2 rounded-xl border border-primary/40 bg-primary/10 px-3 py-2 text-xs">
            <div className="flex min-w-0 items-center gap-2">
              <Crosshair className="h-3.5 w-3.5 shrink-0 text-primary" />
              <span className="truncate font-mono text-[11px] font-semibold text-primary">
                &lt;{visualTarget.selector}&gt;
              </span>
              {visualTarget.textSnippet && (
                <span className="hidden truncate text-muted-foreground sm:inline">
                  &ldquo;{visualTarget.textSnippet}&rdquo;
                </span>
              )}
            </div>
            {onClearVisualTarget && (
              <button
                type="button"
                onClick={onClearVisualTarget}
                className="rounded-md p-0.5 text-muted-foreground hover:bg-background hover:text-foreground"
                aria-label="Hapus target elemen"
              >
                <X className="h-3.5 w-3.5" />
              </button>
            )}
          </div>
        )}

        {attachments.length > 0 && (
          <div className="mb-2.5 space-y-2 px-1">
            <div className="flex flex-wrap gap-2">
              {attachments.map((att) =>
                att.kind === 'image' && att.previewUrl ? (
                  <div key={att.id} className="group relative">
                    <button
                      type="button"
                      onClick={() => onPreviewImage(att.previewUrl!)}
                      className="overflow-hidden rounded-xl border"
                    >
                      <img
                        src={att.previewUrl}
                        alt={att.name}
                        className="h-16 w-20 object-cover transition-transform hover:scale-105"
                      />
                    </button>
                    <span className="pointer-events-none absolute bottom-1 left-1 inline-flex items-center gap-1 rounded bg-black/80 px-1.5 py-0.5 font-mono text-[9px] text-white">
                      {att.countingTokens && <Loader2 className="h-2.5 w-2.5 animate-spin" />}
                      {att.tokenExact ? '' : '~'}
                      {(att.tokenCount ?? 0).toLocaleString()} tok
                    </span>
                    <button
                      type="button"
                      onClick={() =>
                        setAttachments((prev) => prev.filter((item) => item.id !== att.id))
                      }
                      className="absolute -right-1.5 -top-1.5 rounded-full bg-background p-0.5 shadow ring-1 ring-border hover:bg-destructive hover:text-destructive-foreground"
                      aria-label="Hapus lampiran"
                    >
                      <X className="h-3 w-3" />
                    </button>
                  </div>
                ) : (
                  <div
                    key={att.id}
                    className="flex items-center gap-2 rounded-xl border bg-muted/60 px-2.5 py-1.5 text-xs"
                  >
                    {att.kind === 'video' ? (
                      <FileVideo className="h-3.5 w-3.5 shrink-0 text-primary" />
                    ) : att.kind === 'audio' ? (
                      <FileAudio className="h-3.5 w-3.5 shrink-0 text-primary" />
                    ) : att.kind === 'pdf' ? (
                      <FileText className="h-3.5 w-3.5 shrink-0 text-rose-400" />
                    ) : (
                      <FileCode2 className="h-3.5 w-3.5 shrink-0 text-primary" />
                    )}

                    <div className="min-w-0">
                      <div className="flex items-center gap-1.5">
                        <span className="max-w-[120px] truncate font-mono text-[11px] font-medium">
                          {att.name}
                        </span>
                        {att.durationSec && (
                          <span className="font-mono text-[10px] text-muted-foreground">
                            {formatDuration(att.durationSec)}
                          </span>
                        )}
                      </div>
                      <div className="flex items-center gap-1.5 font-mono text-[10px] text-muted-foreground">
                        <span>{formatBytes(att.size)}</span>
                        <span>•</span>
                        <span
                          className={cn(
                            'inline-flex items-center gap-1',
                            att.tokenExact ? 'text-emerald-400' : 'text-primary'
                          )}
                        >
                          {att.countingTokens && <Loader2 className="h-2.5 w-2.5 animate-spin" />}
                          {att.tokenExact ? '' : '~'}
                          {(att.tokenCount ?? 0).toLocaleString()} tok
                        </span>
                      </div>
                    </div>

                    <button
                      type="button"
                      onClick={() =>
                        setAttachments((prev) => prev.filter((item) => item.id !== att.id))
                      }
                      className="rounded p-0.5 text-muted-foreground hover:text-destructive"
                    >
                      <X className="h-3.5 w-3.5" />
                    </button>
                  </div>
                )
              )}
            </div>

            <div className="flex items-center justify-between text-[11px] text-muted-foreground">
              <span>
                {attachments.length} file terlampir (
                {attachments.every((a) => a.tokenExact)
                  ? 'Gemini exact count'
                  : 'estimasi / verifikasi'}
                )
              </span>
              <span className="font-mono font-medium text-primary">
                Total file: {filesTotalTokens.toLocaleString()} tok
              </span>
            </div>
          </div>
        )}

        <StudioScroll ref={inputScrollRef} className="max-h-[185px] w-full" autoHide="scroll">
          <Textarea
            ref={textareaRef}
            rows={1}
            value={input}
            onChange={(e) => {
              setInput(e.target.value);
              adjustHeight();
            }}
            onKeyDown={handleKeyDown}
            onPaste={handlePaste}
            placeholder={
              visualTarget
                ? `Instruksi edit untuk <${visualTarget.selector}>...`
                : `Tanya ${model}...`
            }
            className="min-h-[42px] w-full resize-none overflow-hidden border-0 bg-transparent px-2 py-2 text-[15px] leading-relaxed shadow-none focus-visible:ring-0 focus-visible:ring-offset-0"
          />
        </StudioScroll>

        <div className="mt-1.5 flex items-center justify-between gap-1.5 border-t border-border/40 pt-2">
          <div className="flex flex-wrap items-center gap-1">
            <input
              ref={fileInputRef}
              type="file"
              accept="image/*,video/*,audio/*,application/pdf,.txt,.md,.ts,.tsx,.js,.jsx,.json,.py,.css,.html,.sql,.csv,.xml,.yaml,.yml,.sh"
              multiple
              className="hidden"
              onChange={handleFileChange}
            />

            <Button
              type="button"
              variant="ghost"
              size="icon"
              onClick={() => fileInputRef.current?.click()}
              disabled={isStreaming}
              className="h-8 w-8 rounded-xl text-muted-foreground hover:text-foreground sm:w-auto sm:px-2.5"
              title="Upload Gambar, Video, Audio, PDF, atau Kode"
            >
              <Paperclip className="h-4 w-4" />
              <span className="hidden text-xs sm:inline">File</span>
            </Button>

            <Button
              type="button"
              variant={thinkingOn ? 'secondary' : 'ghost'}
              size="sm"
              disabled={isStreaming}
              onClick={() => {
                const next = !thinkingOn;
                onUpdateSettings({
                  ...settings,
                  includeThoughts: next,
                  thinkingBudget: next && settings.thinkingBudget === 0 ? -1 : settings.thinkingBudget,
                });
                toast.success(next ? 'Thinking Process: Aktif' : 'Thinking Process: Nonaktif');
              }}
              className={cn(
                'h-8 gap-1 rounded-xl px-2 text-xs',
                thinkingOn
                  ? 'border border-primary/35 bg-primary/10 text-primary hover:bg-primary/20'
                  : 'text-muted-foreground hover:text-foreground'
              )}
              title="Tampilkan proses thinking AI"
            >
              <Brain className="h-3.5 w-3.5" />
              <span className="hidden sm:inline">Think</span>
            </Button>

            <Button
              type="button"
              variant="ghost"
              size="icon"
              onClick={() => {
                setInput('/');
                textareaRef.current?.focus();
              }}
              disabled={isStreaming}
              className="h-8 w-8 rounded-xl text-muted-foreground hover:text-foreground"
              title="Slash Templates (/)"
            >
              <Slash className="h-3.5 w-3.5" />
            </Button>

            {input.trim().length > 4 && !isStreaming && (
              <Button
                type="button"
                variant="ghost"
                size="sm"
                onClick={handleEnhancePrompt}
                className="h-8 gap-1 rounded-xl px-2 text-xs text-primary hover:bg-primary/10"
                title="Perkaya struktur prompt"
              >
                <Wand2 className="h-3.5 w-3.5" />
                <span className="hidden sm:inline">Enhance</span>
              </Button>
            )}

            {micAvailable && (
              <Button
                type="button"
                variant={isRecording ? 'destructive' : 'ghost'}
                size="icon"
                onClick={handleToggleMic}
                disabled={isStreaming || isTranscribing}
                className="h-8 w-8 rounded-xl"
                title={
                  isRecording
                    ? 'Selesai & Transkrip'
                    : useAiStt
                      ? 'AI Voice Input (WAV 16kHz + Gemini AI)'
                      : 'Browser Voice Input'
                }
              >
                {isTranscribing ? (
                  <Loader2 className="h-4 w-4 animate-spin text-primary" />
                ) : (
                  <Mic
                    className={cn(
                      'h-4 w-4',
                      isRecording ? 'animate-pulse text-white' : 'text-muted-foreground'
                    )}
                  />
                )}
              </Button>
            )}

            <Popover>
              <PopoverTrigger asChild>
                <Button
                  type="button"
                  variant="ghost"
                  size="sm"
                  className="h-8 gap-1.5 rounded-xl px-2 text-xs text-muted-foreground hover:text-foreground"
                  title="Quick Tuning"
                >
                  <SlidersHorizontal className="h-3.5 w-3.5" />
                  <span className="hidden max-w-[95px] truncate sm:inline">
                    {activePersona?.name || 'Tuning'}
                  </span>
                </Button>
              </PopoverTrigger>
              <PopoverContent align="start" className="w-[calc(100vw-2rem)] max-w-xs space-y-4">
                <div className="space-y-1.5">
                  <div className="text-xs font-semibold">Quick Persona</div>
                  <div className="grid grid-cols-2 gap-1.5">
                    {PERSONA_PRESETS.map((p) => (
                      <button
                        key={p.id}
                        type="button"
                        onClick={() =>
                          onUpdateSettings({
                            ...settings,
                            activePersonaId: p.id,
                            systemInstruction: p.systemInstruction,
                            temperature: p.temperature,
                          })
                        }
                        className={cn(
                          'rounded-lg border px-2.5 py-1.5 text-left text-xs transition-colors',
                          settings.activePersonaId === p.id
                            ? 'border-primary bg-primary/10 font-medium text-primary'
                            : 'hover:bg-accent'
                        )}
                      >
                        {p.name}
                      </button>
                    ))}
                  </div>
                </div>

                <div className="space-y-2">
                  <div className="flex items-center justify-between text-xs">
                    <span className="flex items-center gap-1.5 font-medium">
                      <Mic className="h-3.5 w-3.5 text-primary" />
                      Mode Speech-to-Text
                    </span>
                  </div>
                  <div className="grid grid-cols-2 gap-1.5">
                    <button
                      type="button"
                      onClick={() => onUpdateSettings({ ...settings, sttMode: 'ai' })}
                      className={cn(
                        'rounded-lg border px-2.5 py-1.5 text-left text-[11px] transition-colors',
                        settings.sttMode === 'ai'
                          ? 'border-primary bg-primary/10 font-semibold text-primary'
                          : 'hover:bg-accent'
                      )}
                    >
                      Gemini AI (Akurat)
                    </button>
                    <button
                      type="button"
                      onClick={() => onUpdateSettings({ ...settings, sttMode: 'browser' })}
                      className={cn(
                        'rounded-lg border px-2.5 py-1.5 text-left text-[11px] transition-colors',
                        settings.sttMode === 'browser'
                          ? 'border-primary bg-primary/10 font-semibold text-primary'
                          : 'hover:bg-accent'
                      )}
                    >
                      Browser Live
                    </button>
                  </div>
                </div>

                <div className="space-y-2">
                  <div className="flex items-center justify-between text-xs">
                    <span className="flex items-center gap-1.5 font-medium">
                      <Brain className="h-3.5 w-3.5 text-primary" />
                      Thinking Budget
                    </span>
                    <Switch
                      checked={thinkingOn}
                      onCheckedChange={(checked) =>
                        onUpdateSettings({
                          ...settings,
                          includeThoughts: checked,
                          thinkingBudget: checked && settings.thinkingBudget === 0 ? -1 : settings.thinkingBudget,
                        })
                      }
                    />
                  </div>
                  {thinkingOn && (
                    <div className="grid grid-cols-5 gap-1">
                      {THINKING_BUDGET_OPTIONS.map((opt) => (
                        <button
                          key={opt.value}
                          type="button"
                          onClick={() =>
                            onUpdateSettings({
                              ...settings,
                              includeThoughts: true,
                              thinkingBudget: opt.value,
                            })
                          }
                          className={cn(
                            'rounded-md border py-1 font-mono text-[10px] transition-colors',
                            settings.thinkingBudget === opt.value
                              ? 'border-primary bg-primary/15 font-semibold text-primary'
                              : 'hover:bg-accent'
                          )}
                        >
                          {opt.label}
                        </button>
                      ))}
                    </div>
                  )}
                </div>

                <div className="space-y-2">
                  <div className="flex justify-between text-xs">
                    <span className="font-medium">Temperature</span>
                    <span className="font-mono text-primary">{settings.temperature.toFixed(1)}</span>
                  </div>
                  <Slider
                    min={0}
                    max={2}
                    step={0.1}
                    value={[settings.temperature]}
                    onValueChange={([val]) => onUpdateSettings({ ...settings, temperature: val })}
                  />
                </div>

                <div className="flex items-center justify-between rounded-lg border bg-muted/40 px-3 py-2">
                  <span className="text-xs font-medium">Demo SSE Mode</span>
                  <Switch
                    checked={settings.demoMode}
                    onCheckedChange={(checked) =>
                      onUpdateSettings({ ...settings, demoMode: checked })
                    }
                  />
                </div>
              </PopoverContent>
            </Popover>
          </div>

          <div className="flex items-center gap-2">
            {totalDraftTokens > 0 && (
              <span className="font-mono text-[10px] text-muted-foreground">
                {totalDraftTokens.toLocaleString()} tok
              </span>
            )}

            {isStreaming ? (
              <Button
                type="button"
                size="sm"
                variant="destructive"
                onClick={onStop}
                className="h-8 gap-1.5 rounded-xl px-3"
              >
                <Square className="h-3.5 w-3.5 fill-current" />
                <span>Stop</span>
              </Button>
            ) : (
              <Button
                type="button"
                size="icon"
                onClick={submit}
                disabled={(!input.trim() && attachments.length === 0) || isRecording || isTranscribing}
                className="h-8 w-8 rounded-xl shadow-md"
                aria-label="Kirim pesan"
              >
                <ArrowUp className="h-4 w-4" />
              </Button>
            )}
          </div>
        </div>
      </div>

      <div className="mt-1.5 flex items-center justify-between px-1 text-[10px] text-muted-foreground sm:text-[11px]">
        <span className="inline-flex items-center gap-1 truncate">
          <Sparkles className="h-3 w-3 shrink-0 text-primary" />
          <span className="truncate font-mono">{model}</span>
        </span>
        <span className="hidden sm:inline">Ctrl+K Palette • Ctrl+\ Zen Mode</span>
        <span className="sm:hidden">Ketik / untuk template</span>
      </div>
    </div>
  );
}
